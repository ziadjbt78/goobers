    1| /**
    2|  * Agent — the per-creature driver. v8.
    3|  *
    4|  * The Agent decides ONE thing per frame: the DESIRED velocity that would take
    5|  * the creature to its brain's goal. It never writes a position, never writes a
    6|  * heading, never writes a bone. Everything visible comes from the motion layer:
    7|  *
    8|  *   brain -> desired velocity -> motion.preAnimate -> HeroAnimator.update
    9|  *                             -> motion.postAnimate  -> rig
   10|  *
   11|  * Position, heading, speed, the gait cycle, the juice, the face, the
   12|  * choreography and the dangling pick-up all live in CreatureMotion.
   13|  */

   42| const BODY_RATE = 14;

   45| /** animator action -> the Choreo action the motion layer plays for it. */
   46| const ACTION_PLAY: Record<AnimAction, string | null> = {
   47|   none: null, eat: 'feedEat', chomp: 'feedEat', sniff: 'feedEat',
   48|   play: 'celebrate', greet: 'callArrive', startle: 'callNotice',
   49|   wave: 'inspectWave', dig: 'yuck', pet: 'petRelease',
   50|   sleep: null, carry: null, splash: null, swim: null, fly: null,

   54| export class Agent implements HashEntry {

   64|   name: string;

   66|   goal: THREE.Vector3 | null = null;
   67|   /** turn to face a point WITHOUT walking — Locomotion's heading spring turns it */
   68|   faceTarget: { x: number; z: number } | null = null;
   69|   throttle = 0;
   70|   carried = false;
   71|   bulk: number;
   72|

   82|   /** the Sim resolves motion-layer effects through here */

   86|   readonly feet: Foot[] = [];

   99|   static preMs = 0;
  100|   static animMs = 0;
  101|   static postMs = 0;

  103|   bodyRaw = 0;

  109|   constructor(world: World, template: HeroTemplate, dna: HeroDNA, handle: HeroHandle, seed: number) {
  110|     this.id = Agent.NEXT_ID++;
  111|     this.world = world;
  112|     this.dna = dna;
  113|     this.template = template;
  114|     this.handle = handle;
  115|     this.rng = mulberry32(seed);
  116|     this.name = `${template.name} ${String(this.id).padStart(2, '0')}`;
  117|     this.bulk = template.height * dna.scale;
  118| 
  119|     // The animator runs in EXTERNAL-phase mode: the motion layer supplies the
  120|     // gait cycle every frame, so there is exactly one clock.
  121|     this.hero = new HeroAnimator(template, handle);
  122|     const height = template.height * dna.scale;
  123|     const rig = createRig(template, handle, this.hero, {
  124|       emote: (g, life, punch) => this.say(g, life, punch),
  125|       vfx: (kind, x, y, z, scale) => this.onVfx?.(kind, x, y, z, scale),
  126|       ground: (x, z) => heightAt(x, z),
  127|     });
  128|     this.rig = rig;
  129|     this.restBodyY = rig.body.position.y;
  130|     // reach from the LIVE rig, not the template offsets
  131|     const f0 = template.feet[0];
  132|     const liveReach = f0
  133|       ? handle.bones[f0.chain[1]].position.length() + handle.bones[f0.chain[2]].position.length()
  134|       : 0;
  135|     this.motion = new CreatureMotion(rig, locoConfigFor(template, height, liveReach), new THREE.Vector3(), 0);
  136|     this.legRest = rig.legs.map((b) => b.quaternion.clone());
  137| 
  138|     for (let i = 0; i < template.feet.length; i++) {
  139|       const f = template.feet[i];
  140|       this.feet.push({
  141|         pos: f.plant.clone(), from: f.plant.clone(), to: f.plant.clone(),
  142|         swing: 0, swinging: false, phase: f.phase, step: f.step, lift: f.lift, wasUp: false,
  143|       });
  144|       this.prevSwing.push(false);
  145|     }
  146|   }
  147| 
  148|   /** Personality trait, 0..1. Defensive so a DNA without the field still runs. */
  149|   trait(k: 'bold' | 'curious' | 'sociable' | 'playful' | 'lazy'): number {
  150|     const p = (this.dna as unknown as { personality?: Record<string, number> }).personality;
  151|     return p && typeof p[k] === 'number' ? p[k] : 0.5;
  152|   }
  153| 
  154|   /** live views — the motion layer owns both vectors */
  155|   get pos(): THREE.Vector3 { return this.motion.loco.pos; }
  156|   get heading(): number { return this.motion.loco.heading; }
  157|   set heading(v: number) { this.motion.loco.heading = v; }
  158|   get speed(): number { return this.motion.loco.speed; }
  159|   get moving(): boolean { return this.motion.loco.speed > 0.06; }
  160|   get topSpeed(): number { return this.motion.loco.cfg.runSpeed * (0.78 + this.dna.speed * 0.34); }
  161| 
  162|   /** Drop the creature at a world point. */
  163|   place(x: number, z: number, heading = this.rng() * Math.PI * 2): void {
  164|     let px = x;
  165|     let pz = z;
  166|     if (!isDry(px, pz)) {
  167|       const s = this.world.randomSpot(this.rng);
  168|       px = s.x;
  169|       pz = s.z;
  170|     }
  171|     const rr = Math.hypot(px, pz);
  172|     if (rr > WALK_R) { px *= WALK_R / rr; pz *= WALK_R / rr; }
  173|     const loco = this.motion.loco;
  174|     loco.pos.set(px, heightAt(px, pz), pz);
  175|     loco.heading = heading;
  176|     loco.speed = 0;
  177|     loco.phase = this.rng();
  178|     this.motion.grabPoint = null;
  179|     this.rig.root.position.copy(loco.pos);
  180|     this.rig.root.rotation.set(0, heading, 0);
  181|   }
  182| 
  183|   goTo(p: THREE.Vector3 | { x: number; y: number; z: number } | null, throttle = 1): void {
  184|     this.goal = p ? new THREE.Vector3(p.x, p.y, p.z) : null;
  185|     this.throttle = p ? throttle : 0;
  186|   }
  187| 
  188|   stop(): void { this.goal = null; this.throttle = 0; this.faceTarget = null; }
  189| 
  190|   say(g: Glyph, life = 1.6, punch = 1): void {
  191|     this.emoteGlyph = g;
  192|     this.emoteT = life;
  193|     this.emoteMax = life;
  194|     this.emotePunch = punch;
  195|   }

  197|   /** Begin a choreographed action. Restarting the same action is a no-op. */
  198|   begin(id: AnimAction): void {
  199|     if (this.action === id) return;
  200|     this.action = id;
  201|     this.actionAge = 0;
  202|     const act = ACTION_PLAY[id];
  203|     if (act) this.motion.play(act);
  204|     else this.motion.interrupt();
  205|   }
  206| 
  207|   /** A happy hop — one celebrate cycle, restarted so three in a row read as three. */
  208|   cheer(): void { this.motion.interrupt('celebrate'); }

  210|   /** v9 `G` key: 0 none, 1 pitch+, 2 roll+, 3 left limb forward, 4 wave+. */
  211|   debugPose = 0;
  212| 
  213|   /**
  214|    * The G-key sign check. Each pose drives ONE channel through the adapter's
  215|    * `axes` signs, so the rendered frame says whether that sign moves the way
  216|    * its name claims. Applied on top of the real motion layer, after
  217|    * postAnimate, so nothing else is disturbed.
  218|    */
  219|   private applyDebugPose(): void {
  220|     const p = this.debugPose;
  221|     if (!p) return;
  222|     const ax = this.rig.axes;
  223|     const q = new THREE.Quaternion();
  224|     const AMP = 0.5;
  225|     if (p === 1) {
  226|       // rotation about the BODY bone's own pivot: the head swings relative to
  227|       // that pivot, which is what the probe measures in the body's frame
  228|       q.setFromAxisAngle(new THREE.Vector3(1, 0, 0), ax.pitch * AMP);
  229|       this.rig.body.quaternion.multiply(q);
  230|     } else if (p === 2) {
  231|       q.setFromAxisAngle(new THREE.Vector3(0, 0, 1), ax.roll * AMP);
  232|       this.rig.body.quaternion.multiply(q);
  233|     } else if (p === 3) {
  234|       const leg = this.rig.legs[0];
  235|       if (leg) leg.quaternion.multiply(this.axisIn(leg, ax.limb === 'x' ? 1 : 0, ax.limb === 'z' ? 1 : 0, AMP));
  236|     } else if (p === 4) {
  237|       const w = this.rig.arms[0] ?? this.rig.legs[0];
  238|       if (w) w.quaternion.multiply(this.axisIn(w, 1, 0, ax.wave * -AMP));
  239|     }
  240|   }
