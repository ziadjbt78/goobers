    1| /**
    2|  * Agent — the per-creature driver. v8.
    3|  *
    4|  * The Agent decides ONE thing per frame: the DESIRED velocity that would take
    5|  * the creature to its brain's goal. It never writes a position, never writes a
    6|  * heading, never writes a bone. Everything visible comes from the motion layer:
    7|  *
    8|  *   brain -> desired velocity -> motion.preAnimate -> HeroAnimator.update
    9|  *                             -> motion.postAnimate  -> rig
   10|  *
   11|  * Position, heading, speed, the gait cycle, the juice, the face, the
   12|  * choreography and the dangling pick-up all live in CreatureMotion.
   13|  */

   91|   constructor(world: World, template: HeroTemplate, dna: HeroDNA, handle: HeroHandle, seed: number) {
   92|     this.id = Agent.NEXT_ID++;
   93|     this.world = world;
   94|     this.dna = dna;
   95|     this.template = template;
   96|     this.handle = handle;
   97|     this.rng = mulberry32(seed);
   98|     this.name = `${template.name} ${String(this.id).padStart(2, '0')}`;
   99|     this.bulk = template.height * dna.scale;
  100| 
  101|     // The animator runs in EXTERNAL-phase mode: the motion layer supplies the
  102|     // gait cycle every frame, so there is exactly one clock.
  103|     this.hero = new HeroAnimator(template, handle);
  104|     const height = template.height * dna.scale;
  105|     const rig = createRig(template, handle, this.hero, {
  106|       emote: (g, life, punch) => this.say(g, life, punch),
  107|       vfx: (kind, x, y, z, scale) => this.onVfx?.(kind, x, y, z, scale),
  108|       ground: (x, z) => heightAt(x, z),
  109|     });
  110|     this.rig = rig;
  111|     this.restBodyY = rig.body.position.y;
  112|     // reach from the LIVE rig, not the template offsets
  113|     const f0 = template.feet[0];
  114|     const liveReach = f0
  115|       ? handle.bones[f0.chain[1]].position.length() + handle.bones[f0.chain[2]].position.length()
  116|       : 0;
  117|     this.motion = new CreatureMotion(rig, locoConfigFor(template, height, liveReach), new THREE.Vector3(), 0);
  118| 
  119|     for (let i = 0; i < template.feet.length; i++) {
  120|       const f = template.feet[i];
  121|       this.feet.push({
  122|         pos: f.plant.clone(), from: f.plant.clone(), to: f.plant.clone(),
  123|         swing: 0, swinging: false, phase: f.phase, step: f.step, lift: f.lift, wasUp: false,
  124|       });
  125|       this.prevSwing.push(false);
  126|     }
  127|   }

  129|   /** Personality trait, 0..1. Defensive so a DNA without the field still runs. */
  130|   trait(k: 'bold' | 'curious' | 'sociable' | 'playful' | 'lazy'): number {
  131|     const p = (this.dna as unknown as { personality?: Record<string, number> }).personality;
  132|     return p && typeof p[k] === 'number' ? p[k] : 0.5;

  135|   /** live views — the motion layer owns both vectors */
  136|   get pos(): THREE.Vector3 { return this.motion.loco.pos; }
  137|   get heading(): number { return this.motion.loco.heading; }
  138|   set heading(v: number) { this.motion.loco.heading = v; }
  139|   get speed(): number { return this.motion.loco.speed; }
  140|   get moving(): boolean { return this.motion.loco.speed > 0.06; }
  141|   get topSpeed(): number { return this.motion.loco.cfg.runSpeed * (0.78 + this.dna.speed * 0.34); }
  142| 
  143|   /** Drop the creature at a world point. */
  144|   place(x: number, z: number, heading = this.rng() * Math.PI * 2): void {
  145|     let px = x;
  146|     let pz = z;
  147|     if (!isDry(px, pz)) {
  148|       const s = this.world.randomSpot(this.rng);
  149|       px = s.x;
  150|       pz = s.z;
  151|     }
  152|     const rr = Math.hypot(px, pz);
  153|     if (rr > WALK_R) { px *= WALK_R / rr; pz *= WALK_R / rr; }
  154|     const loco = this.motion.loco;
  155|     loco.pos.set(px, heightAt(px, pz), pz);
  156|     loco.heading = heading;
  157|     loco.speed = 0;
  158|     loco.phase = this.rng();
  159|     this.motion.grabPoint = null;
  160|     this.rig.root.position.copy(loco.pos);
  161|     this.rig.root.rotation.set(0, heading, 0);
  162|   }

  164|   goTo(p: THREE.Vector3 | { x: number; y: number; z: number } | null, throttle = 1): void {

  166|     this.throttle = p ? throttle : 0;

  178|   /** Begin a choreographed action. Restarting the same action is a no-op. */
  179|   begin(id: AnimAction): void {
  180|     if (this.action === id) return;
  181|     this.action = id;
  182|     this.actionAge = 0;
  183|     const act = ACTION_PLAY[id];
  184|     if (act) this.motion.play(act);
  185|     else this.motion.interrupt();
  186|   }

  188|   /** A happy hop — one celebrate cycle, restarted so three in a row read as three. */

  191|   /** v9 `G` key: 0 none, 1 pitch+, 2 roll+, 3 left limb forward, 4 wave+. */
  192|   debugPose = 0;
  193| 
  194|   /**
  195|    * The G-key sign check. Each pose drives ONE channel through the adapter's
  196|    * `axes` signs, so the rendered frame says whether that sign moves the way
  197|    * its name claims. Applied on top of the real motion layer, after
  198|    * postAnimate, so nothing else is disturbed.
  199|    */
  200|   private applyDebugPose(): void {
  201|     const p = this.debugPose;
  202|     if (!p) return;
  203|     const ax = this.rig.axes;
  204|     const q = new THREE.Quaternion();
  205|     const AMP = 0.5;
  206|     if (p === 1) {
  207|       // rotation about the BODY bone's own pivot: the head swings relative to
  208|       // that pivot, which is what the probe measures in the body's frame
  209|       q.setFromAxisAngle(new THREE.Vector3(1, 0, 0), ax.pitch * AMP);
  210|       this.rig.body.quaternion.multiply(q);
  211|     } else if (p === 2) {
  212|       q.setFromAxisAngle(new THREE.Vector3(0, 0, 1), ax.roll * AMP);
  213|       this.rig.body.quaternion.multiply(q);
  214|     } else if (p === 3) {
  215|       const leg = this.rig.legs[0];
  216|       if (leg) leg.quaternion.multiply(this.axisIn(leg, ax.limb === 'x' ? 1 : 0, ax.limb === 'z' ? 1 : 0, AMP));
  217|     } else if (p === 4) {
  218|       const w = this.rig.arms[0] ?? this.rig.legs[0];
  219|       if (w) w.quaternion.multiply(this.axisIn(w, 1, 0, ax.wave * -AMP));
  220|     }
  221|   }

  223|   /** v9 diagnostic: strip the pivot's juice so the slip it causes can be
  224|    *  separated from slip caused by a wrong stride. */
  225|   noPivotJuice = false;
  226|   /** frames (per probe) where a PLANTED foot was past 95% of leg reach */
  227|   overstretchFrames = 0;

  230|   /**
  231|    * Build a rotation about a WORLD axis, expressed in `bone`'s parent space, so
  232|    * a test pose swings the limb the same way no matter how that bone's rest
  233|    * frame happens to be oriented.
  234|    */
  235|   private axisIn(bone: THREE.Object3D, x: number, z: number, ang: number): THREE.Quaternion {
  236|     const axisWorld = new THREE.Vector3(x, 0, z);
  237|     if (axisWorld.lengthSq() < 1e-9) axisWorld.set(1, 0, 0);
  238|     axisWorld.normalize();
  239|     const pw = (bone.parent ?? bone).getWorldQuaternion(new THREE.Quaternion());
  240|     const axisLocal = axisWorld.applyQuaternion(pw.clone().invert()).normalize();
  241|     return new THREE.Quaternion().setFromAxisAngle(axisLocal, ang);
  242|   }

  247| 
  248|     // ---- the ONE thing this class decides: desired velocity ---------------
  249|     this.desired.set(0, 0, 0);
