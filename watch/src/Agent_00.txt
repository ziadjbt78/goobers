    1| /**
    2|  * Agent — the per-creature driver. v8.
    3|  *
    4|  * The Agent decides ONE thing per frame: the DESIRED velocity that would take
    5|  * the creature to its brain's goal. It never writes a position, never writes a
    6|  * heading, never writes a bone. Everything visible comes from the motion layer:
    7|  *
    8|  *   brain -> desired velocity -> motion.preAnimate -> HeroAnimator.update
    9|  *                             -> motion.postAnimate  -> rig
   10|  *
   11|  * Position, heading, speed, the gait cycle, the juice, the face, the
   12|  * choreography and the dangling pick-up all live in CreatureMotion.
   13|  */

   97|   constructor(world: World, template: HeroTemplate, dna: HeroDNA, handle: HeroHandle, seed: number) {
   98|     this.id = Agent.NEXT_ID++;
   99|     this.world = world;
  100|     this.dna = dna;
  101|     this.template = template;
  102|     this.handle = handle;
  103|     this.rng = mulberry32(seed);
  104|     this.name = `${template.name} ${String(this.id).padStart(2, '0')}`;
  105|     this.bulk = template.height * dna.scale;
  106| 
  107|     // The animator runs in EXTERNAL-phase mode: the motion layer supplies the
  108|     // gait cycle every frame, so there is exactly one clock.
  109|     this.hero = new HeroAnimator(template, handle);
  110|     const height = template.height * dna.scale;
  111|     const rig = createRig(template, handle, this.hero, {
  112|       emote: (g, life, punch) => this.say(g, life, punch),
  113|       vfx: (kind, x, y, z, scale) => this.onVfx?.(kind, x, y, z, scale),
  114|       ground: (x, z) => heightAt(x, z),
  115|     });
  116|     this.rig = rig;
  117|     this.restBodyY = rig.body.position.y;
  118|     // reach from the LIVE rig, not the template offsets
  119|     const f0 = template.feet[0];
  120|     const liveReach = f0
  121|       ? handle.bones[f0.chain[1]].position.length() + handle.bones[f0.chain[2]].position.length()
  122|       : 0;
  123|     this.motion = new CreatureMotion(rig, locoConfigFor(template, height, liveReach), new THREE.Vector3(), 0);
  124|     this.legRest = rig.legs.map((b) => b.quaternion.clone());
  125| 
  126|     for (let i = 0; i < template.feet.length; i++) {
  127|       const f = template.feet[i];
  128|       this.feet.push({
  129|         pos: f.plant.clone(), from: f.plant.clone(), to: f.plant.clone(),
  130|         swing: 0, swinging: false, phase: f.phase, step: f.step, lift: f.lift, wasUp: false,
  131|       });
  132|       this.prevSwing.push(false);
  133|     }
  134|   }

  136|   /** Personality trait, 0..1. Defensive so a DNA without the field still runs. */

  142|   /** live views — the motion layer owns both vectors */

  144|   get heading(): number { return this.motion.loco.heading; }
  145|   set heading(v: number) { this.motion.loco.heading = v; }
  146|   get speed(): number { return this.motion.loco.speed; }
  147|   get moving(): boolean { return this.motion.loco.speed > 0.06; }
  148|   get topSpeed(): number { return this.motion.loco.cfg.runSpeed * (0.78 + this.dna.speed * 0.34); }
  149| 
  150|   /** Drop the creature at a world point. */
  151|   place(x: number, z: number, heading = this.rng() * Math.PI * 2): void {
  152|     let px = x;
  153|     let pz = z;
  154|     if (!isDry(px, pz)) {
  155|       const s = this.world.randomSpot(this.rng);
  156|       px = s.x;
  157|       pz = s.z;
  158|     }
  159|     const rr = Math.hypot(px, pz);
  160|     if (rr > WALK_R) { px *= WALK_R / rr; pz *= WALK_R / rr; }
  161|     const loco = this.motion.loco;
  162|     loco.pos.set(px, heightAt(px, pz), pz);
  163|     loco.heading = heading;
  164|     loco.speed = 0;
  165|     loco.phase = this.rng();
  166|     this.motion.grabPoint = null;
  167|     this.rig.root.position.copy(loco.pos);
  168|     this.rig.root.rotation.set(0, heading, 0);
  169|   }

  171|   goTo(p: THREE.Vector3 | { x: number; y: number; z: number } | null, throttle = 1): void {

  185|   /** Begin a choreographed action. Restarting the same action is a no-op. */
  186|   begin(id: AnimAction): void {
  187|     if (this.action === id) return;
  188|     this.action = id;
  189|     this.actionAge = 0;
  190|     const act = ACTION_PLAY[id];
  191|     if (act) this.motion.play(act);
  192|     else this.motion.interrupt();
  193|   }

  198|   /** v9 `G` key: 0 none, 1 pitch+, 2 roll+, 3 left limb forward, 4 wave+. */
  199|   debugPose = 0;

  201|   /**
  202|    * The G-key sign check. Each pose drives ONE channel through the adapter's
  203|    * `axes` signs, so the rendered frame says whether that sign moves the way
  204|    * its name claims. Applied on top of the real motion layer, after
  205|    * postAnimate, so nothing else is disturbed.
  206|    */
  207|   private applyDebugPose(): void {
  208|     const p = this.debugPose;
  209|     if (!p) return;
  210|     const ax = this.rig.axes;
  211|     const q = new THREE.Quaternion();
  212|     const AMP = 0.5;
  213|     if (p === 1) {
  214|       // rotation about the BODY bone's own pivot: the head swings relative to
  215|       // that pivot, which is what the probe measures in the body's frame
  216|       q.setFromAxisAngle(new THREE.Vector3(1, 0, 0), ax.pitch * AMP);
  217|       this.rig.body.quaternion.multiply(q);
  218|     } else if (p === 2) {
  219|       q.setFromAxisAngle(new THREE.Vector3(0, 0, 1), ax.roll * AMP);
  220|       this.rig.body.quaternion.multiply(q);
  221|     } else if (p === 3) {
  222|       const leg = this.rig.legs[0];
  223|       if (leg) leg.quaternion.multiply(this.axisIn(leg, ax.limb === 'x' ? 1 : 0, ax.limb === 'z' ? 1 : 0, AMP));
  224|     } else if (p === 4) {
  225|       const w = this.rig.arms[0] ?? this.rig.legs[0];
  226|       if (w) w.quaternion.multiply(this.axisIn(w, 1, 0, ax.wave * -AMP));
  227|     }
  228|   }

  250|     if (axisWorld.lengthSq() < 1e-9) axisWorld.set(1, 0, 0);
