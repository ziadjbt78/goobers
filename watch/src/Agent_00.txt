    1| /**
    2|  * Agent — the per-creature driver. v8.
    3|  *
    4|  * The Agent decides ONE thing per frame: the DESIRED velocity that would take
    5|  * the creature to its brain's goal. It never writes a position, never writes a
    6|  * heading, never writes a bone. Everything visible comes from the motion layer:
    7|  *
    8|  *   brain -> desired velocity -> motion.preAnimate -> HeroAnimator.update
    9|  *                             -> motion.postAnimate  -> rig
   10|  *
   11|  * Position, heading, speed, the gait cycle, the juice, the face, the
   12|  * choreography and the dangling pick-up all live in CreatureMotion.
   13|  */

   95|   constructor(world: World, template: HeroTemplate, dna: HeroDNA, handle: HeroHandle, seed: number) {
   96|     this.id = Agent.NEXT_ID++;
   97|     this.world = world;
   98|     this.dna = dna;
   99|     this.template = template;
  100|     this.handle = handle;
  101|     this.rng = mulberry32(seed);
  102|     this.name = `${template.name} ${String(this.id).padStart(2, '0')}`;
  103|     this.bulk = template.height * dna.scale;
  104| 
  105|     // The animator runs in EXTERNAL-phase mode: the motion layer supplies the
  106|     // gait cycle every frame, so there is exactly one clock.
  107|     this.hero = new HeroAnimator(template, handle);
  108|     const height = template.height * dna.scale;
  109|     const rig = createRig(template, handle, this.hero, {
  110|       emote: (g, life, punch) => this.say(g, life, punch),
  111|       vfx: (kind, x, y, z, scale) => this.onVfx?.(kind, x, y, z, scale),
  112|       ground: (x, z) => heightAt(x, z),
  113|     });
  114|     this.rig = rig;
  115|     this.restBodyY = rig.body.position.y;
  116|     // reach from the LIVE rig, not the template offsets
  117|     const f0 = template.feet[0];
  118|     const liveReach = f0
  119|       ? handle.bones[f0.chain[1]].position.length() + handle.bones[f0.chain[2]].position.length()
  120|       : 0;
  121|     this.motion = new CreatureMotion(rig, locoConfigFor(template, height, liveReach), new THREE.Vector3(), 0);
  122|     this.legRest = rig.legs.map((b) => b.quaternion.clone());
  123| 
  124|     for (let i = 0; i < template.feet.length; i++) {
  125|       const f = template.feet[i];
  126|       this.feet.push({
  127|         pos: f.plant.clone(), from: f.plant.clone(), to: f.plant.clone(),
  128|         swing: 0, swinging: false, phase: f.phase, step: f.step, lift: f.lift, wasUp: false,
  129|       });
  130|       this.prevSwing.push(false);
  131|     }
  132|   }

  148|   /** Drop the creature at a world point. */
  149|   place(x: number, z: number, heading = this.rng() * Math.PI * 2): void {
  150|     let px = x;
  151|     let pz = z;
  152|     if (!isDry(px, pz)) {
  153|       const s = this.world.randomSpot(this.rng);
  154|       px = s.x;
  155|       pz = s.z;
  156|     }
  157|     const rr = Math.hypot(px, pz);
  158|     if (rr > WALK_R) { px *= WALK_R / rr; pz *= WALK_R / rr; }
  159|     const loco = this.motion.loco;
  160|     loco.pos.set(px, heightAt(px, pz), pz);
  161|     loco.heading = heading;
  162|     loco.speed = 0;
  163|     loco.phase = this.rng();
  164|     this.motion.grabPoint = null;
  165|     this.rig.root.position.copy(loco.pos);
  166|     this.rig.root.rotation.set(0, heading, 0);
  167|   }

  183|   /** Begin a choreographed action. Restarting the same action is a no-op. */
  184|   begin(id: AnimAction): void {

  196|   /** v9 `G` key: 0 none, 1 pitch+, 2 roll+, 3 left limb forward, 4 wave+. */
  197|   debugPose = 0;

  199|   /**
  200|    * The G-key sign check. Each pose drives ONE channel through the adapter's
  201|    * `axes` signs, so the rendered frame says whether that sign moves the way
  202|    * its name claims. Applied on top of the real motion layer, after
  203|    * postAnimate, so nothing else is disturbed.
  204|    */
  205|   private applyDebugPose(): void {
  206|     const p = this.debugPose;
  207|     if (!p) return;
  208|     const ax = this.rig.axes;
  209|     const q = new THREE.Quaternion();
  210|     const AMP = 0.5;
  211|     if (p === 1) {
  212|       // rotation about the BODY bone's own pivot: the head swings relative to
  213|       // that pivot, which is what the probe measures in the body's frame
  214|       q.setFromAxisAngle(new THREE.Vector3(1, 0, 0), ax.pitch * AMP);
  215|       this.rig.body.quaternion.multiply(q);
  216|     } else if (p === 2) {
  217|       q.setFromAxisAngle(new THREE.Vector3(0, 0, 1), ax.roll * AMP);
  218|       this.rig.body.quaternion.multiply(q);
  219|     } else if (p === 3) {
  220|       const leg = this.rig.legs[0];
  221|       if (leg) leg.quaternion.multiply(this.axisIn(leg, ax.limb === 'x' ? 1 : 0, ax.limb === 'z' ? 1 : 0, AMP));
  222|     } else if (p === 4) {
  223|       const w = this.rig.arms[0] ?? this.rig.legs[0];
  224|       if (w) w.quaternion.multiply(this.axisIn(w, 1, 0, ax.wave * -AMP));
  225|     }
  226|   }

  235|   /**
  236|    * Build a rotation about a WORLD axis, expressed in `bone`'s parent space, so
  237|    * a test pose swings the limb the same way no matter how that bone's rest
  238|    * frame happens to be oriented.
  239|    */
  240|   private axisIn(bone: THREE.Object3D, x: number, z: number, ang: number): THREE.Quaternion {
  241|     const axisWorld = new THREE.Vector3(x, 0, z);
  242|     if (axisWorld.lengthSq() < 1e-9) axisWorld.set(1, 0, 0);
  243|     axisWorld.normalize();
  244|     const pw = (bone.parent ?? bone).getWorldQuaternion(new THREE.Quaternion());
  245|     const axisLocal = axisWorld.applyQuaternion(pw.clone().invert()).normalize();
  246|     return new THREE.Quaternion().setFromAxisAngle(axisLocal, ang);
  247|   }
