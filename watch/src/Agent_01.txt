  251|     axisWorld.normalize();
  252|     const pw = (bone.parent ?? bone).getWorldQuaternion(new THREE.Quaternion());
  253|     const axisLocal = axisWorld.applyQuaternion(pw.clone().invert()).normalize();
  254|     return new THREE.Quaternion().setFromAxisAngle(axisLocal, ang);
  255|   }
  256| 
  257|   update(dt: number): void {
  258|     if (!this.alive) return;
  259|     this.actionAge += dt;
  260| 
  261|     // ---- the ONE thing this class decides: desired velocity ---------------
  262|     this.desired.set(0, 0, 0);
  263|     if (!this.carried) {
  264|       if (this.goal) {
  265|         const d = Math.hypot(this.goal.x - this.pos.x, this.goal.z - this.pos.z);
  266|         if (d < ARRIVE) this.goal = null;
  267|         else arrive(this.pos, this.goal, this.topSpeed * Math.max(0, this.throttle), SLOW_RADIUS, this.desired);
  268|       }
  269|       if (this.faceTarget) {
  270|         const dx = this.faceTarget.x - this.pos.x;
  271|         const dz = this.faceTarget.z - this.pos.z;
  272|         if (Math.hypot(dx, dz) < 0.04) this.faceTarget = null;
  273|         else {
  274|           // turn in place: aim a slow creep at the target, and let the
  275|           // heading spring in Locomotion do the turning
  276|           const a = Math.atan2(dx, dz);
  277|           this.desired.set(Math.sin(a) * 0.30, 0, Math.cos(a) * 0.30);
  278|         }
  279|       }
  280|     }
  281| 
  282|     // v12: personal space. Creatures never walk into each other's bodies; an
  283|     // overlapping neighbour adds a gentle push away to the desired velocity.
  284|     if (!this.carried) {
  285|       for (const o of this.world.nearby(this.pos, this.bulk * 2, this.id)) {
  286|         const dx = this.pos.x - o.pos.x;
  287|         const dz = this.pos.z - o.pos.z;
  288|         const d = Math.hypot(dx, dz) || 1e-4;
  289|         const minD = (this.bulk + o.bulk) * 0.42;
  290|         if (d < minD) {
  291|           const push = ((minD - d) / minD) * this.topSpeed * 0.6;
  292|           this.desired.x += (dx / d) * push;
  293|           this.desired.z += (dz / d) * push;
  294|           // v12 hard resolve: steering alone loses to turn-first and to busy creatures
  295|           const k = (minD - d) * 0.5 * Math.min(1, dt * 10);
  296|           this.pos.x += (dx / d) * k;
  297|           this.pos.z += (dz / d) * k;
  298|         }
  299|       }
  300|     }
  301| 
  302|     // ---- the 3-step pipeline ---------------------------------------------
  303|     // the world-foot-planting solver needs the live gait numbers
  304|     const loco = this.motion.loco;
  305|     this.hero.drive.stride = loco.cfg.stride * (1 + 0.4 * loco.runT());
  306|     this.hero.drive.speed = loco.speed;
  307|     this.hero.drive.heading = loco.heading;
  308|     this.hero.drive.groundY = heightAt;
  309| 
  310|     this.motion.preAnimate(dt, this.desired);
  311|     this.hero.update(dt, null);
  312|     this.motion.postAnimate(dt);
  313|     this.applyDebugPose();
  314| 
  315|     // ---- v10: MOVE THE JUICE OFF THE PIVOT AND ONTO THE BODY --------------
  316|     // The pivot sits BELOW the leg roots, so squashing it scaled the IK targets
  317|     // with the body and the solver had to re-solve every frame — that was the
  318|     // knee pop. The body bone sits ABOVE the leg roots, exactly like the
  319|     // pedestal, so the juice no longer touches a planted foot.
  320|     const pivot = this.rig.pivot;
  321|     const body = this.rig.body;
  322|     if (this.noPivotJuice) {
  323|       pivot.position.set(0, 0, 0); pivot.scale.set(1, 1, 1); pivot.rotation.set(0, 0, 0);
  324|     } else {
  325|       body.position.set(body.position.x, this.restBodyY + pivot.position.y, body.position.z);
  326|       body.scale.multiply(pivot.scale);
  327|       // the WHOLE juice rotation moves to the body, not just the roll-over:
  328|       // reading only rotation.z silently discarded every pitch the motion
  329|       // layer applied, which is why the G-key pitch probe read 0.0000 m.
  330|       this.rolled = 2 * Math.acos(Math.min(1, Math.abs(pivot.quaternion.w))) > 0.6;
  331|       // v15: hysteresis, so a bouncy action cannot flicker the feet free/planted
  332|       const py = pivot.position.y;
  333|       this.hopping = this.motion.busy && (this.hopping ? py > 0.025 * this.bulk : py > 0.06 * this.bulk);
  334|       if (pivot.quaternion.x || pivot.quaternion.y || pivot.quaternion.z) {
  335|         body.quaternion.multiply(pivot.quaternion);
  336|       }
  337|       pivot.position.set(0, 0, 0); pivot.scale.set(1, 1, 1); pivot.rotation.set(0, 0, 0);
  338|     }
  339| 
  340|     // v12: legs solve LAST, against the final root / heading / body pose. Any
  341|     // additive pose the motion layer put on a hip (kick, flail, G-probe) is
  342|     // captured first and re-applied on top of the IK.
  343|     const hips = this.hero.legHips();
  344|     const rl = this.rig.legs;
  345|     for (let i = 0; i < rl.length; i++) {
  346|       if (!hips.has(rl[i]) || !this.legRest[i]) continue;
  347|       this.legDelta[i] = (this.legDelta[i] ?? new THREE.Quaternion()).copy(this.legRest[i]).invert().multiply(rl[i].quaternion);
  348|     }
  349|     const free = this.carried || this.rolled || this.hopping || this.motion.dangle.active || this.motion.dangle.airborne;
  350|     const tl = performance.now();
  351|     this.hero.solveWorldLegs(dt, free);
  352|     Agent.legMs += performance.now() - tl;
  353|     this.kicking.clear();
  354|     if (!free) {
  355|       for (let i = 0; i < rl.length; i++) {
  356|         const dq = this.legDelta[i];
  357|         if (!dq || !hips.has(rl[i])) continue;
  358|         const ang = 2 * Math.acos(Math.min(1, Math.abs(dq.w)));
  359|         if (ang < 1e-3) continue;
  360|         rl[i].quaternion.multiply(dq);
  361|         // a visible kick / flail / wave lifts that foot: it is airborne, not sliding
  362|         if (ang > 0.05) { const li = this.hero.legIndexOfHip(rl[i]); if (li >= 0) this.kicking.add(li); }
  363|       }
  364|     }
  365| 
  366|     this.readFeet();
  367|     this.tickEmote(dt);
  368|   }
  369| 
  370|   private tickEmote(dt: number): void {
  371|     if (this.emoteT > 0) {
  372|       this.emoteT -= dt;
  373|       if (this.emoteT <= 0) { this.emoteGlyph = null; this.emoteT = 0; }
  374|     }
  375|     this.wiggle = Math.max(0, this.wiggle - dt * 1.6);
  376|   }

  378|   /**
  379|    * Read the REAL ankle bones back after the animator has solved them, so the
  380|    * motion metrics measure the rendered rig and not a re-derivation of it.
  381|    */
  382|   private readFeet(): void {
  383|     const t = this.template;
  384|     const g = this.handle.group;
  385|     g.updateMatrixWorld(true);
  386|     const loco = this.motion.loco;
  387|     for (let i = 0; i < this.feet.length; i++) {
  388|       const f = this.feet[i];
  389|       const bone = this.handle.bones[t.feet[i].chain[2]];
  390|       const hipBone = this.handle.bones[t.feet[i].chain[0]];
  391|       const kneeBone = this.handle.bones[t.feet[i].chain[1]];
  392|       if (bone) bone.getWorldPosition(f.pos);
  393|       // the stance/swing flag now comes from the real world-planting solver
  394|       const legs = (this.hero as unknown as { legs: { index: number; mode: string }[] }).legs;
  395|       const leg = legs?.find((l) => l.index === i);
  396|       const swinging = (leg ? leg.mode === 'swing' : false) || this.hero.legsFree || this.kicking.has(i) || this.hero.legLifted(i);
  397|       f.swinging = swinging;
  398|       f.swing = 0;
  399|       if (!swinging && hipBone && kneeBone && bone) {
  400|         // v12: live world-space segment lengths; overstretch = leg locked straight
  401|         const hw = hipBone.getWorldPosition(this._hip);
  402|         const kw = kneeBone.getWorldPosition(this._kneeW);
  403|         const reach = (hw.distanceTo(kw) + kw.distanceTo(f.pos)) * 0.98;
  404|         if (hw.distanceTo(f.pos) > reach) this.overstretchFrames++;
  405|       }
  406|       // one dust puff per footfall, fired on the swing -> stance edge
  407|       if (this.prevSwing[i] && !swinging && loco.speed > 0.1) {
  408|         this.onFootfall?.(f.pos.x, f.pos.y, f.pos.z, 0.25 + loco.runT());
  409|       }
  410|       this.prevSwing[i] = swinging;
  411|     }
  412|   }
