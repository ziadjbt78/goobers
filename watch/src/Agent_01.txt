  251|       if (this.goal) {
  252|         const d = Math.hypot(this.goal.x - this.pos.x, this.goal.z - this.pos.z);
  253|         if (d < ARRIVE) this.goal = null;
  254|         else arrive(this.pos, this.goal, this.topSpeed * Math.max(0, this.throttle), SLOW_RADIUS, this.desired);
  255|       }
  256|       if (this.faceTarget) {
  257|         const dx = this.faceTarget.x - this.pos.x;
  258|         const dz = this.faceTarget.z - this.pos.z;
  259|         if (Math.hypot(dx, dz) < 0.04) this.faceTarget = null;
  260|         else {
  261|           // turn in place: aim a slow creep at the target, and let the
  262|           // heading spring in Locomotion do the turning
  263|           const a = Math.atan2(dx, dz);
  264|           this.desired.set(Math.sin(a) * 0.30, 0, Math.cos(a) * 0.30);
  265|         }
  266|       }
  267|     }
  268| 
  269|     // v12: personal space. Creatures never walk into each other's bodies; an
  270|     // overlapping neighbour adds a gentle push away to the desired velocity.
  271|     if (!this.carried) {
  272|       for (const o of this.world.nearby(this.pos, this.bulk * 2, this.id)) {
  273|         const dx = this.pos.x - o.pos.x;
  274|         const dz = this.pos.z - o.pos.z;
  275|         const d = Math.hypot(dx, dz) || 1e-4;
  276|         const minD = (this.bulk + o.bulk) * 0.42;
  277|         if (d < minD) {
  278|           const push = ((minD - d) / minD) * this.topSpeed * 0.6;
  279|           this.desired.x += (dx / d) * push;
  280|           this.desired.z += (dz / d) * push;
  281|         }
  282|       }
  283|     }
  284| 
  285|     // ---- the 3-step pipeline ---------------------------------------------
  286|     // the world-foot-planting solver needs the live gait numbers
  287|     const loco = this.motion.loco;
  288|     this.hero.drive.stride = loco.cfg.stride * (1 + 0.4 * loco.runT());
  289|     this.hero.drive.speed = loco.speed;
  290|     this.hero.drive.heading = loco.heading;
  291|     this.hero.drive.groundY = heightAt;
  292| 
  293|     this.motion.preAnimate(dt, this.desired);
  294|     this.hero.update(dt, null);
  295|     this.motion.postAnimate(dt);
  296|     this.applyDebugPose();
  297| 
  298|     // ---- v10: MOVE THE JUICE OFF THE PIVOT AND ONTO THE BODY --------------
  299|     // The pivot sits BELOW the leg roots, so squashing it scaled the IK targets
  300|     // with the body and the solver had to re-solve every frame — that was the
  301|     // knee pop. The body bone sits ABOVE the leg roots, exactly like the
  302|     // pedestal, so the juice no longer touches a planted foot.
  303|     const pivot = this.rig.pivot;
  304|     const body = this.rig.body;
  305|     if (this.noPivotJuice) {
  306|       pivot.position.set(0, 0, 0); pivot.scale.set(1, 1, 1); pivot.rotation.set(0, 0, 0);
  307|     } else {
  308|       body.position.set(body.position.x, this.restBodyY + pivot.position.y, body.position.z);
  309|       body.scale.multiply(pivot.scale);
  310|       // the WHOLE juice rotation moves to the body, not just the roll-over:
  311|       // reading only rotation.z silently discarded every pitch the motion
  312|       // layer applied, which is why the G-key pitch probe read 0.0000 m.
  313|       if (pivot.quaternion.x || pivot.quaternion.y || pivot.quaternion.z) {
  314|         body.quaternion.multiply(pivot.quaternion);
  315|       }
  316|       pivot.position.set(0, 0, 0); pivot.scale.set(1, 1, 1); pivot.rotation.set(0, 0, 0);
  317|     }
  318| 
  319|     this.readFeet();
  320|     this.tickEmote(dt);
  321|   }
  322| 
  323|   private tickEmote(dt: number): void {
  324|     if (this.emoteT > 0) {
  325|       this.emoteT -= dt;
  326|       if (this.emoteT <= 0) { this.emoteGlyph = null; this.emoteT = 0; }
  327|     }
  328|     this.wiggle = Math.max(0, this.wiggle - dt * 1.6);
  329|   }

  331|   /**
  332|    * Read the REAL ankle bones back after the animator has solved them, so the
  333|    * motion metrics measure the rendered rig and not a re-derivation of it.
  334|    */
  335|   private readFeet(): void {
  336|     const t = this.template;
  337|     const g = this.handle.group;
  338|     g.updateMatrixWorld(true);
  339|     const loco = this.motion.loco;
  340|     for (let i = 0; i < this.feet.length; i++) {
  341|       const f = this.feet[i];
  342|       const bone = this.handle.bones[t.feet[i].chain[2]];
  343|       const hipBone = this.handle.bones[t.feet[i].chain[0]];
  344|       const kneeBone = this.handle.bones[t.feet[i].chain[1]];
  345|       if (bone) bone.getWorldPosition(f.pos);
  346|       // the stance/swing flag now comes from the real world-planting solver
  347|       const legs = (this.hero as unknown as { legs: { index: number; mode: string }[] }).legs;
  348|       const leg = legs?.find((l) => l.index === i);
  349|       const swinging = leg ? leg.mode === 'swing' : false;
  350|       f.swinging = swinging;
  351|       f.swing = 0;
  352|       if (!swinging && hipBone && kneeBone && bone) {
  353|         const reach = (kneeBone.position.length() + bone.position.length()) * 0.95;
  354|         const hw = hipBone.getWorldPosition(this._hip);
  355|         if (hw.distanceTo(f.pos) > reach) this.overstretchFrames++;
  356|       }
  357|       // one dust puff per footfall, fired on the swing -> stance edge
  358|       if (this.prevSwing[i] && !swinging && loco.speed > 0.1) {
  359|         this.onFootfall?.(f.pos.x, f.pos.y, f.pos.z, 0.25 + loco.runT());
  360|       }
  361|       this.prevSwing[i] = swinging;
  362|     }
  363|   }

  365|   emoteAnchor(out: THREE.Vector3): THREE.Vector3 {
  366|     return out.copy(this.pos).setY(this.pos.y + this.template.height * this.dna.scale * 0.98);
  367|   }
