    1| /**
    2|  * HeroAnimator — THE one animation pipeline, and the place all the cartoon
    3|  * juice lives. Every view (Heroes, Zoo, Studio, World) drives this same class.
    4|  *
    5|  * Layers, all additive and all eased — never switched hard:
    6|  *   1. base        idle breath, ambient idle behaviour, blinks, micro-turn
    7|  *   2. locomotion  gait driven by ACTUAL ground speed, eased in over 0.25s
    8|  *   3. action      eat / sniff / play / pet / sleep / greet / carry / startle
    9|  *   4. look        gaze, deliberate eye lead, saccades
   10|  *   5. secondary   damped springs on ears, tails, antennae
   11|  *
   12|  * Exaggeration is the point. Body bob is 8-12% of creature height, every
   13|  * footfall squashes the body, turning rolls it 12 degrees, acceleration pitches
   14|  * it and stopping recoils it. Subtle is a bug.
   15|  *
   16|  * The idle path reproduces the locked baseline: locomotionW 0 and actionW 0
   17|  * with the ambient idle timer parked gives the approved pedestal pose.
   18|  */

  195|   constructor(template: HeroTemplate, handle: HeroHandle) {
  196|     this.template = template;
  197|     this.handle = handle;
  198| 
  199|     for (let i = 0; i < template.feet.length; i++) {
  200|       const f = template.feet[i];
  201|       const [hi, ki, ai] = f.chain;
  202|       const hip = handle.bones[hi];
  203|       const knee = handle.bones[ki];
  204|       const ankle = handle.bones[ai];
  205|       if (!hip || !knee || !ankle) continue;
  206|       // ---- v9 pole, per leg, in creature space ---------------------------
  207|       // A quadruped's REAR knees bend backward and its FRONT knees forward; a
  208|       // hexapod's legs all bow OUTWARD; a biped keeps the pedestal's forward
  209|       // pole. Locking this per leg is what stops the mirrored-branch flip.
  210|       let poleLocal: THREE.Vector3;
  211|       if (template.locomotion === 'skitter') {
  212|         poleLocal = new THREE.Vector3(f.side >= 0 ? 1 : -1, 0, 0);
  213|       } else if (template.feet.length === 4) {
  214|         poleLocal = new THREE.Vector3(0, 0, f.plant.z < 0 ? -1 : 1);
  215|       } else {
  216|         poleLocal = new THREE.Vector3(0, 0, 1);
  217|       }
  218|       this.legs.push({
  219|         hip, knee, ankle,
  220|         plant: f.plant.clone(), step: f.step, lift: f.lift, phase: f.phase, index: i,
  221|         prev: [hip.quaternion.clone(), knee.quaternion.clone(), ankle.quaternion.clone()],
  222|         // rest knee in the HIP-PARENT's space — the space solveIK2 works in
  223|         kneeLocal: hip.position.clone().add(knee.position),
  224|         poleLocal, poleSm: new THREE.Vector3(), wasSwing: false,
  225|         lock: f.plant.clone(), swingFrom: f.plant.clone(), swingTo: f.plant.clone(),
  226|         swingT: 0, mode: 'plant', swingDur: 0.2, init: false,
  227|         bodyAtPlant: new THREE.Vector3(),
  228|         L1: knee.position.length(), L2: ankle.position.length(),
  229|         restAngle: Math.atan2(f.plant.x, f.plant.z), forced: false,
  230|       });
  231|       this.drive.footTargets.push(f.plant.clone());
  232|       this.drive.footSwing.push(false);
  233|     }
  234| 
  235|     for (const b of handle.bones) {
  236|       if (b.name.startsWith('arm') && b.name.includes('upper')) {
  237|         const lower = handle.bones.find((o) => o.name === b.name.replace('upper', 'lower'));
  238|         if (lower) this.arms.push({ upper: b, lower, side: b.position.x < 0 ? -1 : 1 });
  239|       }
  240|     }
  241|     for (const b of handle.bones) if (b.name.startsWith('spine')) this.spine.push(b);
  242| 
  243|     for (const b of handle.bones) {
  244|       const n = b.name;
  245|       if (n.startsWith('ear') || n.startsWith('tail') || n.startsWith('ant') || n.startsWith('sprout_leaf') || n.startsWith('antenna')) {
  246|         this.wiggles.push({
  247|           bone: b, w: new Wobble(),
  248|           axis: n.startsWith('ant') || n.startsWith('antenna') ? 'y' : 'x',
  249|           gain: n.startsWith('tail') ? 1.4 : 1.0,
  250|         });
