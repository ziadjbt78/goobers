  211|     for (let i = 0; i < template.feet.length; i++) {
  212|       const f = template.feet[i];
  213|       const [hi, ki, ai] = f.chain;
  214|       const hip = handle.bones[hi];
  215|       const knee = handle.bones[ki];
  216|       const ankle = handle.bones[ai];
  217|       if (!hip || !knee || !ankle) continue;
  218|       // ---- v9 pole, per leg, in creature space ---------------------------

  220|       // hexapod's legs all bow OUTWARD; a biped keeps the pedestal's forward

  224|         poleLocal = new THREE.Vector3(f.side >= 0 ? 1 : -1, 0, 0);
  225|       } else if (template.feet.length === 4) {
  226|         poleLocal = new THREE.Vector3(0, 0, f.plant.z < 0 ? -1 : 1);
  227|       } else {
  228|         poleLocal = new THREE.Vector3(0, 0, 1);
  229|       }
