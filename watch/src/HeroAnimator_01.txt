  327|   update(dt: number, lookAt: THREE.Vector3 | null): void {
  328|     const h = this.handle;
  329|     const t = this.template;
  330|     this.t += dt;
  331|     this.excited = Math.max(0, this.excited - dt);
  332|     this.actionT += dt;
  333| 
  334|     // ---- layer weights -----------------------------------------------------
  335|     const jumpy = t.locomotion === 'hop';
  336|     const moving = this.driven && !this.external ? (this.drive.speed > 0.05 ? 1 : 0) : (this.gait > 0.02 ? 1 : 0);
  337|     const locoTarget = this.driven || this.external ? moving : 0;
  338|     this.locomotionW += (locoTarget - this.locomotionW) * Math.min(1, dt / 0.25);
  339|     const loco = this.driven || this.external ? this.locomotionW : 0;
  340|     const actTarget = this.action === 'none' ? 0 : 1;
  341|     this.actionBlend += (actTarget - this.actionBlend) * Math.min(1, dt / 0.25);
  342|     const act = this.actionBlend;
  343| 
  344|     // ---- ambient idle behaviour: the creature is never frozen --------------
  345|     // in external mode the motion layer plays ambient behaviour through Choreo
  346|     const standingStill = !this.external && loco < 0.15 && act < 0.2;
  347|     if (standingStill && this.idleAct === 'none') {
  348|       this.idleTimer -= dt;
  349|       if (this.idleTimer <= 0) {
  350|         const pool: IdleAct[] = ['sniff', 'scratch', 'sit', 'yawn', 'sniff'];
  351|         const pick = this.deterministic ? 'sniff' : pool[Math.floor(Math.random() * pool.length)];
  352|         this.idleAct = pick;
  353|         this.idleActT = pick === 'sit' ? 2.4 : 1.3;
  354|         this.idleTimer = 4 + Math.random() * 6;
  355|       }
  356|     }
  357|     if (this.idleAct !== 'none') {
  358|       this.idleActT -= dt;
  359|       if (this.idleActT <= 0) { this.idleAct = 'none'; this.idleActW = 0; }
  360|     }
  361|     const idleTarget = this.idleAct !== 'none' && standingStill ? 1 : 0;
  362|     this.idleActW += (idleTarget - this.idleActW) * Math.min(1, dt / 0.25);
  363|     const ia = this.idleActW;
  364| 
  365|     // ---- the shared clock --------------------------------------------------
  366|     const speed01 = this.driven && !this.external ? Math.min(1, this.drive.speed / 1.4) : this.gait;
  367|     if (this.external) {
  368|       this.phase = this.externalPhase;
  369|     } else if (this.driven) {
  370|       if (jumpy) this.phase += dt * (1.2 + speed01 * 2.6);
  371|       else {
  372|         const stride = Math.max(0.16, this.legs.length ? this.legs[0].step * 2.1 * Math.max(1, this.bulkScale * 0.5) : 0.4);
  373|         this.phase = (this.phase + (this.drive.speed * dt) / stride) % 1;
  374|       }
  375|     } else {
  376|       this.phase += dt * (jumpy ? 1.3 + this.gait * 2.2 : 1.1 + this.gait * 2.4);
  377|     }
  378| 
  379|     // ---- body ---------------------------------------------------------------
  380|     let bodyLift = 0;
  381|     let bodyPitch = 0;
  382|     let bodyRoll = 0;
  383| 
  384|     const beatLegs = this.template.feet.length > 4 ? 3 : 2;
  385|     const beat = Math.abs(Math.sin(this.phase * Math.PI * 2 * beatLegs));
  386| 
  387|     if (this.external) {
  388|       // The motion layer writes lift / squash / pitch / roll / yaw on the pivot
  389|       // and body bones. The animator adds NOTHING here, so no channel is
  390|       // applied twice.
  391|       bodyLift = 0;
  392|       bodyPitch = 0;
  393|       bodyRoll = 0;
  394|       this.squash = 1;
  395|       this.bodyYaw = 0;
  396|       this.headPitch = 0;
  397|     } else if (this.driven) {
  398|       bodyLift = this.drive.lift;
  399|       bodyPitch = this.drive.pitch + this.drive.slopePitch;
  400|       bodyRoll = -this.drive.lean + this.drive.slopeRoll;
  401|       this.squash = this.drive.squash;
  402|       this.bodyYaw = this.drive.yaw;
  403|       this.headPitch = 0;
  404|     } else if (jumpy) {
  405|       this.hopV -= 22 * dt;
  406|       this.hopY += this.hopV * dt;
  407|       if (this.hopY < 0) {
  408|         this.hopY = 0;
  409|         if (this.hopV < -0.6) this.squashV += -this.hopV * 1.6;
  410|         this.hopV = 0;
  411|       }
  412|       if (this.hopY === 0 && moving > 0) {
  413|         const push = Math.max(0, Math.sin(this.phase * Math.PI * 2 * 0.5 + 0.4));
  414|         if (push > 0.97) this.hopV = 2.6 + this.gait * 2.4;
  415|       }
  416|       bodyLift = this.hopY;
  417|       const airborne = this.hopY > 0.001 ? 1 : 0;
  418|       this.squashV += (-(this.squash - 1) * 120 - this.squashV * 14) * dt;
  419|       this.squash += this.squashV * dt;
  420|       this.squash = Math.max(0.72, Math.min(1.22, this.squash));
  421|       bodyPitch = airborne * -0.06;
  422|     } else {
  423|       const bob = (beat - 0.5) * 0.028 * this.gait;
  424|       const idle = Math.sin(this.t * 1.6) * 0.006 * (1 - moving);
  425|       bodyLift = bob + idle + this.excited * Math.abs(Math.sin(this.t * 11)) * 0.05;
  426|       this.squash += (1 - this.squash) * Math.min(1, dt * 9);
  427|       bodyLift += -this.squashV * 0.004;
  428|       this.squashV *= Math.max(0, 1 - dt * 8);
  429|     }
  430| 
  431|     if (this.driven && !this.external) bodyLift += Math.sin(this.t * 1.6) * 0.006;
  432|     bodyLift -= loco * 0.012;
  433|     if (this.external) {
  434|       bodyLift = 0; bodyPitch = 0; bodyRoll = 0;
  435|       this.squash = 1; this.bodyYaw = 0; this.headPitch = 0;
  436|     }

  438|     // ---- ambient idle poses, layered on the base ---------------------------

  441|       const ease = Math.sin(Math.min(1, u) * Math.PI);

  456|         case 'yawn':

  458|           bodyLift += ease * 0.06 * ia;
  459|           this.squash *= 1 + ease * 0.10 * ia;
