  337|   update(dt: number, lookAt: THREE.Vector3 | null): void {
  338|     const h = this.handle;
  339|     const t = this.template;
  340|     this.t += dt;
  341|     this.excited = Math.max(0, this.excited - dt);
  342|     this.actionT += dt;
  343| 
  344|     // ---- layer weights -----------------------------------------------------
  345|     const jumpy = t.locomotion === 'hop';
  346|     const moving = this.driven && !this.external ? (this.drive.speed > 0.05 ? 1 : 0) : (this.gait > 0.02 ? 1 : 0);
  347|     const locoTarget = this.driven || this.external ? moving : 0;
  348|     this.locomotionW += (locoTarget - this.locomotionW) * Math.min(1, dt / 0.25);
  349|     const loco = this.driven || this.external ? this.locomotionW : 0;
  350|     const actTarget = this.action === 'none' ? 0 : 1;
  351|     this.actionBlend += (actTarget - this.actionBlend) * Math.min(1, dt / 0.25);
  352|     const act = this.actionBlend;
  353| 
  354|     // ---- ambient idle behaviour: the creature is never frozen --------------
  355|     // in external mode the motion layer plays ambient behaviour through Choreo
  356|     const standingStill = !this.external && loco < 0.15 && act < 0.2;
  357|     if (standingStill && this.idleAct === 'none') {
  358|       this.idleTimer -= dt;
  359|       if (this.idleTimer <= 0) {
  360|         const pool: IdleAct[] = ['sniff', 'scratch', 'sit', 'yawn', 'sniff'];
  361|         const pick = this.deterministic ? 'sniff' : pool[Math.floor(Math.random() * pool.length)];
  362|         this.idleAct = pick;
  363|         this.idleActT = pick === 'sit' ? 2.4 : 1.3;
  364|         this.idleTimer = 4 + Math.random() * 6;
  365|       }
  366|     }
  367|     if (this.idleAct !== 'none') {
  368|       this.idleActT -= dt;
  369|       if (this.idleActT <= 0) { this.idleAct = 'none'; this.idleActW = 0; }
  370|     }
  371|     const idleTarget = this.idleAct !== 'none' && standingStill ? 1 : 0;
  372|     this.idleActW += (idleTarget - this.idleActW) * Math.min(1, dt / 0.25);
  373|     const ia = this.idleActW;
  374| 
  375|     // ---- the shared clock --------------------------------------------------
  376|     const speed01 = this.driven && !this.external ? Math.min(1, this.drive.speed / 1.4) : this.gait;
  377|     if (this.external) {
  378|       this.phase = this.externalPhase;
  379|     } else if (this.driven) {
  380|       if (jumpy) this.phase += dt * (1.2 + speed01 * 2.6);
  381|       else {
  382|         const stride = Math.max(0.16, this.legs.length ? this.legs[0].step * 2.1 * Math.max(1, this.bulkScale * 0.5) : 0.4);
  383|         this.phase = (this.phase + (this.drive.speed * dt) / stride) % 1;
  384|       }
  385|     } else {
  386|       this.phase += dt * (jumpy ? 1.3 + this.gait * 2.2 : 1.1 + this.gait * 2.4);
  387|     }
  388| 
  389|     // ---- body ---------------------------------------------------------------
  390|     let bodyLift = 0;
  391|     let bodyPitch = 0;
  392|     let bodyRoll = 0;
  393| 
  394|     const beatLegs = this.template.feet.length > 4 ? 3 : 2;
  395|     const beat = Math.abs(Math.sin(this.phase * Math.PI * 2 * beatLegs));
  396| 
  397|     if (this.external) {
  398|       // The motion layer writes lift / squash / pitch / roll / yaw on the pivot
  399|       // and body bones. The animator adds NOTHING here, so no channel is
  400|       // applied twice.
  401|       bodyLift = 0;
  402|       bodyPitch = 0;
  403|       bodyRoll = 0;
  404|       this.squash = 1;
  405|       this.bodyYaw = 0;
  406|       this.headPitch = 0;
  407|     } else if (this.driven) {
  408|       bodyLift = this.drive.lift;
  409|       bodyPitch = this.drive.pitch + this.drive.slopePitch;
  410|       bodyRoll = -this.drive.lean + this.drive.slopeRoll;
  411|       this.squash = this.drive.squash;
  412|       this.bodyYaw = this.drive.yaw;
  413|       this.headPitch = 0;
  414|     } else if (jumpy) {
  415|       this.hopV -= 22 * dt;
  416|       this.hopY += this.hopV * dt;
  417|       if (this.hopY < 0) {
  418|         this.hopY = 0;
  419|         if (this.hopV < -0.6) this.squashV += -this.hopV * 1.6;
  420|         this.hopV = 0;
  421|       }
  422|       if (this.hopY === 0 && moving > 0) {
  423|         const push = Math.max(0, Math.sin(this.phase * Math.PI * 2 * 0.5 + 0.4));
  424|         if (push > 0.97) this.hopV = 2.6 + this.gait * 2.4;
  425|       }
  426|       bodyLift = this.hopY;
  427|       const airborne = this.hopY > 0.001 ? 1 : 0;
  428|       this.squashV += (-(this.squash - 1) * 120 - this.squashV * 14) * dt;
  429|       this.squash += this.squashV * dt;
  430|       this.squash = Math.max(0.72, Math.min(1.22, this.squash));
  431|       bodyPitch = airborne * -0.06;
  432|     } else {
  433|       const bob = (beat - 0.5) * 0.028 * this.gait;
  434|       const idle = Math.sin(this.t * 1.6) * 0.006 * (1 - moving);
  435|       bodyLift = bob + idle + this.excited * Math.abs(Math.sin(this.t * 11)) * 0.05;
  436|       this.squash += (1 - this.squash) * Math.min(1, dt * 9);
  437|       bodyLift += -this.squashV * 0.004;
  438|       this.squashV *= Math.max(0, 1 - dt * 8);
  439|     }
  440| 
  441|     if (this.driven && !this.external) bodyLift += Math.sin(this.t * 1.6) * 0.006;
  442|     bodyLift -= loco * 0.012;
  443|     if (this.external) {
  444|       bodyLift = 0; bodyPitch = 0; bodyRoll = 0;
  445|       this.squash = 1; this.bodyYaw = 0; this.headPitch = 0;
  446|     }
  447| 
  448|     // ---- ambient idle poses, layered on the base ---------------------------
  449|     if (ia > 0.001) {

  451|       const ease = Math.sin(Math.min(1, u) * Math.PI);

  460|           break;
  461|         case 'sit':

  465|           break;
  466|         case 'yawn':

  468|           bodyLift += ease * 0.06 * ia;
  469|           this.squash *= 1 + ease * 0.10 * ia;
  470|           break;
  471|         default: break;
  472|       }

  475|     // ---- attention: gaze, blink --------------------------------------------
