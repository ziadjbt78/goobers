  315|   update(dt: number, lookAt: THREE.Vector3 | null): void {
  316|     const h = this.handle;
  317|     const t = this.template;
  318|     this.t += dt;
  319|     this.excited = Math.max(0, this.excited - dt);
  320|     this.actionT += dt;
  321| 
  322|     // ---- layer weights -----------------------------------------------------
  323|     const jumpy = t.locomotion === 'hop';
  324|     const moving = this.driven && !this.external ? (this.drive.speed > 0.05 ? 1 : 0) : (this.gait > 0.02 ? 1 : 0);
  325|     const locoTarget = this.driven || this.external ? moving : 0;
  326|     this.locomotionW += (locoTarget - this.locomotionW) * Math.min(1, dt / 0.25);
  327|     const loco = this.driven || this.external ? this.locomotionW : 0;
  328|     const actTarget = this.action === 'none' ? 0 : 1;
  329|     this.actionBlend += (actTarget - this.actionBlend) * Math.min(1, dt / 0.25);
  330|     const act = this.actionBlend;
  331| 
  332|     // ---- ambient idle behaviour: the creature is never frozen --------------
  333|     // in external mode the motion layer plays ambient behaviour through Choreo
  334|     const standingStill = !this.external && loco < 0.15 && act < 0.2;
  335|     if (standingStill && this.idleAct === 'none') {
  336|       this.idleTimer -= dt;
  337|       if (this.idleTimer <= 0) {
  338|         const pool: IdleAct[] = ['sniff', 'scratch', 'sit', 'yawn', 'sniff'];
  339|         const pick = this.deterministic ? 'sniff' : pool[Math.floor(Math.random() * pool.length)];
  340|         this.idleAct = pick;
  341|         this.idleActT = pick === 'sit' ? 2.4 : 1.3;
  342|         this.idleTimer = 4 + Math.random() * 6;
  343|       }
  344|     }
  345|     if (this.idleAct !== 'none') {
  346|       this.idleActT -= dt;
  347|       if (this.idleActT <= 0) { this.idleAct = 'none'; this.idleActW = 0; }
  348|     }
  349|     const idleTarget = this.idleAct !== 'none' && standingStill ? 1 : 0;
  350|     this.idleActW += (idleTarget - this.idleActW) * Math.min(1, dt / 0.25);
  351|     const ia = this.idleActW;
  352| 
  353|     // ---- the shared clock --------------------------------------------------
  354|     const speed01 = this.driven && !this.external ? Math.min(1, this.drive.speed / 1.4) : this.gait;
  355|     if (this.external) {
  356|       this.phase = this.externalPhase;
  357|     } else if (this.driven) {
  358|       if (jumpy) this.phase += dt * (1.2 + speed01 * 2.6);
  359|       else {
  360|         const stride = Math.max(0.16, this.legs.length ? this.legs[0].step * 2.1 * Math.max(1, this.bulkScale * 0.5) : 0.4);
  361|         this.phase = (this.phase + (this.drive.speed * dt) / stride) % 1;
  362|       }
  363|     } else {
  364|       this.phase += dt * (jumpy ? 1.3 + this.gait * 2.2 : 1.1 + this.gait * 2.4);
  365|     }
  366| 
  367|     // ---- body ---------------------------------------------------------------
  368|     let bodyLift = 0;
  369|     let bodyPitch = 0;
  370|     let bodyRoll = 0;
  371| 
  372|     const beatLegs = this.template.feet.length > 4 ? 3 : 2;
  373|     const beat = Math.abs(Math.sin(this.phase * Math.PI * 2 * beatLegs));
  374| 
  375|     if (this.external) {
  376|       // The motion layer writes lift / squash / pitch / roll / yaw on the pivot
  377|       // and body bones. The animator adds NOTHING here, so no channel is
  378|       // applied twice.
  379|       bodyLift = 0;
  380|       bodyPitch = 0;
  381|       bodyRoll = 0;
  382|       this.squash = 1;
  383|       this.bodyYaw = 0;
  384|       this.headPitch = 0;
  385|     } else if (this.driven) {
  386|       bodyLift = this.drive.lift;
  387|       bodyPitch = this.drive.pitch + this.drive.slopePitch;
  388|       bodyRoll = -this.drive.lean + this.drive.slopeRoll;
  389|       this.squash = this.drive.squash;
  390|       this.bodyYaw = this.drive.yaw;
  391|       this.headPitch = 0;
  392|     } else if (jumpy) {
  393|       this.hopV -= 22 * dt;
  394|       this.hopY += this.hopV * dt;
  395|       if (this.hopY < 0) {
  396|         this.hopY = 0;
  397|         if (this.hopV < -0.6) this.squashV += -this.hopV * 1.6;
  398|         this.hopV = 0;
  399|       }
  400|       if (this.hopY === 0 && moving > 0) {
  401|         const push = Math.max(0, Math.sin(this.phase * Math.PI * 2 * 0.5 + 0.4));
  402|         if (push > 0.97) this.hopV = 2.6 + this.gait * 2.4;
  403|       }
  404|       bodyLift = this.hopY;
  405|       const airborne = this.hopY > 0.001 ? 1 : 0;
  406|       this.squashV += (-(this.squash - 1) * 120 - this.squashV * 14) * dt;
  407|       this.squash += this.squashV * dt;
  408|       this.squash = Math.max(0.72, Math.min(1.22, this.squash));
  409|       bodyPitch = airborne * -0.06;
  410|     } else {
  411|       const bob = (beat - 0.5) * 0.028 * this.gait;
  412|       const idle = Math.sin(this.t * 1.6) * 0.006 * (1 - moving);
  413|       bodyLift = bob + idle + this.excited * Math.abs(Math.sin(this.t * 11)) * 0.05;
  414|       this.squash += (1 - this.squash) * Math.min(1, dt * 9);
  415|       bodyLift += -this.squashV * 0.004;
  416|       this.squashV *= Math.max(0, 1 - dt * 8);
  417|     }
  418| 
  419|     if (this.driven && !this.external) bodyLift += Math.sin(this.t * 1.6) * 0.006;
  420|     bodyLift -= loco * 0.012;
  421|     if (this.external) {
  422|       bodyLift = 0; bodyPitch = 0; bodyRoll = 0;
  423|       this.squash = 1; this.bodyYaw = 0; this.headPitch = 0;
  424|     }
  425| 
  426|     // ---- ambient idle poses, layered on the base ---------------------------
  427|     if (ia > 0.001) {
  428|       const u = this.idleAct === 'none' ? 0 : Math.min(1, this.idleActT > 0.001 ? 1 - this.idleActT / 1.3 : 1);
  429|       const ease = Math.sin(Math.min(1, u) * Math.PI);
  430|       switch (this.idleAct) {
  431|         case 'sniff':
  432|           bodyPitch += ease * 0.42 * ia;                       // head down to the ground
  433|           bodyLift -= ease * 0.05 * ia;
  434|           break;
  435|         case 'scratch':
  436|           bodyRoll += Math.sin(this.idleActT * 26) * 0.20 * ia;
  437|           bodyPitch += ease * 0.10 * ia;
  438|           break;
  439|         case 'sit':
  440|           bodyPitch -= ease * 0.34 * ia;                       // rock back on the haunches
  441|           bodyLift -= ease * 0.10 * ia;
  442|           this.squash *= 1 - ease * 0.14 * ia;
  443|           break;
  444|         case 'yawn':
  445|           bodyPitch -= ease * 0.22 * ia;                       // big stretch up
  446|           bodyLift += ease * 0.06 * ia;
  447|           this.squash *= 1 + ease * 0.10 * ia;
  448|           break;
  449|         default: break;
  450|       }
  451|     }
  452| 
  453|     // ---- attention: gaze, blink --------------------------------------------
  454|     this.nextLook -= dt;
  455|     if (this.nextLook <= 0) {
  456|       this.nextLook = 0.5 + Math.random() * 1.5;
  457|       this.lookX = this.deterministic ? 0 : (Math.random() - 0.5) * 0.9;
  458|       this.lookY = this.deterministic ? 0 : (Math.random() - 0.5) * 0.5;
  459|     }
  460|     if (lookAt) {
  461|       const eye = this._v.setFromMatrixPosition(h.eyes[0]?.root.matrixWorld ?? h.group.matrixWorld);
  462|       const local = eye.clone().sub(h.group.getWorldPosition(new THREE.Vector3()));
  463|       const yaw = Math.atan2(local.x, local.z);
  464|       const dist = Math.max(0.3, local.length());
  465|       const pitch = Math.atan2(lookAt.y - eye.y, dist);
  466|       this.lookX = THREE.MathUtils.clamp(Math.atan2(Math.sin(yaw), Math.cos(yaw)), -0.9, 0.9);
  467|       this.lookY = THREE.MathUtils.clamp(pitch, -0.5, 0.5);
  468|     }
  469|     // The eyes lead: they snap onto the target FIRST, the head arrives 0.15 s
  470|     // later (drive.headLead is already delayed by the Agent), the body last.
  471|     if (this.driven && Math.abs(this.drive.eyeLead) > 0.01) {
  472|       this.lookX = THREE.MathUtils.clamp(this.lookX + this.drive.eyeLead, -1.1, 1.1);
  473|     }
  474|     this.gazeYaw += (this.lookX - this.gazeYaw) * Math.min(1, dt * 12);
  475|     this.gazePitch += (this.lookY - this.gazePitch) * Math.min(1, dt * 12);
  476| 
  477|     this.blinkTimer -= dt;
  478|     if (this.blinkTimer <= 0) {
  479|       const sleepy = this.mood === 'sleepy';
  480|       this.blinkT = 1;
  481|       this.blinkTimer = (sleepy ? 1.4 : 2.0 + Math.random() * 4);
  482|       if (this.doubleBlink > 0) { this.doubleBlink = 0; this.blinkTimer = 0.16; }
  483|       else if (!this.deterministic && Math.random() < 0.16) this.doubleBlink = 1;
  484|       this.blinkSide = !this.deterministic && Math.random() < 0.08 ? (Math.random() < 0.5 ? -1 : 1) : 0;
  485|     }
  486|     if (this.blinkT > 0) this.blinkT = Math.max(0, this.blinkT - dt * 7);
  487| 
  488|     if (!this.driven) {
  489|       this.nextTurn -= dt;
  490|       if (this.nextTurn <= 0) {
  491|         this.nextTurn = 1.6 + Math.random() * 3;
  492|         this.bodyYawTarget = this.deterministic ? 0 : (Math.random() - 0.5) * 0.55;
  493|       }
  494|       this.headYawIdle = THREE.MathUtils.clamp(this.bodyYawTarget - this.bodyYaw, -0.55, 0.55) * 0.9;
  495|       this.bodyYaw += (this.bodyYawTarget - this.bodyYaw) * Math.min(1, dt * 1.5);
  496|       this.headPitch = 0;
  497|     }
  498|     const headLead = this.driven ? this.drive.headLead : this.headYawIdle;
  499| 
  500|     // ---- action layer -------------------------------------------------------
