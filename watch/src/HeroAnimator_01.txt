  329|   update(dt: number, lookAt: THREE.Vector3 | null): void {
  330|     const h = this.handle;
  331|     const t = this.template;
  332|     this.t += dt;
  333|     this.excited = Math.max(0, this.excited - dt);
  334|     this.actionT += dt;
  335| 
  336|     // ---- layer weights -----------------------------------------------------
  337|     const jumpy = t.locomotion === 'hop';
  338|     const moving = this.driven && !this.external ? (this.drive.speed > 0.05 ? 1 : 0) : (this.gait > 0.02 ? 1 : 0);
  339|     const locoTarget = this.driven || this.external ? moving : 0;
  340|     this.locomotionW += (locoTarget - this.locomotionW) * Math.min(1, dt / 0.25);
  341|     const loco = this.driven || this.external ? this.locomotionW : 0;
  342|     const actTarget = this.action === 'none' ? 0 : 1;
  343|     this.actionBlend += (actTarget - this.actionBlend) * Math.min(1, dt / 0.25);
  344|     const act = this.actionBlend;
  345| 
  346|     // ---- ambient idle behaviour: the creature is never frozen --------------
  347|     // in external mode the motion layer plays ambient behaviour through Choreo
  348|     const standingStill = !this.external && loco < 0.15 && act < 0.2;
  349|     if (standingStill && this.idleAct === 'none') {
  350|       this.idleTimer -= dt;
  351|       if (this.idleTimer <= 0) {
  352|         const pool: IdleAct[] = ['sniff', 'scratch', 'sit', 'yawn', 'sniff'];
  353|         const pick = this.deterministic ? 'sniff' : pool[Math.floor(Math.random() * pool.length)];
  354|         this.idleAct = pick;
  355|         this.idleActT = pick === 'sit' ? 2.4 : 1.3;
  356|         this.idleTimer = 4 + Math.random() * 6;
  357|       }
  358|     }
  359|     if (this.idleAct !== 'none') {
  360|       this.idleActT -= dt;
  361|       if (this.idleActT <= 0) { this.idleAct = 'none'; this.idleActW = 0; }
  362|     }
  363|     const idleTarget = this.idleAct !== 'none' && standingStill ? 1 : 0;
  364|     this.idleActW += (idleTarget - this.idleActW) * Math.min(1, dt / 0.25);
  365|     const ia = this.idleActW;
  366| 
  367|     // ---- the shared clock --------------------------------------------------
  368|     const speed01 = this.driven && !this.external ? Math.min(1, this.drive.speed / 1.4) : this.gait;
  369|     if (this.external) {
  370|       this.phase = this.externalPhase;
  371|     } else if (this.driven) {
  372|       if (jumpy) this.phase += dt * (1.2 + speed01 * 2.6);
  373|       else {
  374|         const stride = Math.max(0.16, this.legs.length ? this.legs[0].step * 2.1 * Math.max(1, this.bulkScale * 0.5) : 0.4);
  375|         this.phase = (this.phase + (this.drive.speed * dt) / stride) % 1;
  376|       }
  377|     } else {
  378|       this.phase += dt * (jumpy ? 1.3 + this.gait * 2.2 : 1.1 + this.gait * 2.4);
  379|     }
  380| 
  381|     // ---- body ---------------------------------------------------------------
  382|     let bodyLift = 0;
  383|     let bodyPitch = 0;
  384|     let bodyRoll = 0;
  385| 
  386|     const beatLegs = this.template.feet.length > 4 ? 3 : 2;
  387|     const beat = Math.abs(Math.sin(this.phase * Math.PI * 2 * beatLegs));
  388| 
  389|     if (this.external) {
  390|       // The motion layer writes lift / squash / pitch / roll / yaw on the pivot
  391|       // and body bones. The animator adds NOTHING here, so no channel is
  392|       // applied twice.
  393|       bodyLift = 0;
  394|       bodyPitch = 0;
  395|       bodyRoll = 0;
  396|       this.squash = 1;
  397|       this.bodyYaw = 0;
  398|       this.headPitch = 0;
  399|     } else if (this.driven) {
  400|       bodyLift = this.drive.lift;
  401|       bodyPitch = this.drive.pitch + this.drive.slopePitch;
  402|       bodyRoll = -this.drive.lean + this.drive.slopeRoll;
  403|       this.squash = this.drive.squash;
  404|       this.bodyYaw = this.drive.yaw;
  405|       this.headPitch = 0;
  406|     } else if (jumpy) {
  407|       this.hopV -= 22 * dt;
  408|       this.hopY += this.hopV * dt;
  409|       if (this.hopY < 0) {
  410|         this.hopY = 0;
  411|         if (this.hopV < -0.6) this.squashV += -this.hopV * 1.6;
  412|         this.hopV = 0;
  413|       }
  414|       if (this.hopY === 0 && moving > 0) {
  415|         const push = Math.max(0, Math.sin(this.phase * Math.PI * 2 * 0.5 + 0.4));
  416|         if (push > 0.97) this.hopV = 2.6 + this.gait * 2.4;
  417|       }
  418|       bodyLift = this.hopY;
  419|       const airborne = this.hopY > 0.001 ? 1 : 0;
  420|       this.squashV += (-(this.squash - 1) * 120 - this.squashV * 14) * dt;
  421|       this.squash += this.squashV * dt;
  422|       this.squash = Math.max(0.72, Math.min(1.22, this.squash));
  423|       bodyPitch = airborne * -0.06;
  424|     } else {
  425|       const bob = (beat - 0.5) * 0.028 * this.gait;
  426|       const idle = Math.sin(this.t * 1.6) * 0.006 * (1 - moving);
  427|       bodyLift = bob + idle + this.excited * Math.abs(Math.sin(this.t * 11)) * 0.05;
  428|       this.squash += (1 - this.squash) * Math.min(1, dt * 9);
  429|       bodyLift += -this.squashV * 0.004;
  430|       this.squashV *= Math.max(0, 1 - dt * 8);
  431|     }
  432| 
  433|     if (this.driven && !this.external) bodyLift += Math.sin(this.t * 1.6) * 0.006;
  434|     bodyLift -= loco * 0.012;
  435|     if (this.external) {
  436|       bodyLift = 0; bodyPitch = 0; bodyRoll = 0;
  437|       this.squash = 1; this.bodyYaw = 0; this.headPitch = 0;
  438|     }
  439| 
  440|     // ---- ambient idle poses, layered on the base ---------------------------
  441|     if (ia > 0.001) {
  442|       const u = this.idleAct === 'none' ? 0 : Math.min(1, this.idleActT > 0.001 ? 1 - this.idleActT / 1.3 : 1);
  443|       const ease = Math.sin(Math.min(1, u) * Math.PI);
  444|       switch (this.idleAct) {
  445|         case 'sniff':
  446|           bodyPitch += ease * 0.42 * ia;                       // head down to the ground
  447|           bodyLift -= ease * 0.05 * ia;
  448|           break;
  449|         case 'scratch':
  450|           bodyRoll += Math.sin(this.idleActT * 26) * 0.20 * ia;
  451|           bodyPitch += ease * 0.10 * ia;
  452|           break;
  453|         case 'sit':
  454|           bodyPitch -= ease * 0.34 * ia;                       // rock back on the haunches
  455|           bodyLift -= ease * 0.10 * ia;
  456|           this.squash *= 1 - ease * 0.14 * ia;
  457|           break;
  458|         case 'yawn':
  459|           bodyPitch -= ease * 0.22 * ia;                       // big stretch up
  460|           bodyLift += ease * 0.06 * ia;
  461|           this.squash *= 1 + ease * 0.10 * ia;
  462|           break;
  463|         default: break;
  464|       }
  465|     }
  466| 
  467|     // ---- attention: gaze, blink --------------------------------------------
  468|     this.nextLook -= dt;
  469|     if (this.nextLook <= 0) {
  470|       this.nextLook = 0.5 + Math.random() * 1.5;
  471|       this.lookX = this.deterministic ? 0 : (Math.random() - 0.5) * 0.9;
  472|       this.lookY = this.deterministic ? 0 : (Math.random() - 0.5) * 0.5;
  473|     }
  474|     if (lookAt) {
  475|       const eye = this._v.setFromMatrixPosition(h.eyes[0]?.root.matrixWorld ?? h.group.matrixWorld);
  476|       const local = eye.clone().sub(h.group.getWorldPosition(new THREE.Vector3()));
  477|       const yaw = Math.atan2(local.x, local.z);
  478|       const dist = Math.max(0.3, local.length());
  479|       const pitch = Math.atan2(lookAt.y - eye.y, dist);
  480|       this.lookX = THREE.MathUtils.clamp(Math.atan2(Math.sin(yaw), Math.cos(yaw)), -0.9, 0.9);
  481|       this.lookY = THREE.MathUtils.clamp(pitch, -0.5, 0.5);
  482|     }
  483|     // The eyes lead: they snap onto the target FIRST, the head arrives 0.15 s
  484|     // later (drive.headLead is already delayed by the Agent), the body last.
  485|     if (this.driven && Math.abs(this.drive.eyeLead) > 0.01) {
  486|       this.lookX = THREE.MathUtils.clamp(this.lookX + this.drive.eyeLead, -1.1, 1.1);
  487|     }
  488|     this.gazeYaw += (this.lookX - this.gazeYaw) * Math.min(1, dt * 12);
  489|     this.gazePitch += (this.lookY - this.gazePitch) * Math.min(1, dt * 12);
  490| 
  491|     this.blinkTimer -= dt;
  492|     if (this.blinkTimer <= 0) {
  493|       const sleepy = this.mood === 'sleepy';
  494|       this.blinkT = 1;
  495|       this.blinkTimer = (sleepy ? 1.4 : 2.0 + Math.random() * 4);
  496|       if (this.doubleBlink > 0) { this.doubleBlink = 0; this.blinkTimer = 0.16; }
  497|       else if (!this.deterministic && Math.random() < 0.16) this.doubleBlink = 1;
  498|       this.blinkSide = !this.deterministic && Math.random() < 0.08 ? (Math.random() < 0.5 ? -1 : 1) : 0;
  499|     }
