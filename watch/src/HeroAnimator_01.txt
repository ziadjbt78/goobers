  334|   update(dt: number, lookAt: THREE.Vector3 | null): void {
  335|     const h = this.handle;
  336|     const t = this.template;
  337|     this.t += dt;
  338|     this.excited = Math.max(0, this.excited - dt);
  339|     this.actionT += dt;
  340| 
  341|     // ---- layer weights -----------------------------------------------------
  342|     const jumpy = t.locomotion === 'hop';
  343|     const moving = this.driven && !this.external ? (this.drive.speed > 0.05 ? 1 : 0) : (this.gait > 0.02 ? 1 : 0);
  344|     const locoTarget = this.driven || this.external ? moving : 0;
  345|     this.locomotionW += (locoTarget - this.locomotionW) * Math.min(1, dt / 0.25);
  346|     const loco = this.driven || this.external ? this.locomotionW : 0;
  347|     const actTarget = this.action === 'none' ? 0 : 1;
  348|     this.actionBlend += (actTarget - this.actionBlend) * Math.min(1, dt / 0.25);
  349|     const act = this.actionBlend;
  350| 
  351|     // ---- ambient idle behaviour: the creature is never frozen --------------
  352|     // in external mode the motion layer plays ambient behaviour through Choreo
  353|     const standingStill = !this.external && loco < 0.15 && act < 0.2;
  354|     if (standingStill && this.idleAct === 'none') {
  355|       this.idleTimer -= dt;
  356|       if (this.idleTimer <= 0) {
  357|         const pool: IdleAct[] = ['sniff', 'scratch', 'sit', 'yawn', 'sniff'];
  358|         const pick = this.deterministic ? 'sniff' : pool[Math.floor(Math.random() * pool.length)];
  359|         this.idleAct = pick;
  360|         this.idleActT = pick === 'sit' ? 2.4 : 1.3;
  361|         this.idleTimer = 4 + Math.random() * 6;
  362|       }
  363|     }
  364|     if (this.idleAct !== 'none') {
  365|       this.idleActT -= dt;
  366|       if (this.idleActT <= 0) { this.idleAct = 'none'; this.idleActW = 0; }
  367|     }
  368|     const idleTarget = this.idleAct !== 'none' && standingStill ? 1 : 0;
  369|     this.idleActW += (idleTarget - this.idleActW) * Math.min(1, dt / 0.25);
  370|     const ia = this.idleActW;
  371| 
  372|     // ---- the shared clock --------------------------------------------------
  373|     const speed01 = this.driven && !this.external ? Math.min(1, this.drive.speed / 1.4) : this.gait;
  374|     if (this.external) {
  375|       this.phase = this.externalPhase;
  376|     } else if (this.driven) {
  377|       if (jumpy) this.phase += dt * (1.2 + speed01 * 2.6);
  378|       else {
  379|         const stride = Math.max(0.16, this.legs.length ? this.legs[0].step * 2.1 * Math.max(1, this.bulkScale * 0.5) : 0.4);
  380|         this.phase = (this.phase + (this.drive.speed * dt) / stride) % 1;
  381|       }
  382|     } else {
  383|       this.phase += dt * (jumpy ? 1.3 + this.gait * 2.2 : 1.1 + this.gait * 2.4);
  384|     }
  385| 
  386|     // ---- body ---------------------------------------------------------------
  387|     let bodyLift = 0;
  388|     let bodyPitch = 0;
  389|     let bodyRoll = 0;
  390| 
  391|     const beatLegs = this.template.feet.length > 4 ? 3 : 2;
  392|     const beat = Math.abs(Math.sin(this.phase * Math.PI * 2 * beatLegs));
  393| 
  394|     if (this.external) {
  395|       // The motion layer writes lift / squash / pitch / roll / yaw on the pivot
  396|       // and body bones. The animator adds NOTHING here, so no channel is
  397|       // applied twice.
  398|       bodyLift = 0;
  399|       bodyPitch = 0;
  400|       bodyRoll = 0;
  401|       this.squash = 1;
  402|       this.bodyYaw = 0;
  403|       this.headPitch = 0;
  404|     } else if (this.driven) {
  405|       bodyLift = this.drive.lift;
  406|       bodyPitch = this.drive.pitch + this.drive.slopePitch;
  407|       bodyRoll = -this.drive.lean + this.drive.slopeRoll;
  408|       this.squash = this.drive.squash;
  409|       this.bodyYaw = this.drive.yaw;
  410|       this.headPitch = 0;
  411|     } else if (jumpy) {
  412|       this.hopV -= 22 * dt;
  413|       this.hopY += this.hopV * dt;
  414|       if (this.hopY < 0) {
  415|         this.hopY = 0;
  416|         if (this.hopV < -0.6) this.squashV += -this.hopV * 1.6;
  417|         this.hopV = 0;
  418|       }
  419|       if (this.hopY === 0 && moving > 0) {
  420|         const push = Math.max(0, Math.sin(this.phase * Math.PI * 2 * 0.5 + 0.4));
  421|         if (push > 0.97) this.hopV = 2.6 + this.gait * 2.4;
  422|       }
  423|       bodyLift = this.hopY;
  424|       const airborne = this.hopY > 0.001 ? 1 : 0;
  425|       this.squashV += (-(this.squash - 1) * 120 - this.squashV * 14) * dt;
  426|       this.squash += this.squashV * dt;
  427|       this.squash = Math.max(0.72, Math.min(1.22, this.squash));
  428|       bodyPitch = airborne * -0.06;
  429|     } else {
  430|       const bob = (beat - 0.5) * 0.028 * this.gait;
  431|       const idle = Math.sin(this.t * 1.6) * 0.006 * (1 - moving);
  432|       bodyLift = bob + idle + this.excited * Math.abs(Math.sin(this.t * 11)) * 0.05;
  433|       this.squash += (1 - this.squash) * Math.min(1, dt * 9);
  434|       bodyLift += -this.squashV * 0.004;
  435|       this.squashV *= Math.max(0, 1 - dt * 8);
  436|     }
  437| 
  438|     if (this.driven && !this.external) bodyLift += Math.sin(this.t * 1.6) * 0.006;
  439|     bodyLift -= loco * 0.012;
  440|     if (this.external) {
  441|       bodyLift = 0; bodyPitch = 0; bodyRoll = 0;
  442|       this.squash = 1; this.bodyYaw = 0; this.headPitch = 0;
  443|     }
  444| 
  445|     // ---- ambient idle poses, layered on the base ---------------------------
  446|     if (ia > 0.001) {
  447|       const u = this.idleAct === 'none' ? 0 : Math.min(1, this.idleActT > 0.001 ? 1 - this.idleActT / 1.3 : 1);
  448|       const ease = Math.sin(Math.min(1, u) * Math.PI);
  449|       switch (this.idleAct) {
  450|         case 'sniff':
  451|           bodyPitch += ease * 0.42 * ia;                       // head down to the ground
  452|           bodyLift -= ease * 0.05 * ia;
  453|           break;
  454|         case 'scratch':
  455|           bodyRoll += Math.sin(this.idleActT * 26) * 0.20 * ia;
  456|           bodyPitch += ease * 0.10 * ia;
  457|           break;
  458|         case 'sit':
  459|           bodyPitch -= ease * 0.34 * ia;                       // rock back on the haunches
  460|           bodyLift -= ease * 0.10 * ia;
  461|           this.squash *= 1 - ease * 0.14 * ia;
  462|           break;
  463|         case 'yawn':
  464|           bodyPitch -= ease * 0.22 * ia;                       // big stretch up
  465|           bodyLift += ease * 0.06 * ia;
  466|           this.squash *= 1 + ease * 0.10 * ia;
  467|           break;
  468|         default: break;
  469|       }
  470|     }
  471| 
  472|     // ---- attention: gaze, blink --------------------------------------------
  473|     this.nextLook -= dt;
  474|     if (this.nextLook <= 0) {
  475|       this.nextLook = 0.5 + Math.random() * 1.5;
  476|       this.lookX = this.deterministic ? 0 : (Math.random() - 0.5) * 0.9;
  477|       this.lookY = this.deterministic ? 0 : (Math.random() - 0.5) * 0.5;
  478|     }
  479|     if (lookAt) {
  480|       const eye = this._v.setFromMatrixPosition(h.eyes[0]?.root.matrixWorld ?? h.group.matrixWorld);
  481|       const local = eye.clone().sub(h.group.getWorldPosition(new THREE.Vector3()));
  482|       const yaw = Math.atan2(local.x, local.z);
  483|       const dist = Math.max(0.3, local.length());
  484|       const pitch = Math.atan2(lookAt.y - eye.y, dist);
  485|       this.lookX = THREE.MathUtils.clamp(Math.atan2(Math.sin(yaw), Math.cos(yaw)), -0.9, 0.9);
  486|       this.lookY = THREE.MathUtils.clamp(pitch, -0.5, 0.5);
  487|     }
  488|     // The eyes lead: they snap onto the target FIRST, the head arrives 0.15 s
  489|     // later (drive.headLead is already delayed by the Agent), the body last.
  490|     if (this.driven && Math.abs(this.drive.eyeLead) > 0.01) {
  491|       this.lookX = THREE.MathUtils.clamp(this.lookX + this.drive.eyeLead, -1.1, 1.1);
  492|     }
  493|     this.gazeYaw += (this.lookX - this.gazeYaw) * Math.min(1, dt * 12);
  494|     this.gazePitch += (this.lookY - this.gazePitch) * Math.min(1, dt * 12);
  495| 
  496|     this.blinkTimer -= dt;
  497|     if (this.blinkTimer <= 0) {
  498|       const sleepy = this.mood === 'sleepy';
  499|       this.blinkT = 1;
  500|       this.blinkTimer = (sleepy ? 1.4 : 2.0 + Math.random() * 4);
