  323|   update(dt: number, lookAt: THREE.Vector3 | null): void {
  324|     const h = this.handle;
  325|     const t = this.template;
  326|     this.t += dt;
  327|     this.excited = Math.max(0, this.excited - dt);
  328|     this.actionT += dt;
  329| 
  330|     // ---- layer weights -----------------------------------------------------
  331|     const jumpy = t.locomotion === 'hop';
  332|     const moving = this.driven && !this.external ? (this.drive.speed > 0.05 ? 1 : 0) : (this.gait > 0.02 ? 1 : 0);
  333|     const locoTarget = this.driven || this.external ? moving : 0;
  334|     this.locomotionW += (locoTarget - this.locomotionW) * Math.min(1, dt / 0.25);
  335|     const loco = this.driven || this.external ? this.locomotionW : 0;
  336|     const actTarget = this.action === 'none' ? 0 : 1;
  337|     this.actionBlend += (actTarget - this.actionBlend) * Math.min(1, dt / 0.25);
  338|     const act = this.actionBlend;
  339| 
  340|     // ---- ambient idle behaviour: the creature is never frozen --------------
  341|     // in external mode the motion layer plays ambient behaviour through Choreo
  342|     const standingStill = !this.external && loco < 0.15 && act < 0.2;
  343|     if (standingStill && this.idleAct === 'none') {
  344|       this.idleTimer -= dt;
  345|       if (this.idleTimer <= 0) {
  346|         const pool: IdleAct[] = ['sniff', 'scratch', 'sit', 'yawn', 'sniff'];
  347|         const pick = this.deterministic ? 'sniff' : pool[Math.floor(Math.random() * pool.length)];
  348|         this.idleAct = pick;
  349|         this.idleActT = pick === 'sit' ? 2.4 : 1.3;
  350|         this.idleTimer = 4 + Math.random() * 6;
  351|       }
  352|     }
  353|     if (this.idleAct !== 'none') {
  354|       this.idleActT -= dt;
  355|       if (this.idleActT <= 0) { this.idleAct = 'none'; this.idleActW = 0; }
  356|     }
  357|     const idleTarget = this.idleAct !== 'none' && standingStill ? 1 : 0;
  358|     this.idleActW += (idleTarget - this.idleActW) * Math.min(1, dt / 0.25);
  359|     const ia = this.idleActW;
  360| 
  361|     // ---- the shared clock --------------------------------------------------
  362|     const speed01 = this.driven && !this.external ? Math.min(1, this.drive.speed / 1.4) : this.gait;
  363|     if (this.external) {
  364|       this.phase = this.externalPhase;
  365|     } else if (this.driven) {
  366|       if (jumpy) this.phase += dt * (1.2 + speed01 * 2.6);
  367|       else {
  368|         const stride = Math.max(0.16, this.legs.length ? this.legs[0].step * 2.1 * Math.max(1, this.bulkScale * 0.5) : 0.4);
  369|         this.phase = (this.phase + (this.drive.speed * dt) / stride) % 1;
  370|       }
  371|     } else {
  372|       this.phase += dt * (jumpy ? 1.3 + this.gait * 2.2 : 1.1 + this.gait * 2.4);
  373|     }
  374| 
  375|     // ---- body ---------------------------------------------------------------
  376|     let bodyLift = 0;
  377|     let bodyPitch = 0;
  378|     let bodyRoll = 0;
  379| 
  380|     const beatLegs = this.template.feet.length > 4 ? 3 : 2;
  381|     const beat = Math.abs(Math.sin(this.phase * Math.PI * 2 * beatLegs));
  382| 
  383|     if (this.external) {
  384|       // The motion layer writes lift / squash / pitch / roll / yaw on the pivot
  385|       // and body bones. The animator adds NOTHING here, so no channel is
  386|       // applied twice.
  387|       bodyLift = 0;
  388|       bodyPitch = 0;
  389|       bodyRoll = 0;
  390|       this.squash = 1;
  391|       this.bodyYaw = 0;
  392|       this.headPitch = 0;
  393|     } else if (this.driven) {
  394|       bodyLift = this.drive.lift;
  395|       bodyPitch = this.drive.pitch + this.drive.slopePitch;
  396|       bodyRoll = -this.drive.lean + this.drive.slopeRoll;
  397|       this.squash = this.drive.squash;
  398|       this.bodyYaw = this.drive.yaw;
  399|       this.headPitch = 0;
  400|     } else if (jumpy) {
  401|       this.hopV -= 22 * dt;
  402|       this.hopY += this.hopV * dt;
  403|       if (this.hopY < 0) {
  404|         this.hopY = 0;
  405|         if (this.hopV < -0.6) this.squashV += -this.hopV * 1.6;
  406|         this.hopV = 0;
  407|       }
  408|       if (this.hopY === 0 && moving > 0) {
  409|         const push = Math.max(0, Math.sin(this.phase * Math.PI * 2 * 0.5 + 0.4));
  410|         if (push > 0.97) this.hopV = 2.6 + this.gait * 2.4;
  411|       }
  412|       bodyLift = this.hopY;
  413|       const airborne = this.hopY > 0.001 ? 1 : 0;
  414|       this.squashV += (-(this.squash - 1) * 120 - this.squashV * 14) * dt;
  415|       this.squash += this.squashV * dt;
  416|       this.squash = Math.max(0.72, Math.min(1.22, this.squash));
  417|       bodyPitch = airborne * -0.06;
  418|     } else {
  419|       const bob = (beat - 0.5) * 0.028 * this.gait;
  420|       const idle = Math.sin(this.t * 1.6) * 0.006 * (1 - moving);
  421|       bodyLift = bob + idle + this.excited * Math.abs(Math.sin(this.t * 11)) * 0.05;
  422|       this.squash += (1 - this.squash) * Math.min(1, dt * 9);
  423|       bodyLift += -this.squashV * 0.004;
  424|       this.squashV *= Math.max(0, 1 - dt * 8);
  425|     }
  426| 
  427|     if (this.driven && !this.external) bodyLift += Math.sin(this.t * 1.6) * 0.006;
  428|     bodyLift -= loco * 0.012;
  429|     if (this.external) {
  430|       bodyLift = 0; bodyPitch = 0; bodyRoll = 0;
  431|       this.squash = 1; this.bodyYaw = 0; this.headPitch = 0;
  432|     }
  433| 
  434|     // ---- ambient idle poses, layered on the base ---------------------------
  435|     if (ia > 0.001) {
  436|       const u = this.idleAct === 'none' ? 0 : Math.min(1, this.idleActT > 0.001 ? 1 - this.idleActT / 1.3 : 1);
  437|       const ease = Math.sin(Math.min(1, u) * Math.PI);
  438|       switch (this.idleAct) {
  439|         case 'sniff':
  440|           bodyPitch += ease * 0.42 * ia;                       // head down to the ground
  441|           bodyLift -= ease * 0.05 * ia;
  442|           break;
  443|         case 'scratch':
  444|           bodyRoll += Math.sin(this.idleActT * 26) * 0.20 * ia;
  445|           bodyPitch += ease * 0.10 * ia;
  446|           break;
  447|         case 'sit':
  448|           bodyPitch -= ease * 0.34 * ia;                       // rock back on the haunches
  449|           bodyLift -= ease * 0.10 * ia;
  450|           this.squash *= 1 - ease * 0.14 * ia;
  451|           break;
  452|         case 'yawn':
  453|           bodyPitch -= ease * 0.22 * ia;                       // big stretch up
  454|           bodyLift += ease * 0.06 * ia;
  455|           this.squash *= 1 + ease * 0.10 * ia;
  456|           break;
  457|         default: break;
  458|       }
  459|     }
  460| 
  461|     // ---- attention: gaze, blink --------------------------------------------
  462|     this.nextLook -= dt;
  463|     if (this.nextLook <= 0) {
  464|       this.nextLook = 0.5 + Math.random() * 1.5;
  465|       this.lookX = this.deterministic ? 0 : (Math.random() - 0.5) * 0.9;
  466|       this.lookY = this.deterministic ? 0 : (Math.random() - 0.5) * 0.5;
  467|     }

  469|       const eye = this._v.setFromMatrixPosition(h.eyes[0]?.root.matrixWorld ?? h.group.matrixWorld);
  470|       const local = eye.clone().sub(h.group.getWorldPosition(new THREE.Vector3()));
  471|       const yaw = Math.atan2(local.x, local.z);
  472|       const dist = Math.max(0.3, local.length());
  473|       const pitch = Math.atan2(lookAt.y - eye.y, dist);
  474|       this.lookX = THREE.MathUtils.clamp(Math.atan2(Math.sin(yaw), Math.cos(yaw)), -0.9, 0.9);
  475|       this.lookY = THREE.MathUtils.clamp(pitch, -0.5, 0.5);
  476|     }
  477|     // The eyes lead: they snap onto the target FIRST, the head arrives 0.15 s
  478|     // later (drive.headLead is already delayed by the Agent), the body last.
  479|     if (this.driven && Math.abs(this.drive.eyeLead) > 0.01) {
  480|       this.lookX = THREE.MathUtils.clamp(this.lookX + this.drive.eyeLead, -1.1, 1.1);
  481|     }

  483|     this.gazePitch += (this.lookY - this.gazePitch) * Math.min(1, dt * 12);

  491|       else if (!this.deterministic && Math.random() < 0.16) this.doubleBlink = 1;
  492|       this.blinkSide = !this.deterministic && Math.random() < 0.08 ? (Math.random() < 0.5 ? -1 : 1) : 0;
  493|     }
