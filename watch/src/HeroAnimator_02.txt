  501|       const sleepy = this.mood === 'sleepy';
  502|       this.blinkT = 1;
  503|       this.blinkTimer = (sleepy ? 1.4 : 2.0 + Math.random() * 4);
  504|       if (this.doubleBlink > 0) { this.doubleBlink = 0; this.blinkTimer = 0.16; }
  505|       else if (!this.deterministic && Math.random() < 0.16) this.doubleBlink = 1;
  506|       this.blinkSide = !this.deterministic && Math.random() < 0.08 ? (Math.random() < 0.5 ? -1 : 1) : 0;
  507|     }
  508|     if (this.blinkT > 0) this.blinkT = Math.max(0, this.blinkT - dt * 7);
  509| 
  510|     if (!this.driven) {
  511|       this.nextTurn -= dt;
  512|       if (this.nextTurn <= 0) {
  513|         this.nextTurn = 1.6 + Math.random() * 3;
  514|         this.bodyYawTarget = this.deterministic ? 0 : (Math.random() - 0.5) * 0.55;
  515|       }
  516|       this.headYawIdle = THREE.MathUtils.clamp(this.bodyYawTarget - this.bodyYaw, -0.55, 0.55) * 0.9;
  517|       this.bodyYaw += (this.bodyYawTarget - this.bodyYaw) * Math.min(1, dt * 1.5);
  518|       this.headPitch = 0;
  519|     }
  520|     const headLead = this.driven ? this.drive.headLead : this.headYawIdle;
  521| 
  522|     // ---- action layer -------------------------------------------------------
  523|     let actionPitch = 0;
  524|     let actionLift = 0;
  525|     let actionSquash = 1;
  526|     let actionRoll = 0;
  527|     if (act > 0.001) {
  528|       const a = this._actionPose(this.action, this.actionT);
  529|       actionPitch = a.pitch; actionLift = a.lift; actionSquash = a.squash; actionRoll = a.roll;
  530|     }
  531| 
  532|     // the Face map owns every face mesh in external mode (RigAdapter.setFace)
  533|     if (!this.external) this._updateFace(dt, act);
  534| 
  535|     // ---- write the body ----------------------------------------------------
  536|     const root = h.bones[0];
  537|     const body = h.bones[t.bodyBone];
  538|     const head = h.bones[t.headBone];
  539|     const breathe = 1 + Math.sin(this.t * 2.4) * 0.02;
  540|     const idle = Math.sin(this.t * 1.6) * 0.006;
  541|     if (this.external) root.position.set(0, 0, 0);
  542|     else if (!this.driven) root.position.set(0, bodyLift - idle * 0.5, 0);
  543|     else root.position.set(0, bodyLift - idle * 0.5 + actionLift * act, 0);
  544| 
  545|     const style = this._gaitFlavour(loco, speed01);
  546|     const roll = bodyRoll + style.roll + actionRoll * act;
  547|     this._q.setFromEuler(this._e.set(
  548|       bodyPitch + actionPitch * act,
  549|       this.bodyYaw + (this.driven ? headLead * 0.25 : 0),
  550|       roll,
  551|     ));
  552|     body.quaternion.copy(this._q);
  553| 
  554|     const sq = this.squash * (1 + (actionSquash - 1) * act);
  555|     body.scale.set(1 / Math.sqrt(Math.max(0.2, sq)), sq * breathe, 1 / Math.sqrt(Math.max(0.2, sq)));
  556| 
  557|     for (let i = 0; i < this.spine.length; i++) {
  558|       const b = this.spine[i];
  559|       const lag = i * 0.16;
  560|       const w = Math.sin((this.phase - lag) * Math.PI * 2) * style.spine * (1 - i * 0.25);
  561|       b.rotateX(w);
  562|     }
  563| 
  564|     head.quaternion.setFromEuler(this._e.set(
  565|       this.headPitch - bodyPitch * 0.5 + style.headPitch - actionPitch * act * 0.6,
  566|       headLead,
  567|       0,
  568|     ));
  569| 
  570|     // ---- feet --------------------------------------------------------------
  571|     // v12: the World path solves its legs LAST (Agent -> solveWorldLegs), after
  572|     // root, heading and body juice are final. Solving here, before those moved,
  573|     // is what dragged planted feet and popped knees.
  574|     if (!this.external) this._solveLegsPedestal(dt, moving);
  575| 
  576|     // ---- a petted creature kicks a back leg ---------------------------------
  577|     if (this.legs.length > 0) {
  578|       this.kickT = this.petted ? this.kickT + dt : 0;
  579|       if (this.petted) {
  580|         const back = this.legs[this.legs.length - 1];
  581|         const kick = Math.max(0, Math.sin(this.kickT * 9)) * 0.85;
  582|         if (kick > 0.02) back.hip.rotateX(-kick);
  583|       }
  584|     }
  585| 
  586|     // ---- arms swing opposite the legs, and wave when greeting --------------
  587|     for (const a of this.arms) {
  588|       let sw = 0;
  589|       if (loco > 0.01) sw = Math.sin(this.phase * Math.PI * 2 + (a.side > 0 ? Math.PI : 0)) * 0.52 * loco;
  590|       if (this.action === 'wave' || this.action === 'greet') {
  591|         sw += act * (a.side > 0 ? Math.sin(this.t * 13) * 0.9 + 0.7 : 0.1);
  592|       }
  593|       if (this.held) sw += Math.sin(this.t * 15 + a.side) * 0.55;
  594|       a.upper.rotateX(sw);
  595|       a.lower.rotateX(Math.max(0, -sw) * 0.5);
  596|     }
  597| 
  598|     // ---- wobble springs on ears, tails, antennae ---------------------------
  599|     const accel = this.driven ? this.drive.speed * 0.9 : this.gait * 0.6;
  600|     const driveW = -this.bodyYaw * 0.6 + bodyPitch * 2.4 + (accel * 0.10 * Math.sin(this.phase * Math.PI * 4));
  601|     for (const w of this.wiggles) {
  602|       const extra = this.template.locomotion === 'skitter' ? Math.sin(this.t * 17 + w.gain) * 0.04 * loco : 0;
  603|       const kick = this.mood === 'happy' ? Math.sin(this.t * 9) * 0.12 : 0;
  604|       const x = w.w.step((driveW + kick) * w.gain, Math.min(dt, 1 / 60));
  605|       w.bone.quaternion.setFromAxisAngle(w.axis === 'y' ? Y : X, x * 0.9 + extra);
  606|     }
  607|   }

  610|   /**
  611|    * v10 — world-space foot planting for the World path.
  612|    *
  613|    * The pedestal cycle moves a body-space target, which is why a planted foot
  614|    * used to travel with the body (25 cm of slide). Here the foot is instead
  615|    * pinned to a WORLD point for its whole stance, and the swing is a
  616|    * predicted-arc step to the next plant. Guards force an early lift before
  617|    * anything can overstretch or slide: leg reach, rest-offset error, turn
  618|    * error, and idle body drift.
  619|    */
  620|   /** Pedestal / driven leg solve, moved here VERBATIM from update(). */
  621|   private _solveLegsPedestal(dt: number, moving: number): void {
  622|     const h = this.handle;
  623|     // ---- feet --------------------------------------------------------------
  624|     if (this.legs.length > 0) {
  625|       h.group.updateMatrixWorld(true);
  626|       const world = new THREE.Matrix4().copy(h.group.matrixWorld);
  627|       // The IK pole must live in the CREATURE's frame, not the world's: a knee
  628|       // that always bends toward world +Z folds backwards the moment the
  629|       // creature faces any other way. On a pedestal the creature never turns,
  630|       // so the pole stays exactly the old world +Z and the baseline holds.
  631|       const drivenRig = this.driven || this.external;
  632|       const groupWorldQuat = drivenRig ? h.group.getWorldQuaternion(this._q) : null;
  633|       for (let i = 0; i < this.legs.length; i++) {
  634|         const L = this.legs[i];
  635|         if (drivenRig && groupWorldQuat) {
  636|           // per-leg pole in creature space, rotated into the world, then eased.
  637|           // An unsmoothed pole that grazes the limb axis is exactly what makes a
  638|           // two-bone solve flip; the ease removes that.
  639|           this._pole.copy(L.poleLocal).applyQuaternion(groupWorldQuat);
  640|           if (L.poleSm.lengthSq() < 1e-6) L.poleSm.copy(this._pole);
  641|           else L.poleSm.lerp(this._pole, 1 - Math.exp(-dt / 0.08)).normalize();
  642|           this._pole.copy(L.poleSm);
  643|         } else {
  644|           this._pole.set(0, 0, 1);
  645|         }
  646|         const target = new THREE.Vector3();
  647|         if (this.external) {
  648|           // v10 WORLD PLANTING: the foot is pinned to a world point for the
  649|           // whole stance, so it cannot slide. See _worldFootTarget.
  650|           this._worldFootTarget(L, target, dt);
  651|         } else if (this.driven && this.drive.footTargets[L.index]) {
  652|           target.copy(this.drive.footTargets[L.index]);
  653|         } else {
  654|           const u = (this.phase + L.phase) % 1;
  655|           let zOff = 0;
  656|           let yOff = 0;
  657|           if (moving > 0) {
  658|             if (u < 0.5) zOff = (0.5 - u * 2) * L.step;
  659|             else {
  660|               const s = (u - 0.5) * 2;
  661|               zOff = (s * 2 - 1) * L.step * 0.5;
  662|               yOff = Math.sin(s * Math.PI) * L.lift;
  663|             }
  664|           }
  665|           target.set(L.plant.x, L.plant.y + yOff * moving, L.plant.z + zOff * moving);
  666|           target.applyMatrix4(world);
  667|         }
  668|         solveLeg({
  669|           hip: L.hip, knee: L.knee, ankle: L.ankle,
  670|           targetWorld: target,
  671|           poleWorld: this._pole,
  672|           prevKnee: L.kneeLocal,
  673|         });
  674|         // A two-bone solve flips branch when the pole crosses the limb axis, and
  675|         // a splayed insect leg does that constantly. An airborne leg is capped
  676|         // at 12 rad/s so a flip reads as a fast move, never a pop. Planted legs
  677|         // are exempt, so foot slip stays exactly zero.
  678|         //
  679|         // v9: EXTERNAL mode derives the swing from the external cycle itself,
  680|         // and the plant-transition frame counts as airborne (that is the frame
  681|         // the solve is most prone to snap on).
  682|         const airborne = this.external ? L.mode === 'swing' : false;
  683|         const swingNow = this.external
  684|           ? (airborne || L.wasSwing)
  685|           : !!(this.driven && this.drive.footSwing[L.index]);
  686|         if (this.external) L.wasSwing = airborne;
  687|         if ((this.driven || this.external) && swingNow) {
  688|           const maxD = LEG_RATE * dt;
  689|           for (let k = 0; k < 3; k++) {
  690|             const bone = k === 0 ? L.hip : k === 1 ? L.knee : L.ankle;
  691|             const q = L.prev[k];
  692|             const ang = 2 * Math.acos(Math.min(1, Math.abs(bone.quaternion.dot(q))));
  693|             if (ang > maxD && ang > 1e-6) {
  694|               _lim.copy(bone.quaternion);
  695|               bone.quaternion.copy(q).slerp(_lim, maxD / ang);
  696|             }
  697|           }
  698|         }
  699|         L.prev[0].copy(L.hip.quaternion);
  700|         L.prev[1].copy(L.knee.quaternion);
  701|         L.prev[2].copy(L.ankle.quaternion);
  702|       }
  703|       void world;
  704|     }
  705| 
  706|   }
