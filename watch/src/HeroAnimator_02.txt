  501| 
  502|     if (!this.driven) {

  507|       }

  509|       this.bodyYaw += (this.bodyYawTarget - this.bodyYaw) * Math.min(1, dt * 1.5);
  510|       this.headPitch = 0;
  511|     }

  513| 
  514|     // ---- action layer -------------------------------------------------------

  516|     let actionLift = 0;
  517|     let actionSquash = 1;
  518|     let actionRoll = 0;
  519|     if (act > 0.001) {
  520|       const a = this._actionPose(this.action, this.actionT);
  521|       actionPitch = a.pitch; actionLift = a.lift; actionSquash = a.squash; actionRoll = a.roll;
  522|     }

  524|     // the Face map owns every face mesh in external mode (RigAdapter.setFace)
  525|     if (!this.external) this._updateFace(dt, act);
  526| 
  527|     // ---- write the body ----------------------------------------------------
  528|     const root = h.bones[0];
  529|     const body = h.bones[t.bodyBone];
  530|     const head = h.bones[t.headBone];
  531|     const breathe = 1 + Math.sin(this.t * 2.4) * 0.02;
  532|     const idle = Math.sin(this.t * 1.6) * 0.006;
  533|     if (this.external) root.position.set(0, 0, 0);
  534|     else if (!this.driven) root.position.set(0, bodyLift - idle * 0.5, 0);
  535|     else root.position.set(0, bodyLift - idle * 0.5 + actionLift * act, 0);
  536| 
  537|     const style = this._gaitFlavour(loco, speed01);
  538|     const roll = bodyRoll + style.roll + actionRoll * act;
  539|     this._q.setFromEuler(new THREE.Euler(
  540|       bodyPitch + actionPitch * act,
  541|       this.bodyYaw + (this.driven ? headLead * 0.25 : 0),
  542|       roll,
  543|     ));
  544|     body.quaternion.copy(this._q);
  545| 
  546|     const sq = this.squash * (1 + (actionSquash - 1) * act);
  547|     body.scale.set(1 / Math.sqrt(Math.max(0.2, sq)), sq * breathe, 1 / Math.sqrt(Math.max(0.2, sq)));
  548| 
  549|     for (let i = 0; i < this.spine.length; i++) {
  550|       const b = this.spine[i];
  551|       const lag = i * 0.16;
  552|       const w = Math.sin((this.phase - lag) * Math.PI * 2) * style.spine * (1 - i * 0.25);
  553|       b.rotateX(w);
  554|     }
  555| 
  556|     head.quaternion.setFromEuler(new THREE.Euler(
  557|       this.headPitch - bodyPitch * 0.5 + style.headPitch - actionPitch * act * 0.6,
  558|       headLead,
  559|       0,
  560|     ));
  561| 
  562|     // ---- feet --------------------------------------------------------------
  563|     // v12: the World path solves its legs LAST (Agent -> solveWorldLegs), after
  564|     // root, heading and body juice are final. Solving here, before those moved,
  565|     // is what dragged planted feet and popped knees.
  566|     if (!this.external) this._solveLegsPedestal(dt, moving);
  567| 
  568|     // ---- a petted creature kicks a back leg ---------------------------------
  569|     if (this.legs.length > 0) {
  570|       this.kickT = this.petted ? this.kickT + dt : 0;
  571|       if (this.petted) {
  572|         const back = this.legs[this.legs.length - 1];
  573|         const kick = Math.max(0, Math.sin(this.kickT * 9)) * 0.85;
  574|         if (kick > 0.02) back.hip.rotateX(-kick);
  575|       }
  576|     }
  577| 
  578|     // ---- arms swing opposite the legs, and wave when greeting --------------
  579|     for (const a of this.arms) {
  580|       let sw = 0;
  581|       if (loco > 0.01) sw = Math.sin(this.phase * Math.PI * 2 + (a.side > 0 ? Math.PI : 0)) * 0.52 * loco;
  582|       if (this.action === 'wave' || this.action === 'greet') {
  583|         sw += act * (a.side > 0 ? Math.sin(this.t * 13) * 0.9 + 0.7 : 0.1);
  584|       }
  585|       if (this.held) sw += Math.sin(this.t * 15 + a.side) * 0.55;
  586|       a.upper.rotateX(sw);
  587|       a.lower.rotateX(Math.max(0, -sw) * 0.5);
  588|     }
  589| 
  590|     // ---- wobble springs on ears, tails, antennae ---------------------------
  591|     const accel = this.driven ? this.drive.speed * 0.9 : this.gait * 0.6;
  592|     const driveW = -this.bodyYaw * 0.6 + bodyPitch * 2.4 + (accel * 0.10 * Math.sin(this.phase * Math.PI * 4));
  593|     for (const w of this.wiggles) {

  595|       const kick = this.mood === 'happy' ? Math.sin(this.t * 9) * 0.12 : 0;
  596|       const x = w.w.step((driveW + kick) * w.gain, Math.min(dt, 1 / 60));
  597|       w.bone.quaternion.setFromAxisAngle(w.axis === 'y' ? Y : X, x * 0.9 + extra);
  598|     }
  599|   }

  602|   /**
  603|    * v10 — world-space foot planting for the World path.
  604|    *
  605|    * The pedestal cycle moves a body-space target, which is why a planted foot
  606|    * used to travel with the body (25 cm of slide). Here the foot is instead
  607|    * pinned to a WORLD point for its whole stance, and the swing is a
  608|    * predicted-arc step to the next plant. Guards force an early lift before
  609|    * anything can overstretch or slide: leg reach, rest-offset error, turn
  610|    * error, and idle body drift.
  611|    */
  612|   /** Pedestal / driven leg solve, moved here VERBATIM from update(). */
  613|   private _solveLegsPedestal(dt: number, moving: number): void {
  614|     const h = this.handle;
  615|     // ---- feet --------------------------------------------------------------
  616|     if (this.legs.length > 0) {
  617|       h.group.updateMatrixWorld(true);
  618|       const world = new THREE.Matrix4().copy(h.group.matrixWorld);
  619|       // The IK pole must live in the CREATURE's frame, not the world's: a knee
  620|       // that always bends toward world +Z folds backwards the moment the
  621|       // creature faces any other way. On a pedestal the creature never turns,
  622|       // so the pole stays exactly the old world +Z and the baseline holds.
  623|       const drivenRig = this.driven || this.external;
  624|       const groupWorldQuat = drivenRig ? h.group.getWorldQuaternion(this._q) : null;
  625|       for (let i = 0; i < this.legs.length; i++) {
  626|         const L = this.legs[i];
  627|         if (drivenRig && groupWorldQuat) {
  628|           // per-leg pole in creature space, rotated into the world, then eased.
  629|           // An unsmoothed pole that grazes the limb axis is exactly what makes a
  630|           // two-bone solve flip; the ease removes that.
  631|           this._pole.copy(L.poleLocal).applyQuaternion(groupWorldQuat);
  632|           if (L.poleSm.lengthSq() < 1e-6) L.poleSm.copy(this._pole);
  633|           else L.poleSm.lerp(this._pole, 1 - Math.exp(-dt / 0.08)).normalize();
  634|           this._pole.copy(L.poleSm);
  635|         } else {
  636|           this._pole.set(0, 0, 1);
  637|         }
  638|         const target = new THREE.Vector3();
  639|         if (this.external) {
  640|           // v10 WORLD PLANTING: the foot is pinned to a world point for the
  641|           // whole stance, so it cannot slide. See _worldFootTarget.
  642|           this._worldFootTarget(L, target, dt);
  643|         } else if (this.driven && this.drive.footTargets[L.index]) {
  644|           target.copy(this.drive.footTargets[L.index]);
  645|         } else {
  646|           const u = (this.phase + L.phase) % 1;
  647|           let zOff = 0;
  648|           let yOff = 0;
  649|           if (moving > 0) {
  650|             if (u < 0.5) zOff = (0.5 - u * 2) * L.step;
  651|             else {
  652|               const s = (u - 0.5) * 2;
  653|               zOff = (s * 2 - 1) * L.step * 0.5;
  654|               yOff = Math.sin(s * Math.PI) * L.lift;
  655|             }
  656|           }
  657|           target.set(L.plant.x, L.plant.y + yOff * moving, L.plant.z + zOff * moving);
  658|           target.applyMatrix4(world);
  659|         }
  660|         solveLeg({
  661|           hip: L.hip, knee: L.knee, ankle: L.ankle,
  662|           targetWorld: target,
  663|           poleWorld: this._pole,
  664|           prevKnee: L.kneeLocal,
  665|         });
  666|         // A two-bone solve flips branch when the pole crosses the limb axis, and
  667|         // a splayed insect leg does that constantly. An airborne leg is capped
  668|         // at 12 rad/s so a flip reads as a fast move, never a pop. Planted legs
  669|         // are exempt, so foot slip stays exactly zero.
  670|         //
  671|         // v9: EXTERNAL mode derives the swing from the external cycle itself,
  672|         // and the plant-transition frame counts as airborne (that is the frame
  673|         // the solve is most prone to snap on).
  674|         const airborne = this.external ? L.mode === 'swing' : false;
  675|         const swingNow = this.external
  676|           ? (airborne || L.wasSwing)
  677|           : !!(this.driven && this.drive.footSwing[L.index]);
  678|         if (this.external) L.wasSwing = airborne;
  679|         if ((this.driven || this.external) && swingNow) {
  680|           const maxD = LEG_RATE * dt;
  681|           for (let k = 0; k < 3; k++) {
  682|             const bone = k === 0 ? L.hip : k === 1 ? L.knee : L.ankle;
  683|             const q = L.prev[k];
  684|             const ang = 2 * Math.acos(Math.min(1, Math.abs(bone.quaternion.dot(q))));
  685|             if (ang > maxD && ang > 1e-6) {
  686|               _lim.copy(bone.quaternion);
  687|               bone.quaternion.copy(q).slerp(_lim, maxD / ang);
  688|             }
  689|           }
  690|         }
  691|         L.prev[0].copy(L.hip.quaternion);
  692|         L.prev[1].copy(L.knee.quaternion);
  693|         L.prev[2].copy(L.ankle.quaternion);
  694|       }
  695|       void world;
  696|     }
  697|
