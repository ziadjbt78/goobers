  501|       this.bodyYaw += (this.bodyYawTarget - this.bodyYaw) * Math.min(1, dt * 1.5);
  502|       this.headPitch = 0;
  503|     }
  504|     const headLead = this.driven ? this.drive.headLead : this.headYawIdle;
  505| 
  506|     // ---- action layer -------------------------------------------------------
  507|     let actionPitch = 0;
  508|     let actionLift = 0;
  509|     let actionSquash = 1;
  510|     let actionRoll = 0;
  511|     if (act > 0.001) {
  512|       const a = this._actionPose(this.action, this.actionT);
  513|       actionPitch = a.pitch; actionLift = a.lift; actionSquash = a.squash; actionRoll = a.roll;
  514|     }
  515| 
  516|     // the Face map owns every face mesh in external mode (RigAdapter.setFace)
  517|     if (!this.external) this._updateFace(dt, act);
  518| 
  519|     // ---- write the body ----------------------------------------------------
  520|     const root = h.bones[0];
  521|     const body = h.bones[t.bodyBone];
  522|     const head = h.bones[t.headBone];
  523|     const breathe = 1 + Math.sin(this.t * 2.4) * 0.02;
  524|     const idle = Math.sin(this.t * 1.6) * 0.006;
  525|     if (this.external) root.position.set(0, 0, 0);
  526|     else if (!this.driven) root.position.set(0, bodyLift - idle * 0.5, 0);
  527|     else root.position.set(0, bodyLift - idle * 0.5 + actionLift * act, 0);
  528| 
  529|     const style = this._gaitFlavour(loco, speed01);
  530|     const roll = bodyRoll + style.roll + actionRoll * act;
  531|     this._q.setFromEuler(new THREE.Euler(
  532|       bodyPitch + actionPitch * act,
  533|       this.bodyYaw + (this.driven ? headLead * 0.25 : 0),
  534|       roll,
  535|     ));
  536|     body.quaternion.copy(this._q);
  537| 
  538|     const sq = this.squash * (1 + (actionSquash - 1) * act);
  539|     body.scale.set(1 / Math.sqrt(Math.max(0.2, sq)), sq * breathe, 1 / Math.sqrt(Math.max(0.2, sq)));
  540| 
  541|     for (let i = 0; i < this.spine.length; i++) {
  542|       const b = this.spine[i];
  543|       const lag = i * 0.16;
  544|       const w = Math.sin((this.phase - lag) * Math.PI * 2) * style.spine * (1 - i * 0.25);
  545|       b.rotateX(w);
  546|     }
  547| 
  548|     head.quaternion.setFromEuler(new THREE.Euler(
  549|       this.headPitch - bodyPitch * 0.5 + style.headPitch - actionPitch * act * 0.6,
  550|       headLead,
  551|       0,
  552|     ));
  553| 
  554|     // ---- feet --------------------------------------------------------------
  555|     // v12: the World path solves its legs LAST (Agent -> solveWorldLegs), after
  556|     // root, heading and body juice are final. Solving here, before those moved,
  557|     // is what dragged planted feet and popped knees.
  558|     if (!this.external) this._solveLegsPedestal(dt, moving);
  559| 
  560|     // ---- a petted creature kicks a back leg ---------------------------------
  561|     if (this.legs.length > 0) {
  562|       this.kickT = this.petted ? this.kickT + dt : 0;
  563|       if (this.petted) {
  564|         const back = this.legs[this.legs.length - 1];
  565|         const kick = Math.max(0, Math.sin(this.kickT * 9)) * 0.85;
  566|         if (kick > 0.02) back.hip.rotateX(-kick);
  567|       }
  568|     }
  569| 
  570|     // ---- arms swing opposite the legs, and wave when greeting --------------
  571|     for (const a of this.arms) {
  572|       let sw = 0;
  573|       if (loco > 0.01) sw = Math.sin(this.phase * Math.PI * 2 + (a.side > 0 ? Math.PI : 0)) * 0.52 * loco;
  574|       if (this.action === 'wave' || this.action === 'greet') {
  575|         sw += act * (a.side > 0 ? Math.sin(this.t * 13) * 0.9 + 0.7 : 0.1);
  576|       }
  577|       if (this.held) sw += Math.sin(this.t * 15 + a.side) * 0.55;
  578|       a.upper.rotateX(sw);
  579|       a.lower.rotateX(Math.max(0, -sw) * 0.5);
  580|     }
  581| 
  582|     // ---- wobble springs on ears, tails, antennae ---------------------------
  583|     const accel = this.driven ? this.drive.speed * 0.9 : this.gait * 0.6;
  584|     const driveW = -this.bodyYaw * 0.6 + bodyPitch * 2.4 + (accel * 0.10 * Math.sin(this.phase * Math.PI * 4));
  585|     for (const w of this.wiggles) {
  586|       const extra = this.template.locomotion === 'skitter' ? Math.sin(this.t * 17 + w.gain) * 0.04 * loco : 0;
  587|       const kick = this.mood === 'happy' ? Math.sin(this.t * 9) * 0.12 : 0;
  588|       const x = w.w.step((driveW + kick) * w.gain, Math.min(dt, 1 / 60));
  589|       w.bone.quaternion.setFromAxisAngle(w.axis === 'y' ? Y : X, x * 0.9 + extra);
  590|     }
  591|   }
  592| 
  593| 
  594|   /**
  595|    * v10 — world-space foot planting for the World path.
  596|    *
  597|    * The pedestal cycle moves a body-space target, which is why a planted foot
  598|    * used to travel with the body (25 cm of slide). Here the foot is instead
  599|    * pinned to a WORLD point for its whole stance, and the swing is a
  600|    * predicted-arc step to the next plant. Guards force an early lift before
  601|    * anything can overstretch or slide: leg reach, rest-offset error, turn
  602|    * error, and idle body drift.
  603|    */
  604|   /** Pedestal / driven leg solve, moved here VERBATIM from update(). */
  605|   private _solveLegsPedestal(dt: number, moving: number): void {
  606|     const h = this.handle;
  607|     // ---- feet --------------------------------------------------------------
  608|     if (this.legs.length > 0) {
  609|       h.group.updateMatrixWorld(true);
  610|       const world = new THREE.Matrix4().copy(h.group.matrixWorld);
  611|       // The IK pole must live in the CREATURE's frame, not the world's: a knee
  612|       // that always bends toward world +Z folds backwards the moment the
  613|       // creature faces any other way. On a pedestal the creature never turns,
  614|       // so the pole stays exactly the old world +Z and the baseline holds.
  615|       const drivenRig = this.driven || this.external;
  616|       const groupWorldQuat = drivenRig ? h.group.getWorldQuaternion(this._q) : null;
  617|       for (let i = 0; i < this.legs.length; i++) {
  618|         const L = this.legs[i];
  619|         if (drivenRig && groupWorldQuat) {
  620|           // per-leg pole in creature space, rotated into the world, then eased.
  621|           // An unsmoothed pole that grazes the limb axis is exactly what makes a
  622|           // two-bone solve flip; the ease removes that.
  623|           this._pole.copy(L.poleLocal).applyQuaternion(groupWorldQuat);
  624|           if (L.poleSm.lengthSq() < 1e-6) L.poleSm.copy(this._pole);
  625|           else L.poleSm.lerp(this._pole, 1 - Math.exp(-dt / 0.08)).normalize();
  626|           this._pole.copy(L.poleSm);
  627|         } else {
  628|           this._pole.set(0, 0, 1);
  629|         }
  630|         const target = new THREE.Vector3();
  631|         if (this.external) {
  632|           // v10 WORLD PLANTING: the foot is pinned to a world point for the
  633|           // whole stance, so it cannot slide. See _worldFootTarget.
  634|           this._worldFootTarget(L, target, dt);
  635|         } else if (this.driven && this.drive.footTargets[L.index]) {
  636|           target.copy(this.drive.footTargets[L.index]);
  637|         } else {
  638|           const u = (this.phase + L.phase) % 1;
  639|           let zOff = 0;
  640|           let yOff = 0;
  641|           if (moving > 0) {
  642|             if (u < 0.5) zOff = (0.5 - u * 2) * L.step;
  643|             else {
  644|               const s = (u - 0.5) * 2;
  645|               zOff = (s * 2 - 1) * L.step * 0.5;
  646|               yOff = Math.sin(s * Math.PI) * L.lift;
  647|             }
  648|           }
  649|           target.set(L.plant.x, L.plant.y + yOff * moving, L.plant.z + zOff * moving);
  650|           target.applyMatrix4(world);
  651|         }
  652|         solveLeg({
  653|           hip: L.hip, knee: L.knee, ankle: L.ankle,
  654|           targetWorld: target,
  655|           poleWorld: this._pole,
  656|           prevKnee: L.kneeLocal,
  657|         });
  658|         // A two-bone solve flips branch when the pole crosses the limb axis, and
  659|         // a splayed insect leg does that constantly. An airborne leg is capped
  660|         // at 12 rad/s so a flip reads as a fast move, never a pop. Planted legs
  661|         // are exempt, so foot slip stays exactly zero.
  662|         //
  663|         // v9: EXTERNAL mode derives the swing from the external cycle itself,
  664|         // and the plant-transition frame counts as airborne (that is the frame
  665|         // the solve is most prone to snap on).
  666|         const airborne = this.external ? L.mode === 'swing' : false;
  667|         const swingNow = this.external
  668|           ? (airborne || L.wasSwing)
  669|           : !!(this.driven && this.drive.footSwing[L.index]);
  670|         if (this.external) L.wasSwing = airborne;
  671|         if ((this.driven || this.external) && swingNow) {
  672|           const maxD = LEG_RATE * dt;
  673|           for (let k = 0; k < 3; k++) {
  674|             const bone = k === 0 ? L.hip : k === 1 ? L.knee : L.ankle;
  675|             const q = L.prev[k];
  676|             const ang = 2 * Math.acos(Math.min(1, Math.abs(bone.quaternion.dot(q))));
  677|             if (ang > maxD && ang > 1e-6) {
  678|               _lim.copy(bone.quaternion);
  679|               bone.quaternion.copy(q).slerp(_lim, maxD / ang);
  680|             }
  681|           }
  682|         }
  683|         L.prev[0].copy(L.hip.quaternion);
  684|         L.prev[1].copy(L.knee.quaternion);
  685|         L.prev[2].copy(L.ankle.quaternion);
  686|       }
  687|       void world;
  688|     }
  689| 
  690|   }

  692|   private _hips: Set<THREE.Object3D> | null = null;
  693|   /** v12: the hip bones the leg IK owns, so the Agent can preserve additive limb poses. */
  694|   legHips(): Set<THREE.Object3D> {
  695|     if (!this._hips) this._hips = new Set(this.legs.map((l) => l.hip));
  696|     return this._hips;
  697|   }

  699|   /**
  700|    * v12 World leg solve. Called by the Agent AFTER the motion layer has written
  701|    * the final root position, heading and body juice for this frame, so a planted
  702|    * foot is solved against the body it is actually attached to.
  703|    * `free` = held / airborne: nothing plants, locks re-seed on landing.
  704|    */
  705|   /** v13: world-units drop of the skeleton root that keeps every hip inside its leg */
  706|   crouch = 0;
  707|   liftStats = { beat: 0, hard: 0, strain: 0, soft: 0, maxErr: 0 };
  708|   resetLiftStats(): void { this.liftStats = { beat: 0, hard: 0, strain: 0, soft: 0, maxErr: 0 }; }
  709| 
  710|   solveWorldLegs(dt: number, free: boolean): void {
  711|     if (!this.external || this.legs.length === 0) return;
  712|     void this._wrapPi;
  713|     if (free) {
  714|       for (const L of this.legs) { L.init = false; L.mode = 'plant'; L.err = 0; }
  715|       this.crouch *= Math.exp(-dt / 0.2);
  716|       return;
  717|     }
  718|     const h = this.handle;
  719|     h.group.updateMatrixWorld(true);
  720| 
  721|     // v13 CROUCH: stubby legs plus the cartoon bob put the hips higher than the
  722|     // legs can reach, so a planted foot had zero horizontal room and HAD to slide.
  723|     // Drop the skeleton root just enough that every hip sits inside its leg.
  724|     const ground = this.drive.groundY;
  725|     const kH = 0.92 - 0.12 * Math.min(1, Math.max(0, this.locomotionW));
  726|     let need = 0;
  727|     for (const L of this.legs) {
  728|       const hip = L.hip.getWorldPosition(this._vw);
  729|       const knee = L.knee.getWorldPosition(this._kw);
  730|       const ank = L.ankle.getWorldPosition(this._aw);
  731|       L.reachW = hip.distanceTo(knee) + knee.distanceTo(ank);
  732|       this._rw.set(L.plant.x, 0, L.plant.z).applyMatrix4(h.group.matrixWorld);
  733|       const gy = (ground ? ground(this._rw.x, this._rw.z) : h.group.position.y) + L.plant.y;
  734|       need = Math.max(need, Math.min(0.5 * L.reachW, hip.y - gy - kH * L.reachW));
  735|     }
  736|     this.crouch = need > this.crouch ? need : this.crouch + (need - this.crouch) * Math.min(1, dt / 0.25);
  737|     if (this.crouch > 1e-5) {
  738|       const root = h.bones[0];
  739|       const ps = root.parent ? root.parent.getWorldScale(this._sc).y : 1;
  740|       root.position.y -= this.crouch / Math.max(1e-4, ps);
  741|       h.group.updateMatrixWorld(true);
  742|     }
  743| 
  744|     const gq = h.group.getWorldQuaternion(this._q);
  745|     for (const L of this.legs) {
  746|       this._pole.copy(L.poleLocal).applyQuaternion(gq);
  747|       if (L.poleSm.lengthSq() < 1e-6) L.poleSm.copy(this._pole);
  748|       else L.poleSm.lerp(this._pole, 1 - Math.exp(-dt / 0.08)).normalize();
  749|       this._pole.copy(L.poleSm);
  750|       this._worldFootTarget(L, this._ft, dt);
