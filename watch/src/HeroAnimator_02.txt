  501|     let actionPitch = 0;
  502|     let actionLift = 0;
  503|     let actionSquash = 1;
  504|     let actionRoll = 0;
  505|     if (act > 0.001) {
  506|       const a = this._actionPose(this.action, this.actionT);
  507|       actionPitch = a.pitch; actionLift = a.lift; actionSquash = a.squash; actionRoll = a.roll;
  508|     }
  509| 
  510|     // the Face map owns every face mesh in external mode (RigAdapter.setFace)
  511|     if (!this.external) this._updateFace(dt, act);
  512| 
  513|     // ---- write the body ----------------------------------------------------
  514|     const root = h.bones[0];
  515|     const body = h.bones[t.bodyBone];
  516|     const head = h.bones[t.headBone];
  517|     const breathe = 1 + Math.sin(this.t * 2.4) * 0.02;
  518|     const idle = Math.sin(this.t * 1.6) * 0.006;
  519|     if (this.external) root.position.set(0, 0, 0);
  520|     else if (!this.driven) root.position.set(0, bodyLift - idle * 0.5, 0);
  521|     else root.position.set(0, bodyLift - idle * 0.5 + actionLift * act, 0);
  522| 
  523|     const style = this._gaitFlavour(loco, speed01);
  524|     const roll = bodyRoll + style.roll + actionRoll * act;
  525|     this._q.setFromEuler(new THREE.Euler(
  526|       bodyPitch + actionPitch * act,
  527|       this.bodyYaw + (this.driven ? headLead * 0.25 : 0),
  528|       roll,
  529|     ));
  530|     body.quaternion.copy(this._q);
  531| 
  532|     const sq = this.squash * (1 + (actionSquash - 1) * act);
  533|     body.scale.set(1 / Math.sqrt(Math.max(0.2, sq)), sq * breathe, 1 / Math.sqrt(Math.max(0.2, sq)));
  534| 
  535|     for (let i = 0; i < this.spine.length; i++) {
  536|       const b = this.spine[i];
  537|       const lag = i * 0.16;
  538|       const w = Math.sin((this.phase - lag) * Math.PI * 2) * style.spine * (1 - i * 0.25);
  539|       b.rotateX(w);
  540|     }
  541| 
  542|     head.quaternion.setFromEuler(new THREE.Euler(
  543|       this.headPitch - bodyPitch * 0.5 + style.headPitch - actionPitch * act * 0.6,
  544|       headLead,
  545|       0,
  546|     ));
  547| 
  548|     // ---- feet --------------------------------------------------------------
  549|     // v12: the World path solves its legs LAST (Agent -> solveWorldLegs), after
  550|     // root, heading and body juice are final. Solving here, before those moved,
  551|     // is what dragged planted feet and popped knees.
  552|     if (!this.external) this._solveLegsPedestal(dt, moving);
  553| 
  554|     // ---- a petted creature kicks a back leg ---------------------------------
  555|     if (this.legs.length > 0) {
  556|       this.kickT = this.petted ? this.kickT + dt : 0;
  557|       if (this.petted) {
  558|         const back = this.legs[this.legs.length - 1];
  559|         const kick = Math.max(0, Math.sin(this.kickT * 9)) * 0.85;
  560|         if (kick > 0.02) back.hip.rotateX(-kick);
  561|       }
  562|     }
  563| 
  564|     // ---- arms swing opposite the legs, and wave when greeting --------------
  565|     for (const a of this.arms) {
  566|       let sw = 0;
  567|       if (loco > 0.01) sw = Math.sin(this.phase * Math.PI * 2 + (a.side > 0 ? Math.PI : 0)) * 0.52 * loco;
  568|       if (this.action === 'wave' || this.action === 'greet') {
  569|         sw += act * (a.side > 0 ? Math.sin(this.t * 13) * 0.9 + 0.7 : 0.1);
  570|       }
  571|       if (this.held) sw += Math.sin(this.t * 15 + a.side) * 0.55;
  572|       a.upper.rotateX(sw);
  573|       a.lower.rotateX(Math.max(0, -sw) * 0.5);
  574|     }
  575| 
  576|     // ---- wobble springs on ears, tails, antennae ---------------------------
  577|     const accel = this.driven ? this.drive.speed * 0.9 : this.gait * 0.6;
  578|     const driveW = -this.bodyYaw * 0.6 + bodyPitch * 2.4 + (accel * 0.10 * Math.sin(this.phase * Math.PI * 4));
  579|     for (const w of this.wiggles) {
  580|       const extra = this.template.locomotion === 'skitter' ? Math.sin(this.t * 17 + w.gain) * 0.04 * loco : 0;
  581|       const kick = this.mood === 'happy' ? Math.sin(this.t * 9) * 0.12 : 0;
  582|       const x = w.w.step((driveW + kick) * w.gain, Math.min(dt, 1 / 60));
  583|       w.bone.quaternion.setFromAxisAngle(w.axis === 'y' ? Y : X, x * 0.9 + extra);
  584|     }
  585|   }

  588|   /**
  589|    * v10 — world-space foot planting for the World path.
  590|    *
  591|    * The pedestal cycle moves a body-space target, which is why a planted foot
  592|    * used to travel with the body (25 cm of slide). Here the foot is instead
  593|    * pinned to a WORLD point for its whole stance, and the swing is a
  594|    * predicted-arc step to the next plant. Guards force an early lift before
  595|    * anything can overstretch or slide: leg reach, rest-offset error, turn
  596|    * error, and idle body drift.
  597|    */
  598|   /** Pedestal / driven leg solve, moved here VERBATIM from update(). */
  599|   private _solveLegsPedestal(dt: number, moving: number): void {
  600|     const h = this.handle;
  601|     // ---- feet --------------------------------------------------------------
  602|     if (this.legs.length > 0) {
  603|       h.group.updateMatrixWorld(true);
  604|       const world = new THREE.Matrix4().copy(h.group.matrixWorld);
  605|       // The IK pole must live in the CREATURE's frame, not the world's: a knee
  606|       // that always bends toward world +Z folds backwards the moment the
  607|       // creature faces any other way. On a pedestal the creature never turns,
  608|       // so the pole stays exactly the old world +Z and the baseline holds.
  609|       const drivenRig = this.driven || this.external;
  610|       const groupWorldQuat = drivenRig ? h.group.getWorldQuaternion(this._q) : null;
  611|       for (let i = 0; i < this.legs.length; i++) {
  612|         const L = this.legs[i];
  613|         if (drivenRig && groupWorldQuat) {
  614|           // per-leg pole in creature space, rotated into the world, then eased.
  615|           // An unsmoothed pole that grazes the limb axis is exactly what makes a
  616|           // two-bone solve flip; the ease removes that.
  617|           this._pole.copy(L.poleLocal).applyQuaternion(groupWorldQuat);
  618|           if (L.poleSm.lengthSq() < 1e-6) L.poleSm.copy(this._pole);
  619|           else L.poleSm.lerp(this._pole, 1 - Math.exp(-dt / 0.08)).normalize();
  620|           this._pole.copy(L.poleSm);
  621|         } else {
  622|           this._pole.set(0, 0, 1);
  623|         }
  624|         const target = new THREE.Vector3();
  625|         if (this.external) {
  626|           // v10 WORLD PLANTING: the foot is pinned to a world point for the
  627|           // whole stance, so it cannot slide. See _worldFootTarget.
  628|           this._worldFootTarget(L, target, dt);
  629|         } else if (this.driven && this.drive.footTargets[L.index]) {
  630|           target.copy(this.drive.footTargets[L.index]);
  631|         } else {
  632|           const u = (this.phase + L.phase) % 1;
  633|           let zOff = 0;
  634|           let yOff = 0;
  635|           if (moving > 0) {
  636|             if (u < 0.5) zOff = (0.5 - u * 2) * L.step;
  637|             else {
  638|               const s = (u - 0.5) * 2;
  639|               zOff = (s * 2 - 1) * L.step * 0.5;
  640|               yOff = Math.sin(s * Math.PI) * L.lift;
  641|             }
  642|           }
  643|           target.set(L.plant.x, L.plant.y + yOff * moving, L.plant.z + zOff * moving);
  644|           target.applyMatrix4(world);
  645|         }
  646|         solveLeg({
  647|           hip: L.hip, knee: L.knee, ankle: L.ankle,
  648|           targetWorld: target,
  649|           poleWorld: this._pole,
  650|           prevKnee: L.kneeLocal,
  651|         });
  652|         // A two-bone solve flips branch when the pole crosses the limb axis, and
  653|         // a splayed insect leg does that constantly. An airborne leg is capped
  654|         // at 12 rad/s so a flip reads as a fast move, never a pop. Planted legs
  655|         // are exempt, so foot slip stays exactly zero.
  656|         //
  657|         // v9: EXTERNAL mode derives the swing from the external cycle itself,
  658|         // and the plant-transition frame counts as airborne (that is the frame
  659|         // the solve is most prone to snap on).
  660|         const airborne = this.external ? L.mode === 'swing' : false;
  661|         const swingNow = this.external
  662|           ? (airborne || L.wasSwing)
  663|           : !!(this.driven && this.drive.footSwing[L.index]);
  664|         if (this.external) L.wasSwing = airborne;
  665|         if ((this.driven || this.external) && swingNow) {
  666|           const maxD = LEG_RATE * dt;
  667|           for (let k = 0; k < 3; k++) {
  668|             const bone = k === 0 ? L.hip : k === 1 ? L.knee : L.ankle;
  669|             const q = L.prev[k];
  670|             const ang = 2 * Math.acos(Math.min(1, Math.abs(bone.quaternion.dot(q))));
  671|             if (ang > maxD && ang > 1e-6) {
  672|               _lim.copy(bone.quaternion);
  673|               bone.quaternion.copy(q).slerp(_lim, maxD / ang);
  674|             }
  675|           }
  676|         }
  677|         L.prev[0].copy(L.hip.quaternion);
  678|         L.prev[1].copy(L.knee.quaternion);
  679|         L.prev[2].copy(L.ankle.quaternion);
  680|       }
  681|       void world;
  682|     }
  683| 
  684|   }

  686|   private _hips: Set<THREE.Object3D> | null = null;
  687|   /** v12: the hip bones the leg IK owns, so the Agent can preserve additive limb poses. */
  688|   legHips(): Set<THREE.Object3D> {
  689|     if (!this._hips) this._hips = new Set(this.legs.map((l) => l.hip));
  690|     return this._hips;
  691|   }

  693|   /**
  694|    * v12 World leg solve. Called by the Agent AFTER the motion layer has written
  695|    * the final root position, heading and body juice for this frame, so a planted
  696|    * foot is solved against the body it is actually attached to.
  697|    * `free` = held / airborne: nothing plants, locks re-seed on landing.
  698|    */
  699|   solveWorldLegs(dt: number, free: boolean): void {
  700|     if (!this.external || this.legs.length === 0) return;
  701|     void this._wrapPi;
  702|     if (free) { for (const L of this.legs) { L.init = false; L.mode = 'plant'; } return; }
  703|     const h = this.handle;
  704|     h.group.updateMatrixWorld(true);
  705|     const gq = h.group.getWorldQuaternion(this._q);
  706|     for (const L of this.legs) {
  707|       this._pole.copy(L.poleLocal).applyQuaternion(gq);
  708|       if (L.poleSm.lengthSq() < 1e-6) L.poleSm.copy(this._pole);
  709|       else L.poleSm.lerp(this._pole, 1 - Math.exp(-dt / 0.08)).normalize();
  710|       this._pole.copy(L.poleSm);
  711|       this._worldFootTarget(L, this._ft, dt);
  712|       solveLeg({ hip: L.hip, knee: L.knee, ankle: L.ankle, targetWorld: this._ft, poleWorld: this._pole, prevKnee: L.kneeLocal });
  713|       // airborne legs (and the landing frame) are rate limited; planted legs never are
  714|       if (L.mode === 'swing' || L.landPending) {
  715|         const maxD = LEG_RATE * dt;
  716|         for (let k = 0; k < 3; k++) {
  717|           const bone = k === 0 ? L.hip : k === 1 ? L.knee : L.ankle;
  718|           const q = L.prev[k];
  719|           const ang = 2 * Math.acos(Math.min(1, Math.abs(bone.quaternion.dot(q))));
  720|           if (ang > maxD && ang > 1e-6) { _lim.copy(bone.quaternion); bone.quaternion.copy(q).slerp(_lim, maxD / ang); }
  721|         }
  722|       }
  723|       L.prev[0].copy(L.hip.quaternion);
  724|       L.prev[1].copy(L.knee.quaternion);
  725|       L.prev[2].copy(L.ankle.quaternion);
  726|       // lock where the foot REALLY landed: a limiter lag can never become a snap
  727|       if (L.landPending) { L.ankle.getWorldPosition(L.lock); L.landPending = false; }
  728|     }
  729|     if (this.petted) {
  730|       const back = this.legs[this.legs.length - 1];
  731|       const kick = Math.max(0, Math.sin(this.kickT * 9)) * 0.85;
  732|       if (kick > 0.02) back.hip.rotateX(-kick);
  733|     }
  734|   }

  736|   /**
  737|    * v12 world stepper. Leg geometry is measured LIVE in world space (creature
  738|    * scale and squash included). Lifts are edge-triggered on the gait beat; a
  739|    * swing re-aims every frame at where the foot must land so the coming stance
  740|    * is centred under the hip; guards lift early only when the leg truly runs
  741|    * out of reach, with an airborne-count limit so a creature never lifts every
  742|    * foot at once.
  743|    */
  744|   private _worldFootTarget(L: LegRig, out: THREE.Vector3, dt: number): void {
  745|     const g = this.handle.group;
  746|     const gp = g.position;
  747|     const hd = this.drive.heading;
  748|     const fx = Math.sin(hd), fz = Math.cos(hd);
  749|     const speed = this.drive.speed;
  750|     const moving = speed > 0.05;
