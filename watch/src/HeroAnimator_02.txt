  501|       this.nextTurn -= dt;
  502|       if (this.nextTurn <= 0) {
  503|         this.nextTurn = 1.6 + Math.random() * 3;
  504|         this.bodyYawTarget = this.deterministic ? 0 : (Math.random() - 0.5) * 0.55;
  505|       }
  506|       this.headYawIdle = THREE.MathUtils.clamp(this.bodyYawTarget - this.bodyYaw, -0.55, 0.55) * 0.9;
  507|       this.bodyYaw += (this.bodyYawTarget - this.bodyYaw) * Math.min(1, dt * 1.5);
  508|       this.headPitch = 0;
  509|     }
  510|     const headLead = this.driven ? this.drive.headLead : this.headYawIdle;
  511| 
  512|     // ---- action layer -------------------------------------------------------
  513|     let actionPitch = 0;
  514|     let actionLift = 0;
  515|     let actionSquash = 1;
  516|     let actionRoll = 0;
  517|     if (act > 0.001) {
  518|       const a = this._actionPose(this.action, this.actionT);
  519|       actionPitch = a.pitch; actionLift = a.lift; actionSquash = a.squash; actionRoll = a.roll;
  520|     }
  521| 
  522|     // the Face map owns every face mesh in external mode (RigAdapter.setFace)
  523|     if (!this.external) this._updateFace(dt, act);
  524| 
  525|     // ---- write the body ----------------------------------------------------
  526|     const root = h.bones[0];
  527|     const body = h.bones[t.bodyBone];
  528|     const head = h.bones[t.headBone];
  529|     const breathe = 1 + Math.sin(this.t * 2.4) * 0.02;
  530|     const idle = Math.sin(this.t * 1.6) * 0.006;
  531|     if (this.external) root.position.set(0, 0, 0);
  532|     else if (!this.driven) root.position.set(0, bodyLift - idle * 0.5, 0);
  533|     else root.position.set(0, bodyLift - idle * 0.5 + actionLift * act, 0);
  534| 
  535|     const style = this._gaitFlavour(loco, speed01);
  536|     const roll = bodyRoll + style.roll + actionRoll * act;
  537|     this._q.setFromEuler(new THREE.Euler(
  538|       bodyPitch + actionPitch * act,
  539|       this.bodyYaw + (this.driven ? headLead * 0.25 : 0),
  540|       roll,
  541|     ));
  542|     body.quaternion.copy(this._q);
  543| 
  544|     const sq = this.squash * (1 + (actionSquash - 1) * act);
  545|     body.scale.set(1 / Math.sqrt(Math.max(0.2, sq)), sq * breathe, 1 / Math.sqrt(Math.max(0.2, sq)));
  546| 
  547|     for (let i = 0; i < this.spine.length; i++) {
  548|       const b = this.spine[i];
  549|       const lag = i * 0.16;
  550|       const w = Math.sin((this.phase - lag) * Math.PI * 2) * style.spine * (1 - i * 0.25);
  551|       b.rotateX(w);
  552|     }
  553| 
  554|     head.quaternion.setFromEuler(new THREE.Euler(
  555|       this.headPitch - bodyPitch * 0.5 + style.headPitch - actionPitch * act * 0.6,
  556|       headLead,
  557|       0,
  558|     ));
  559| 
  560|     // ---- feet --------------------------------------------------------------
  561|     // v12: the World path solves its legs LAST (Agent -> solveWorldLegs), after
  562|     // root, heading and body juice are final. Solving here, before those moved,
  563|     // is what dragged planted feet and popped knees.
  564|     if (!this.external) this._solveLegsPedestal(dt, moving);
  565| 
  566|     // ---- a petted creature kicks a back leg ---------------------------------
  567|     if (this.legs.length > 0) {
  568|       this.kickT = this.petted ? this.kickT + dt : 0;
  569|       if (this.petted) {
  570|         const back = this.legs[this.legs.length - 1];
  571|         const kick = Math.max(0, Math.sin(this.kickT * 9)) * 0.85;
  572|         if (kick > 0.02) back.hip.rotateX(-kick);
  573|       }
  574|     }
  575| 
  576|     // ---- arms swing opposite the legs, and wave when greeting --------------
  577|     for (const a of this.arms) {
  578|       let sw = 0;
  579|       if (loco > 0.01) sw = Math.sin(this.phase * Math.PI * 2 + (a.side > 0 ? Math.PI : 0)) * 0.52 * loco;
  580|       if (this.action === 'wave' || this.action === 'greet') {
  581|         sw += act * (a.side > 0 ? Math.sin(this.t * 13) * 0.9 + 0.7 : 0.1);
  582|       }
  583|       if (this.held) sw += Math.sin(this.t * 15 + a.side) * 0.55;
  584|       a.upper.rotateX(sw);
  585|       a.lower.rotateX(Math.max(0, -sw) * 0.5);
  586|     }
  587| 
  588|     // ---- wobble springs on ears, tails, antennae ---------------------------
  589|     const accel = this.driven ? this.drive.speed * 0.9 : this.gait * 0.6;
  590|     const driveW = -this.bodyYaw * 0.6 + bodyPitch * 2.4 + (accel * 0.10 * Math.sin(this.phase * Math.PI * 4));
  591|     for (const w of this.wiggles) {
  592|       const extra = this.template.locomotion === 'skitter' ? Math.sin(this.t * 17 + w.gain) * 0.04 * loco : 0;
  593|       const kick = this.mood === 'happy' ? Math.sin(this.t * 9) * 0.12 : 0;
  594|       const x = w.w.step((driveW + kick) * w.gain, Math.min(dt, 1 / 60));
  595|       w.bone.quaternion.setFromAxisAngle(w.axis === 'y' ? Y : X, x * 0.9 + extra);
  596|     }
  597|   }

  600|   /**
  601|    * v10 — world-space foot planting for the World path.
  602|    *
  603|    * The pedestal cycle moves a body-space target, which is why a planted foot
  604|    * used to travel with the body (25 cm of slide). Here the foot is instead
  605|    * pinned to a WORLD point for its whole stance, and the swing is a
  606|    * predicted-arc step to the next plant. Guards force an early lift before
  607|    * anything can overstretch or slide: leg reach, rest-offset error, turn
  608|    * error, and idle body drift.
  609|    */
  610|   /** Pedestal / driven leg solve, moved here VERBATIM from update(). */
  611|   private _solveLegsPedestal(dt: number, moving: number): void {
  612|     const h = this.handle;
  613|     // ---- feet --------------------------------------------------------------
  614|     if (this.legs.length > 0) {
  615|       h.group.updateMatrixWorld(true);
  616|       const world = new THREE.Matrix4().copy(h.group.matrixWorld);
  617|       // The IK pole must live in the CREATURE's frame, not the world's: a knee
  618|       // that always bends toward world +Z folds backwards the moment the
  619|       // creature faces any other way. On a pedestal the creature never turns,
  620|       // so the pole stays exactly the old world +Z and the baseline holds.
  621|       const drivenRig = this.driven || this.external;
  622|       const groupWorldQuat = drivenRig ? h.group.getWorldQuaternion(this._q) : null;
  623|       for (let i = 0; i < this.legs.length; i++) {
  624|         const L = this.legs[i];
  625|         if (drivenRig && groupWorldQuat) {
  626|           // per-leg pole in creature space, rotated into the world, then eased.
  627|           // An unsmoothed pole that grazes the limb axis is exactly what makes a
  628|           // two-bone solve flip; the ease removes that.
  629|           this._pole.copy(L.poleLocal).applyQuaternion(groupWorldQuat);
  630|           if (L.poleSm.lengthSq() < 1e-6) L.poleSm.copy(this._pole);
  631|           else L.poleSm.lerp(this._pole, 1 - Math.exp(-dt / 0.08)).normalize();
  632|           this._pole.copy(L.poleSm);
  633|         } else {
  634|           this._pole.set(0, 0, 1);
  635|         }
  636|         const target = new THREE.Vector3();
  637|         if (this.external) {
  638|           // v10 WORLD PLANTING: the foot is pinned to a world point for the
  639|           // whole stance, so it cannot slide. See _worldFootTarget.
  640|           this._worldFootTarget(L, target, dt);
  641|         } else if (this.driven && this.drive.footTargets[L.index]) {
  642|           target.copy(this.drive.footTargets[L.index]);
  643|         } else {
  644|           const u = (this.phase + L.phase) % 1;
  645|           let zOff = 0;
  646|           let yOff = 0;
  647|           if (moving > 0) {
  648|             if (u < 0.5) zOff = (0.5 - u * 2) * L.step;
  649|             else {
  650|               const s = (u - 0.5) * 2;
  651|               zOff = (s * 2 - 1) * L.step * 0.5;
  652|               yOff = Math.sin(s * Math.PI) * L.lift;
  653|             }
  654|           }
  655|           target.set(L.plant.x, L.plant.y + yOff * moving, L.plant.z + zOff * moving);
  656|           target.applyMatrix4(world);
  657|         }
  658|         solveLeg({
  659|           hip: L.hip, knee: L.knee, ankle: L.ankle,
  660|           targetWorld: target,
  661|           poleWorld: this._pole,
  662|           prevKnee: L.kneeLocal,
  663|         });
  664|         // A two-bone solve flips branch when the pole crosses the limb axis, and
  665|         // a splayed insect leg does that constantly. An airborne leg is capped
  666|         // at 12 rad/s so a flip reads as a fast move, never a pop. Planted legs
  667|         // are exempt, so foot slip stays exactly zero.
  668|         //
  669|         // v9: EXTERNAL mode derives the swing from the external cycle itself,
  670|         // and the plant-transition frame counts as airborne (that is the frame
  671|         // the solve is most prone to snap on).
  672|         const airborne = this.external ? L.mode === 'swing' : false;
  673|         const swingNow = this.external
  674|           ? (airborne || L.wasSwing)
  675|           : !!(this.driven && this.drive.footSwing[L.index]);
  676|         if (this.external) L.wasSwing = airborne;
  677|         if ((this.driven || this.external) && swingNow) {
  678|           const maxD = LEG_RATE * dt;
  679|           for (let k = 0; k < 3; k++) {
  680|             const bone = k === 0 ? L.hip : k === 1 ? L.knee : L.ankle;
  681|             const q = L.prev[k];
  682|             const ang = 2 * Math.acos(Math.min(1, Math.abs(bone.quaternion.dot(q))));
  683|             if (ang > maxD && ang > 1e-6) {
  684|               _lim.copy(bone.quaternion);
  685|               bone.quaternion.copy(q).slerp(_lim, maxD / ang);
  686|             }
  687|           }
  688|         }
  689|         L.prev[0].copy(L.hip.quaternion);
  690|         L.prev[1].copy(L.knee.quaternion);
  691|         L.prev[2].copy(L.ankle.quaternion);
  692|       }
  693|       void world;
  694|     }
  695| 
  696|   }

  710|   /** v15: rate-limit all three bones of a leg against last frame, then remember them. */

  726|   /**
  727|    * v12 World leg solve. Called by the Agent AFTER the motion layer has written
  728|    * the final root position, heading and body juice for this frame, so a planted
  729|    * foot is solved against the body it is actually attached to.
  730|    * `free` = held / airborne: nothing plants, locks re-seed on landing.
  731|    */
  732|   /** v13: world-units drop of the skeleton root that keeps every hip inside its leg */
  733|   crouch = 0;
  734|   /** v14: legs hang free this frame (held, rolled, mid-jump) */
  735|   legsFree = false;
  736|   liftStats = { beat: 0, hard: 0, strain: 0, soft: 0, maxErr: 0 };
  737|   resetLiftStats(): void { this.liftStats = { beat: 0, hard: 0, strain: 0, soft: 0, maxErr: 0 }; }

  739|   solveWorldLegs(dt: number, free: boolean): void {
  740|     if (!this.external || this.legs.length === 0) return;
  741|     void this._wrapPi;
  742|     const restQ = this._restQ ?? (this._restQ = this.legs.map((L) => [L.knee.quaternion.clone(), L.ankle.quaternion.clone()]));
  743|     this.legsFree = free;
  744|     if (free) {
  745|       // v15: hanging legs EASE, never snap. Hips keep the motion layer's pose
  746|       // (dangle flail), knees/ankles relax toward rest at the swing rate, and the
  747|       // crouch stays applied (decaying) so the body cannot pop up.
  748|       const hf = this.handle;
  749|       this.crouch *= Math.exp(-dt / 0.2);
  750|       if (this.crouch > 1e-5) {
