  751|         const root = hf.bones[0];
  752|         const ps = root.parent ? root.parent.getWorldScale(this._sc).y : 1;
  753|         root.position.y -= this.crouch / Math.max(1e-4, ps);
  754|       }
  755|       const maxF = LEG_RATE * dt;
  756|       for (let li = 0; li < this.legs.length; li++) {
  757|         const L = this.legs[li];
  758|         L.relaunch = true; L.mode = 'plant'; L.err = 0; L.postErr = 0;
  759|         L.knee.quaternion.copy(restQ[li][0]);
  760|         L.ankle.quaternion.copy(restQ[li][1]);
  761|         this._limitLeg(L, maxF);
  762|       }
  763|       return;
  764|     }
  765|     const h = this.handle;
  766|     h.group.updateMatrixWorld(true);
  767| 
  768|     // v13 CROUCH: stubby legs plus the cartoon bob put the hips higher than the
  769|     // legs can reach, so a planted foot had zero horizontal room and HAD to slide.
  770|     // Drop the skeleton root just enough that every hip sits inside its leg.
  771|     const ground = this.drive.groundY;
  772|     const kH = 0.92 - 0.12 * Math.min(1, Math.max(0, this.locomotionW));
  773|     let need = 0;
  774|     for (const L of this.legs) {
  775|       const hip = L.hip.getWorldPosition(this._vw);
  776|       const knee = L.knee.getWorldPosition(this._kw);
  777|       const ank = L.ankle.getWorldPosition(this._aw);
  778|       L.reachW = hip.distanceTo(knee) + knee.distanceTo(ank);
  779|       this._rw.set(L.plant.x, 0, L.plant.z).applyMatrix4(h.group.matrixWorld);
  780|       const gy = (ground ? ground(this._rw.x, this._rw.z) : h.group.position.y) + L.plant.y;
  781|       need = Math.max(need, Math.min(0.5 * L.reachW, hip.y - gy - kH * L.reachW));
  782|     }
  783|     this.crouch = need > this.crouch ? need : this.crouch + (need - this.crouch) * Math.min(1, dt / 0.25);
  784|     if (this.crouch > 1e-5) {
  785|       const root = h.bones[0];
  786|       const ps = root.parent ? root.parent.getWorldScale(this._sc).y : 1;
  787|       root.position.y -= this.crouch / Math.max(1e-4, ps);
  788|       h.group.updateMatrixWorld(true);
  789|     }
  790| 
  791|     const gq = h.group.getWorldQuaternion(this._q);
  792|     for (const L of this.legs) {
  793|       // v15: the pole rides the hip's PARENT (the body), so an action that pitches
  794|       // or rolls the body carries the knee direction with it instead of flipping it
  795|       const par = L.hip.parent;
  796|       if (par) {
  797|         par.getWorldQuaternion(this._pq);
  798|         if (!L.parRel) L.parRel = gq.clone().invert().multiply(this._pq);
  799|         this._pq.multiply(this._iq.copy(L.parRel).invert());
  800|         this._pole.copy(L.poleLocal).applyQuaternion(this._pq);
  801|       } else {
  802|         this._pole.copy(L.poleLocal).applyQuaternion(gq);
  803|       }
  804|       if (L.poleSm.lengthSq() < 1e-6) L.poleSm.copy(this._pole);
  805|       else L.poleSm.lerp(this._pole, 1 - Math.exp(-dt / 0.08)).normalize();
  806|       this._pole.copy(L.poleSm);
  807|       if (L.relaunch) {
  808|         // v15: back from hanging: swing in from where the ankle REALLY is
  809|         L.relaunch = false;
  810|         if (L.init) {
  811|           L.ankle.getWorldPosition(this._aw);
  812|           L.mode = 'swing'; L.swingT = 0; L.swingDur = 0.18; L.forced = true;
  813|           L.swingFrom.copy(this._aw); L.swingTo.copy(this._aw);
  814|         }
  815|       }
  816|       this._worldFootTarget(L, this._ft, dt);
  817|       this._goal.copy(this._ft);
  818|       const hip = L.hip.getWorldPosition(this._hw2);
  819|       const rMax = 0.985 * L.reachW;
  820|       this._clampReach(this._ft, hip, rMax);
  821|       // v13 CLOSED-LOOP IK: squash shears the leg's parent space, so the aim
  822|       // misses. Measure where the ankle REALLY went and re-aim, up to 3 times.
  823|       let err = 0;
  824|       for (let it = 0; it < 3; it++) {
  825|         solveLeg({ hip: L.hip, knee: L.knee, ankle: L.ankle, targetWorld: this._ft, poleWorld: this._pole, prevKnee: L.kneeLocal });
  826|         L.ankle.updateWorldMatrix(true, false);
  827|         L.ankle.getWorldPosition(this._aw);
  828|         const ex = this._goal.x - this._aw.x, ey = this._goal.y - this._aw.y, ez = this._goal.z - this._aw.z;
  829|         err = Math.hypot(ex, ey, ez);
  830|         if (err < 0.0015) break;
  831|         this._ft.x += ex; this._ft.y += ey; this._ft.z += ez;
  832|         this._clampReach(this._ft, hip, rMax);
  833|       }
  834|       L.err = err;
  835|       if (L.mode === 'plant' && !L.landPending && err > this.liftStats.maxErr) this.liftStats.maxErr = err;
  836|       const maxD = (L.mode === 'swing' || L.landPending ? LEG_RATE : PLANT_RATE) * dt;
  837|       for (let k = 0; k < 3; k++) {
  838|         const bone = k === 0 ? L.hip : k === 1 ? L.knee : L.ankle;
  839|         const q = L.prev[k];
  840|         const ang = 2 * Math.acos(Math.min(1, Math.abs(bone.quaternion.dot(q))));
  841|         if (ang > maxD && ang > 1e-6) { _lim.copy(bone.quaternion); bone.quaternion.copy(q).slerp(_lim, maxD / ang); }
  842|       }
  843|       L.ankle.updateWorldMatrix(true, false);
  844|       L.postErr = L.ankle.getWorldPosition(this._aw).distanceTo(this._goal);
  845|       L.prev[0].copy(L.hip.quaternion);
  846|       L.prev[1].copy(L.knee.quaternion);
  847|       L.prev[2].copy(L.ankle.quaternion);
  848|       if (L.landPending) { L.ankle.updateWorldMatrix(true, false); L.ankle.getWorldPosition(L.lock); L.landPending = false; }
  849|     }
  850|     if (this.petted) {
  851|       const back = this.legs[this.legs.length - 1];
  852|       const kick = Math.max(0, Math.sin(this.kickT * 9)) * 0.85;
  853|       if (kick > 0.02) back.hip.rotateX(-kick);
  854|     }
  855|   }

  857|   private _clampReach(v: THREE.Vector3, hip: THREE.Vector3, r: number): void {
  858|     if (r <= 1e-4) return;
  859|     const d = v.distanceTo(hip);
  860|     if (d > r) v.sub(hip).multiplyScalar(r / d).add(hip);
  861|   }

  867|   /**
  868|    * v12 world stepper. Leg geometry is measured LIVE in world space (creature
  869|    * scale and squash included). Lifts are edge-triggered on the gait beat; a
  870|    * swing re-aims every frame at where the foot must land so the coming stance
  871|    * is centred under the hip; guards lift early only when the leg truly runs
  872|    * out of reach, with an airborne-count limit so a creature never lifts every
  873|    * foot at once.
  874|    */
  875|   private _worldFootTarget(L: LegRig, out: THREE.Vector3, dt: number): void {
  876|     const g = this.handle.group;
  877|     const gp = g.position;
  878|     const hd = this.drive.heading;
  879|     const fx = Math.sin(hd), fz = Math.cos(hd);
  880|     const speed = this.drive.speed;
  881|     const moving = speed > 0.05;
  882|     const stride = Math.max(0.05, this.drive.stride);
  883|     const ground = this.drive.groundY;
  884|     const gy = (x: number, z: number): number => (ground ? ground(x, z) : gp.y) + L.plant.y;
  885| 
  886|     const hip = L.hip.getWorldPosition(this._vw);
  887|     const knee = L.knee.getWorldPosition(this._kw);
  888|     const ank = L.ankle.getWorldPosition(this._aw);
  889|     const reach = Math.max(1e-3, hip.distanceTo(knee) + knee.distanceTo(ank));
  890|     const rest = this._rw.set(L.plant.x, 0, L.plant.z).applyMatrix4(g.matrixWorld);
  891|     rest.y = gy(rest.x, rest.z);
  892| 
  893|     if (!L.init) {
  894|       L.lock.copy(rest); L.mode = 'plant'; L.landT = 1;
  895|       L.prevU = (this.externalPhase + L.phase) % 1;
  896|       L.init = true;
  897|     }
  898| 
  899|     const dyHip = Math.max(0, hip.y - rest.y);
  900|     const r92 = 0.92 * reach;
  901|     const restFlat = Math.hypot(rest.x - hip.x, rest.z - hip.z);
  902|     const flat = Math.max(0.25 * reach, 1.3 * restFlat, Math.sqrt(Math.max(0, r92 * r92 - dyHip * dyHip)));
  903|     const hardR = Math.max(0.97 * reach, 1.08 * hip.distanceTo(rest));
  904|     const cycle = moving ? stride / speed : 1.2;
  905|     const half = Math.min(0.5 * DUTY * stride, 0.6 * flat);
  906| 
  907|     const u = (this.externalPhase + L.phase) % 1;
  908|     const onBeat = moving && crossedPhase(L.prevU, u, DUTY);
  909|     L.prevU = u;
  910|     L.landT += dt;
  911| 
  912|     // landing point: half a stance AHEAD of where the body will be at touchdown,
  913|     // clamped inside the leg's reach from where the hip will be
  914|     const plan = (remaining: number, o: THREE.Vector3): THREE.Vector3 => {
  915|       const travel = moving ? speed * remaining : 0;
  916|       const ahead = moving ? travel + half : 0;
  917|       o.set(rest.x + fx * ahead, 0, rest.z + fz * ahead);
  918|       const hx = hip.x + fx * travel, hz = hip.z + fz * travel;
  919|       const vx = o.x - hx, vz = o.z - hz;
  920|       const vd = Math.hypot(vx, vz), lim = 0.85 * flat;
  921|       if (vd > lim) { o.x = hx + (vx / vd) * lim; o.z = hz + (vz / vd) * lim; }
  922|       o.y = gy(o.x, o.z);
  923|       return o;
  924|     };
  925| 
  926|     if (L.mode === 'plant') {
  927|       // v13: a planted foot the solver cannot hold steps instead of sliding
  928|       const strained = L.err > 0.01 && L.landT > 0.05;
  929|       const hard = strained || Math.hypot(L.lock.x - hip.x, L.lock.z - hip.z) > flat || hip.distanceTo(L.lock) > hardR;
  930|       const rx = L.lock.x - rest.x, rz = L.lock.z - rest.z;
  931|       const along = rx * fx + rz * fz;
  932|       const lagging = moving && along < -1.35 * half;
  933|       const drifted = !moving && Math.hypot(rx, rz) > Math.max(0.3 * flat, 0.05 * reach);
  934|       let air = 0;
  935|       for (const o of this.legs) if (o !== L && o.mode === 'swing') air++;
  936|       const maxAir = this.legs.length >= 6 ? 3 : this.legs.length >= 4 ? 2 : 1;
  937|       const beat = onBeat && L.landT > 0.2 * cycle;
  938|       const soft = (lagging || drifted) && air < maxAir && L.landT > 0.08;
  939|       if (hard || beat || soft) {
  940|         const ls = this.liftStats;
  941|         if (strained) ls.strain++; else if (hard) ls.hard++; else if (beat) ls.beat++; else ls.soft++;
  942|         L.mode = 'swing';
  943|         L.forced = !beat;
  944|         L.swingT = 0;
  945|         L.swingFrom.copy(L.lock);
  946|         if (beat && !hard) {
  947|           L.swingDur = THREE.MathUtils.clamp((1 - DUTY) * cycle, 0.12, 0.40);
  948|         } else {
  949|           plan(0.2, L.swingTo);
  950|           const dd = Math.hypot(L.swingTo.x - L.lock.x, L.swingTo.z - L.lock.z);
  951|           L.swingDur = THREE.MathUtils.clamp(0.14 + 0.35 * dd / reach, 0.14, 0.30);
  952|         }
  953|         plan(L.swingDur, L.swingTo);
  954|       }
  955|     }
  956| 
  957|     if (L.mode === 'swing') {
  958|       L.swingT = Math.min(1, L.swingT + dt / Math.max(0.05, L.swingDur));
  959|       plan((1 - L.swingT) * L.swingDur, this._tw);
  960|       L.swingTo.lerp(this._tw, 1 - Math.exp(-dt / 0.05));
  961|       const sT = L.swingT;
  962|       const e = sT * sT * (3 - 2 * sT);
  963|       out.lerpVectors(L.swingFrom, L.swingTo, e);
  964|       const dd = Math.hypot(L.swingTo.x - L.swingFrom.x, L.swingTo.z - L.swingFrom.z);
  965|       const k = reach / Math.max(1e-3, L.L1 + L.L2);
  966|       const liftH = L.lift * k * Math.min(1, 0.35 + dd / stride);
  967|       out.y = Math.max(out.y, gy(out.x, out.z)) + Math.sin(sT * Math.PI) * liftH;
  968|       if (sT >= 1) { L.mode = 'plant'; L.landPending = true; L.landT = 0; L.lock.copy(out); }
  969|       return;
  970|     }
  971|     L.lock.y += (gy(L.lock.x, L.lock.z) - L.lock.y) * Math.min(1, dt * 20);
  972|     out.copy(L.lock);
  973|   }

  984|     while (d < -Math.PI) d += Math.PI * 2;
  985|     return d;

  994|    * The face. One place writes eyes, lids, brows, mouth and blush, so an

  997|   private _updateFace(dt: number, act: number): void {
  998|     const h = this.handle;
  999|     const closed = this.blinkT > 0 ? Math.sin(Math.PI * (1 - this.blinkT)) : 0;
 1000|     const f = this.face;
