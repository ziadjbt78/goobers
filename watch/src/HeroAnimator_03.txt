  751|       this._goal.copy(this._ft);
  752|       const hip = L.hip.getWorldPosition(this._hw2);
  753|       const rMax = 0.985 * L.reachW;
  754|       this._clampReach(this._ft, hip, rMax);
  755|       // v13 CLOSED-LOOP IK: squash shears the leg's parent space, so the aim
  756|       // misses. Measure where the ankle REALLY went and re-aim, up to 3 times.
  757|       let err = 0;
  758|       for (let it = 0; it < 3; it++) {
  759|         solveLeg({ hip: L.hip, knee: L.knee, ankle: L.ankle, targetWorld: this._ft, poleWorld: this._pole, prevKnee: L.kneeLocal });
  760|         L.ankle.updateWorldMatrix(true, false);
  761|         L.ankle.getWorldPosition(this._aw);
  762|         const ex = this._goal.x - this._aw.x, ey = this._goal.y - this._aw.y, ez = this._goal.z - this._aw.z;
  763|         err = Math.hypot(ex, ey, ez);
  764|         if (err < 0.0015) break;
  765|         this._ft.x += ex; this._ft.y += ey; this._ft.z += ez;
  766|         this._clampReach(this._ft, hip, rMax);
  767|       }
  768|       L.err = err;
  769|       if (L.mode === 'plant' && !L.landPending && err > this.liftStats.maxErr) this.liftStats.maxErr = err;
  770|       const maxD = (L.mode === 'swing' || L.landPending ? LEG_RATE : PLANT_RATE) * dt;
  771|       for (let k = 0; k < 3; k++) {
  772|         const bone = k === 0 ? L.hip : k === 1 ? L.knee : L.ankle;
  773|         const q = L.prev[k];
  774|         const ang = 2 * Math.acos(Math.min(1, Math.abs(bone.quaternion.dot(q))));
  775|         if (ang > maxD && ang > 1e-6) { _lim.copy(bone.quaternion); bone.quaternion.copy(q).slerp(_lim, maxD / ang); }
  776|       }
  777|       L.prev[0].copy(L.hip.quaternion);
  778|       L.prev[1].copy(L.knee.quaternion);
  779|       L.prev[2].copy(L.ankle.quaternion);
  780|       if (L.landPending) { L.ankle.updateWorldMatrix(true, false); L.ankle.getWorldPosition(L.lock); L.landPending = false; }
  781|     }
  782|     if (this.petted) {
  783|       const back = this.legs[this.legs.length - 1];
  784|       const kick = Math.max(0, Math.sin(this.kickT * 9)) * 0.85;
  785|       if (kick > 0.02) back.hip.rotateX(-kick);
  786|     }
  787|   }

  789|   private _clampReach(v: THREE.Vector3, hip: THREE.Vector3, r: number): void {
  790|     if (r <= 1e-4) return;
  791|     const d = v.distanceTo(hip);
  792|     if (d > r) v.sub(hip).multiplyScalar(r / d).add(hip);
  793|   }

  799|   /**
  800|    * v12 world stepper. Leg geometry is measured LIVE in world space (creature
  801|    * scale and squash included). Lifts are edge-triggered on the gait beat; a
  802|    * swing re-aims every frame at where the foot must land so the coming stance
  803|    * is centred under the hip; guards lift early only when the leg truly runs
  804|    * out of reach, with an airborne-count limit so a creature never lifts every
  805|    * foot at once.
  806|    */
  807|   private _worldFootTarget(L: LegRig, out: THREE.Vector3, dt: number): void {
  808|     const g = this.handle.group;
  809|     const gp = g.position;
  810|     const hd = this.drive.heading;
  811|     const fx = Math.sin(hd), fz = Math.cos(hd);
  812|     const speed = this.drive.speed;
  813|     const moving = speed > 0.05;
  814|     const stride = Math.max(0.05, this.drive.stride);
  815|     const ground = this.drive.groundY;
  816|     const gy = (x: number, z: number): number => (ground ? ground(x, z) : gp.y) + L.plant.y;
  817| 
  818|     const hip = L.hip.getWorldPosition(this._vw);
  819|     const knee = L.knee.getWorldPosition(this._kw);
  820|     const ank = L.ankle.getWorldPosition(this._aw);
  821|     const reach = Math.max(1e-3, hip.distanceTo(knee) + knee.distanceTo(ank));
  822|     const rest = this._rw.set(L.plant.x, 0, L.plant.z).applyMatrix4(g.matrixWorld);
  823|     rest.y = gy(rest.x, rest.z);
  824| 
  825|     if (!L.init) {
  826|       L.lock.copy(rest); L.mode = 'plant'; L.landT = 1;
  827|       L.prevU = (this.externalPhase + L.phase) % 1;
  828|       L.init = true;
  829|     }
  830| 
  831|     const dyHip = Math.max(0, hip.y - rest.y);
  832|     const r92 = 0.92 * reach;
  833|     const restFlat = Math.hypot(rest.x - hip.x, rest.z - hip.z);
  834|     const flat = Math.max(0.25 * reach, 1.3 * restFlat, Math.sqrt(Math.max(0, r92 * r92 - dyHip * dyHip)));
  835|     const hardR = Math.max(0.97 * reach, 1.08 * hip.distanceTo(rest));
  836|     const cycle = moving ? stride / speed : 1.2;
  837|     const half = Math.min(0.5 * DUTY * stride, 0.6 * flat);
  838| 
  839|     const u = (this.externalPhase + L.phase) % 1;
  840|     const onBeat = moving && crossedPhase(L.prevU, u, DUTY);
  841|     L.prevU = u;
  842|     L.landT += dt;
  843| 
  844|     // landing point: half a stance AHEAD of where the body will be at touchdown,
  845|     // clamped inside the leg's reach from where the hip will be
  846|     const plan = (remaining: number, o: THREE.Vector3): THREE.Vector3 => {
  847|       const travel = moving ? speed * remaining : 0;
  848|       const ahead = moving ? travel + half : 0;
  849|       o.set(rest.x + fx * ahead, 0, rest.z + fz * ahead);
  850|       const hx = hip.x + fx * travel, hz = hip.z + fz * travel;
  851|       const vx = o.x - hx, vz = o.z - hz;
  852|       const vd = Math.hypot(vx, vz), lim = 0.85 * flat;
  853|       if (vd > lim) { o.x = hx + (vx / vd) * lim; o.z = hz + (vz / vd) * lim; }
  854|       o.y = gy(o.x, o.z);
  855|       return o;
  856|     };
  857| 
  858|     if (L.mode === 'plant') {
  859|       // v13: a planted foot the solver cannot hold steps instead of sliding
  860|       const strained = L.err > 0.01 && L.landT > 0.05;
  861|       const hard = strained || Math.hypot(L.lock.x - hip.x, L.lock.z - hip.z) > flat || hip.distanceTo(L.lock) > hardR;
  862|       const rx = L.lock.x - rest.x, rz = L.lock.z - rest.z;
  863|       const along = rx * fx + rz * fz;
  864|       const lagging = moving && along < -1.35 * half;
  865|       const drifted = !moving && Math.hypot(rx, rz) > Math.max(0.3 * flat, 0.05 * reach);
  866|       let air = 0;
  867|       for (const o of this.legs) if (o !== L && o.mode === 'swing') air++;
  868|       const maxAir = this.legs.length >= 6 ? 3 : this.legs.length >= 4 ? 2 : 1;
  869|       const beat = onBeat && L.landT > 0.2 * cycle;
  870|       const soft = (lagging || drifted) && air < maxAir && L.landT > 0.08;
  871|       if (hard || beat || soft) {
  872|         const ls = this.liftStats;
  873|         if (strained) ls.strain++; else if (hard) ls.hard++; else if (beat) ls.beat++; else ls.soft++;
  874|         L.mode = 'swing';
  875|         L.forced = !beat;
  876|         L.swingT = 0;
  877|         L.swingFrom.copy(L.lock);
  878|         if (beat && !hard) {
  879|           L.swingDur = THREE.MathUtils.clamp((1 - DUTY) * cycle, 0.12, 0.40);
  880|         } else {
  881|           plan(0.2, L.swingTo);
  882|           const dd = Math.hypot(L.swingTo.x - L.lock.x, L.swingTo.z - L.lock.z);
  883|           L.swingDur = THREE.MathUtils.clamp(0.14 + 0.35 * dd / reach, 0.14, 0.30);
  884|         }
  885|         plan(L.swingDur, L.swingTo);
  886|       }
  887|     }
  888| 
  889|     if (L.mode === 'swing') {
  890|       L.swingT = Math.min(1, L.swingT + dt / Math.max(0.05, L.swingDur));
  891|       plan((1 - L.swingT) * L.swingDur, this._tw);
  892|       L.swingTo.lerp(this._tw, 1 - Math.exp(-dt / 0.05));
  893|       const sT = L.swingT;
  894|       const e = sT * sT * (3 - 2 * sT);
  895|       out.lerpVectors(L.swingFrom, L.swingTo, e);
  896|       const dd = Math.hypot(L.swingTo.x - L.swingFrom.x, L.swingTo.z - L.swingFrom.z);
  897|       const k = reach / Math.max(1e-3, L.L1 + L.L2);
  898|       const liftH = L.lift * k * Math.min(1, 0.35 + dd / stride);
  899|       out.y = Math.max(out.y, gy(out.x, out.z)) + Math.sin(sT * Math.PI) * liftH;
  900|       if (sT >= 1) { L.mode = 'plant'; L.landPending = true; L.landT = 0; L.lock.copy(out); }
  901|       return;
  902|     }
  903|     L.lock.y += (gy(L.lock.x, L.lock.z) - L.lock.y) * Math.min(1, dt * 20);
  904|     out.copy(L.lock);
  905|   }
