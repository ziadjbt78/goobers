  751|       this._pole.copy(L.poleLocal).applyQuaternion(gq);

  774|       if (L.mode === 'plant' && !L.landPending && err > this.liftStats.maxErr) this.liftStats.maxErr = err;

  777|         const bone = k === 0 ? L.hip : k === 1 ? L.knee : L.ankle;
  778|         const q = L.prev[k];
  779|         const ang = 2 * Math.acos(Math.min(1, Math.abs(bone.quaternion.dot(q))));
  780|         if (ang > maxD && ang > 1e-6) { _lim.copy(bone.quaternion); bone.quaternion.copy(q).slerp(_lim, maxD / ang); }
  781|       }
  782|       L.ankle.updateWorldMatrix(true, false);

  785|       L.prev[1].copy(L.knee.quaternion);
  786|       L.prev[2].copy(L.ankle.quaternion);

  788|     }
  789|     if (this.petted) {
  790|       const back = this.legs[this.legs.length - 1];
  791|       const kick = Math.max(0, Math.sin(this.kickT * 9)) * 0.85;
  792|       if (kick > 0.02) back.hip.rotateX(-kick);
  793|     }
  794|   }

  806|   /**
  807|    * v12 world stepper. Leg geometry is measured LIVE in world space (creature
  808|    * scale and squash included). Lifts are edge-triggered on the gait beat; a
  809|    * swing re-aims every frame at where the foot must land so the coming stance
  810|    * is centred under the hip; guards lift early only when the leg truly runs
  811|    * out of reach, with an airborne-count limit so a creature never lifts every
  812|    * foot at once.
  813|    */
  814|   private _worldFootTarget(L: LegRig, out: THREE.Vector3, dt: number): void {
  815|     const g = this.handle.group;
  816|     const gp = g.position;
  817|     const hd = this.drive.heading;
  818|     const fx = Math.sin(hd), fz = Math.cos(hd);
  819|     const speed = this.drive.speed;
  820|     const moving = speed > 0.05;
  821|     const stride = Math.max(0.05, this.drive.stride);
  822|     const ground = this.drive.groundY;
  823|     const gy = (x: number, z: number): number => (ground ? ground(x, z) : gp.y) + L.plant.y;
  824| 
  825|     const hip = L.hip.getWorldPosition(this._vw);
  826|     const knee = L.knee.getWorldPosition(this._kw);
  827|     const ank = L.ankle.getWorldPosition(this._aw);
  828|     const reach = Math.max(1e-3, hip.distanceTo(knee) + knee.distanceTo(ank));
  829|     const rest = this._rw.set(L.plant.x, 0, L.plant.z).applyMatrix4(g.matrixWorld);
  830|     rest.y = gy(rest.x, rest.z);
  831| 
  832|     if (!L.init) {
  833|       L.lock.copy(rest); L.mode = 'plant'; L.landT = 1;
  834|       L.prevU = (this.externalPhase + L.phase) % 1;
  835|       L.init = true;
  836|     }
  837| 
  838|     const dyHip = Math.max(0, hip.y - rest.y);
  839|     const r92 = 0.92 * reach;
  840|     const restFlat = Math.hypot(rest.x - hip.x, rest.z - hip.z);
  841|     const flat = Math.max(0.25 * reach, 1.3 * restFlat, Math.sqrt(Math.max(0, r92 * r92 - dyHip * dyHip)));
  842|     const hardR = Math.max(0.97 * reach, 1.08 * hip.distanceTo(rest));
  843|     const cycle = moving ? stride / speed : 1.2;
  844|     const half = Math.min(0.5 * DUTY * stride, 0.6 * flat);
  845| 
  846|     const u = (this.externalPhase + L.phase) % 1;
  847|     const onBeat = moving && crossedPhase(L.prevU, u, DUTY);
  848|     L.prevU = u;
  849|     L.landT += dt;
  850| 
  851|     // landing point: half a stance AHEAD of where the body will be at touchdown,
  852|     // clamped inside the leg's reach from where the hip will be
  853|     const plan = (remaining: number, o: THREE.Vector3): THREE.Vector3 => {
  854|       const travel = moving ? speed * remaining : 0;
  855|       const ahead = moving ? travel + half : 0;
  856|       o.set(rest.x + fx * ahead, 0, rest.z + fz * ahead);
  857|       const hx = hip.x + fx * travel, hz = hip.z + fz * travel;
  858|       const vx = o.x - hx, vz = o.z - hz;
  859|       const vd = Math.hypot(vx, vz), lim = 0.85 * flat;
  860|       if (vd > lim) { o.x = hx + (vx / vd) * lim; o.z = hz + (vz / vd) * lim; }
  861|       o.y = gy(o.x, o.z);
  862|       return o;
  863|     };
  864| 
  865|     if (L.mode === 'plant') {
  866|       // v13: a planted foot the solver cannot hold steps instead of sliding
  867|       const strained = L.err > 0.01 && L.landT > 0.05;
  868|       const hard = strained || Math.hypot(L.lock.x - hip.x, L.lock.z - hip.z) > flat || hip.distanceTo(L.lock) > hardR;
  869|       const rx = L.lock.x - rest.x, rz = L.lock.z - rest.z;
  870|       const along = rx * fx + rz * fz;
  871|       const lagging = moving && along < -1.35 * half;
  872|       const drifted = !moving && Math.hypot(rx, rz) > Math.max(0.3 * flat, 0.05 * reach);
  873|       let air = 0;
  874|       for (const o of this.legs) if (o !== L && o.mode === 'swing') air++;
  875|       const maxAir = this.legs.length >= 6 ? 3 : this.legs.length >= 4 ? 2 : 1;
  876|       const beat = onBeat && L.landT > 0.2 * cycle;
  877|       const soft = (lagging || drifted) && air < maxAir && L.landT > 0.08;
  878|       if (hard || beat || soft) {
  879|         const ls = this.liftStats;
  880|         if (strained) ls.strain++; else if (hard) ls.hard++; else if (beat) ls.beat++; else ls.soft++;
  881|         L.mode = 'swing';
  882|         L.forced = !beat;
  883|         L.swingT = 0;
  884|         L.swingFrom.copy(L.lock);
  885|         if (beat && !hard) {
  886|           L.swingDur = THREE.MathUtils.clamp((1 - DUTY) * cycle, 0.12, 0.40);
  887|         } else {
  888|           plan(0.2, L.swingTo);
  889|           const dd = Math.hypot(L.swingTo.x - L.lock.x, L.swingTo.z - L.lock.z);
  890|           L.swingDur = THREE.MathUtils.clamp(0.14 + 0.35 * dd / reach, 0.14, 0.30);
  891|         }
  892|         plan(L.swingDur, L.swingTo);
  893|       }
  894|     }
  895| 
  896|     if (L.mode === 'swing') {
  897|       L.swingT = Math.min(1, L.swingT + dt / Math.max(0.05, L.swingDur));
  898|       plan((1 - L.swingT) * L.swingDur, this._tw);
  899|       L.swingTo.lerp(this._tw, 1 - Math.exp(-dt / 0.05));
  900|       const sT = L.swingT;
  901|       const e = sT * sT * (3 - 2 * sT);
  902|       out.lerpVectors(L.swingFrom, L.swingTo, e);
  903|       const dd = Math.hypot(L.swingTo.x - L.swingFrom.x, L.swingTo.z - L.swingFrom.z);
  904|       const k = reach / Math.max(1e-3, L.L1 + L.L2);
  905|       const liftH = L.lift * k * Math.min(1, 0.35 + dd / stride);
  906|       out.y = Math.max(out.y, gy(out.x, out.z)) + Math.sin(sT * Math.PI) * liftH;
  907|       if (sT >= 1) { L.mode = 'plant'; L.landPending = true; L.landT = 0; L.lock.copy(out); }
  908|       return;
  909|     }
  910|     L.lock.y += (gy(L.lock.x, L.lock.z) - L.lock.y) * Math.min(1, dt * 20);
  911|     out.copy(L.lock);
  912|   }
