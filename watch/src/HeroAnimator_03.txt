  784|     const h = this.handle;

  787|     // v13 CROUCH: stubby legs plus the cartoon bob put the hips higher than the

  800|       const gy = (ground ? ground(this._rw.x, this._rw.z) : h.group.position.y) + L.plant.y;
  801|       need = Math.max(need, Math.min(0.5 * L.reachW, hip.y - gy - kH * L.reachW));
  802|     }

  805|     this.crouch += (need - this.crouch) * Math.min(1, dt / (need > this.crouch ? 0.06 : 0.25));

  810|       root.position.y -= this.crouch / Math.max(1e-4, ps);

  817|       // or rolls the body carries the knee direction with it instead of flipping it
  818|       const par = L.hip.parent;
  819|       if (par) {

  822|         this._pq.multiply(this._iq.copy(L.parRel).invert());
  823|         this._pole.copy(L.poleLocal).applyQuaternion(this._pq);
  824|       } else {
  825|         this._pole.copy(L.poleLocal).applyQuaternion(gq);
  826|       }

  828|       else L.poleSm.lerp(this._pole, 1 - Math.exp(-dt / 0.08)).normalize();
  829|       this._pole.copy(L.poleSm);
  830|       if (L.relaunch) {
  831|         // v15: back from hanging: swing in from where the ankle REALLY is
  832|         L.relaunch = false;

  835|           L.mode = 'swing'; L.swingT = 0; L.swingDur = 0.18; L.forced = true;
  836|           L.swingFrom.copy(this._aw); L.swingTo.copy(this._aw);
  837|         }

  841|       const rMax = 0.985 * L.reachW;

  850|           if (this._ft.y < minY) { L.lifted = minY - this._ft.y > 0.01; this._ft.y = minY; }
  851|         }

  853|       this._softReach(this._ft, hip, L.reachW);
  854|       // v17: clamped along hip->lock = raised off the ground, not sliding on it
  855|       if (L.mode === 'plant' && !L.landPending && this._ft.distanceTo(this._lk) > 0.01) L.lifted = true;

  857|       // The IK stays exact, so the rendered foot IS the target: smooth AND no slip.
  858|       {

  860|         if (!LT.tPrev) LT.tPrev = this._ft.clone();

  864|           if (mv > cap && mv > 1e-6) this._ft.sub(LT.tPrev).multiplyScalar(cap / mv).add(LT.tPrev);

  871|       const LX = L as unknown as { fb: boolean; sw: boolean; bendRate: number; bendPrev: number };
  872|       LX.fb = false; LX.sw = false;
  873|       // v20: knee swivel is limited against the knee of the PREVIOUS FRAME (body space)
  874|       this._kp.copy(L.kneeLocal);
  875|       let err = 0;
  876|       for (let it = 0; it < 3; it++) {
  877|         // v19 EXACT solve: in the hip parent's own (squashed) space, no world-frame twist, feet face the heading
  878|         if (solveLeg({ hip: L.hip, knee: L.knee, ankle: L.ankle, targetWorld: this._ft, poleWorld: this._pole, prevKnee: L.kneeLocal, exact: true, footWorld: gq, swivelFrom: this._kp, maxSwivel: SWIVEL_RATE * dt })) LX.fb = true;

  911|       L.prev[1].copy(L.knee.quaternion);

  922|   /** v18 soft IK: below SOFT_START the target is untouched; beyond it the reach approaches
  923|    *  SOFT_TOP asymptotically, so a near-straight knee eases instead of snapping. */
  924|   private _softReach(v: THREE.Vector3, hip: THREE.Vector3, reach: number): void {
  925|     if (reach <= 1e-4) return;
  926|     const d = v.distanceTo(hip);
  927|     const ds = SOFT_START * reach;
  928|     if (d <= ds) return;
  929|     const s = (SOFT_TOP - SOFT_START) * reach;
  930|     const d2 = ds + s * (1 - Math.exp(-(d - ds) / s));
  931|     v.sub(hip).multiplyScalar(d2 / d).add(hip);
  932|   }

  934|   private _clampReach(v: THREE.Vector3, hip: THREE.Vector3, r: number): void {
  935|     if (r <= 1e-4) return;
  936|     const d = v.distanceTo(hip);
  937|     if (d > r) v.sub(hip).multiplyScalar(r / d).add(hip);
  938|   }

  948|   /**
  949|    * v12 world stepper. Leg geometry is measured LIVE in world space (creature
  950|    * scale and squash included). Lifts are edge-triggered on the gait beat; a
  951|    * swing re-aims every frame at where the foot must land so the coming stance
  952|    * is centred under the hip; guards lift early only when the leg truly runs
  953|    * out of reach, with an airborne-count limit so a creature never lifts every

  956|   private _worldFootTarget(L: LegRig, out: THREE.Vector3, dt: number): void {
  957|     const g = this.handle.group;
  958|     const gp = g.position;
  959|     const hd = this.drive.heading;
  960|     const fx = Math.sin(hd), fz = Math.cos(hd);
  961|     const speed = this.drive.speed;
  962|     const moving = speed > 0.05;
  963|     const stride = Math.max(0.05, this.drive.stride);
  964|     const ground = this.drive.groundY;
  965|     const gy = (x: number, z: number): number => (ground ? ground(x, z) : gp.y) + L.plant.y;
  966| 
  967|     const hip = this._vw.setFromMatrixPosition(L.hip.matrixWorld);
  968|     const knee = this._kw.setFromMatrixPosition(L.knee.matrixWorld);
  969|     const ank = this._aw.setFromMatrixPosition(L.ankle.matrixWorld);
  970|     const reach = Math.max(1e-3, hip.distanceTo(knee) + knee.distanceTo(ank));
  971|     const rest = this._rw.set(L.plant.x, 0, L.plant.z).applyMatrix4(g.matrixWorld);
  972|     rest.y = gy(rest.x, rest.z);
  973| 
  974|     if (!L.init) {
  975|       L.lock.copy(rest); L.mode = 'plant'; L.landT = 1;
  976|       L.prevU = (this.externalPhase + L.phase) % 1;
  977|       L.init = true;
  978|     }
  979| 
  980|     const dyHip = Math.max(0, hip.y - rest.y);
  981|     const r92 = 0.92 * reach;
  982|     const restFlat = Math.hypot(rest.x - hip.x, rest.z - hip.z);
  983|     const flat = Math.max(0.25 * reach, 1.3 * restFlat, Math.sqrt(Math.max(0, r92 * r92 - dyHip * dyHip)));
  984|     const hardR = Math.max(0.97 * reach, 1.08 * hip.distanceTo(rest));
  985|     const cycle = moving ? stride / speed : 1.2;
  986|     const half = Math.min(0.5 * DUTY * stride, 0.6 * flat);
  987| 
  988|     const u = (this.externalPhase + L.phase) % 1;
  989|     const onBeat = moving && crossedPhase(L.prevU, u, DUTY);
  990|     L.prevU = u;
  991|     L.landT += dt;
  992| 
  993|     // landing point: half a stance AHEAD of where the body will be at touchdown,
  994|     // clamped inside the leg's reach from where the hip will be
  995|     const plan = (remaining: number, o: THREE.Vector3): THREE.Vector3 => {
  996|       const travel = moving ? speed * remaining : 0;
  997|       const ahead = moving ? travel + half : 0;
  998|       o.set(rest.x + fx * ahead, 0, rest.z + fz * ahead);
  999|       const hx = hip.x + fx * travel, hz = hip.z + fz * travel;
 1000|       const vx = o.x - hx, vz = o.z - hz;
