  828|         L.relaunch = false;

  832|           L.swingFrom.copy(this._aw); L.swingTo.copy(this._aw);

  902|   /** v18 soft IK: below SOFT_START the target is untouched; beyond it the reach approaches
  903|    *  SOFT_TOP asymptotically, so a near-straight knee eases instead of snapping. */
  904|   private _softReach(v: THREE.Vector3, hip: THREE.Vector3, reach: number): void {
  905|     if (reach <= 1e-4) return;
  906|     const d = v.distanceTo(hip);
  907|     const ds = SOFT_START * reach;
  908|     if (d <= ds) return;
  909|     const s = (SOFT_TOP - SOFT_START) * reach;
  910|     const d2 = ds + s * (1 - Math.exp(-(d - ds) / s));
  911|     v.sub(hip).multiplyScalar(d2 / d).add(hip);
  912|   }
  913| 
  914|   private _clampReach(v: THREE.Vector3, hip: THREE.Vector3, r: number): void {
  915|     if (r <= 1e-4) return;
  916|     const d = v.distanceTo(hip);
  917|     if (d > r) v.sub(hip).multiplyScalar(r / d).add(hip);
  918|   }

  927|   /**
  928|    * v12 world stepper. Leg geometry is measured LIVE in world space (creature
  929|    * scale and squash included). Lifts are edge-triggered on the gait beat; a
  930|    * swing re-aims every frame at where the foot must land so the coming stance
  931|    * is centred under the hip; guards lift early only when the leg truly runs
  932|    * out of reach, with an airborne-count limit so a creature never lifts every
  933|    * foot at once.
  934|    */
  935|   private _worldFootTarget(L: LegRig, out: THREE.Vector3, dt: number): void {
  936|     const g = this.handle.group;
  937|     const gp = g.position;
  938|     const hd = this.drive.heading;
  939|     const fx = Math.sin(hd), fz = Math.cos(hd);
  940|     const speed = this.drive.speed;
  941|     const moving = speed > 0.05;
  942|     const stride = Math.max(0.05, this.drive.stride);
  943|     const ground = this.drive.groundY;
  944|     const gy = (x: number, z: number): number => (ground ? ground(x, z) : gp.y) + L.plant.y;
  945| 
  946|     const hip = this._vw.setFromMatrixPosition(L.hip.matrixWorld);
  947|     const knee = this._kw.setFromMatrixPosition(L.knee.matrixWorld);
  948|     const ank = this._aw.setFromMatrixPosition(L.ankle.matrixWorld);
  949|     const reach = Math.max(1e-3, hip.distanceTo(knee) + knee.distanceTo(ank));
  950|     const rest = this._rw.set(L.plant.x, 0, L.plant.z).applyMatrix4(g.matrixWorld);
  951|     rest.y = gy(rest.x, rest.z);
  952| 
  953|     if (!L.init) {
  954|       L.lock.copy(rest); L.mode = 'plant'; L.landT = 1;
  955|       L.prevU = (this.externalPhase + L.phase) % 1;
  956|       L.init = true;
  957|     }
  958| 
  959|     const dyHip = Math.max(0, hip.y - rest.y);
  960|     const r92 = 0.92 * reach;
  961|     const restFlat = Math.hypot(rest.x - hip.x, rest.z - hip.z);
  962|     const flat = Math.max(0.25 * reach, 1.3 * restFlat, Math.sqrt(Math.max(0, r92 * r92 - dyHip * dyHip)));
  963|     const hardR = Math.max(0.97 * reach, 1.08 * hip.distanceTo(rest));
  964|     const cycle = moving ? stride / speed : 1.2;
  965|     const half = Math.min(0.5 * DUTY * stride, 0.6 * flat);
  966| 
  967|     const u = (this.externalPhase + L.phase) % 1;
  968|     const onBeat = moving && crossedPhase(L.prevU, u, DUTY);
  969|     L.prevU = u;
  970|     L.landT += dt;
  971| 
  972|     // landing point: half a stance AHEAD of where the body will be at touchdown,
  973|     // clamped inside the leg's reach from where the hip will be
  974|     const plan = (remaining: number, o: THREE.Vector3): THREE.Vector3 => {
  975|       const travel = moving ? speed * remaining : 0;
  976|       const ahead = moving ? travel + half : 0;
  977|       o.set(rest.x + fx * ahead, 0, rest.z + fz * ahead);
  978|       const hx = hip.x + fx * travel, hz = hip.z + fz * travel;
  979|       const vx = o.x - hx, vz = o.z - hz;
  980|       const vd = Math.hypot(vx, vz), lim = 0.85 * flat;
  981|       if (vd > lim) { o.x = hx + (vx / vd) * lim; o.z = hz + (vz / vd) * lim; }
  982|       o.y = gy(o.x, o.z);
  983|       return o;
  984|     };
  985| 
  986|     if (L.mode === 'plant') {
  987|       // v13: a planted foot the solver cannot hold steps instead of sliding
  988|       const strained = L.err > 0.01 && L.landT > 0.05;
  989|       const hard = strained || Math.hypot(L.lock.x - hip.x, L.lock.z - hip.z) > flat || hip.distanceTo(L.lock) > hardR;
  990|       const rx = L.lock.x - rest.x, rz = L.lock.z - rest.z;
  991|       const along = rx * fx + rz * fz;
  992|       const lagging = moving && along < -1.35 * half;
  993|       const drifted = !moving && Math.hypot(rx, rz) > Math.max(0.3 * flat, 0.05 * reach);
  994|       let air = 0;
  995|       for (const o of this.legs) if (o !== L && o.mode === 'swing') air++;
  996|       const maxAir = this.legs.length >= 6 ? 3 : this.legs.length >= 4 ? 2 : 1;
  997|       const beat = onBeat && L.landT > 0.2 * cycle;
  998|       const soft = (lagging || drifted) && air < maxAir && L.landT > 0.08;
  999|       if (hard || beat || soft) {
 1000|         const ls = this.liftStats;
