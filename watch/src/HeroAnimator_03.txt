  755|     let d = a;
  756|     while (d > Math.PI) d -= Math.PI * 2;
  757|     while (d < -Math.PI) d += Math.PI * 2;
  758|     return d;
  759|   }

  766|   /**
  767|    * The face. One place writes eyes, lids, brows, mouth and blush, so an
  768|    * expression can never half-apply.
  769|    */
  770|   private _updateFace(dt: number, act: number): void {
  771|     const h = this.handle;
  772|     const closed = this.blinkT > 0 ? Math.sin(Math.PI * (1 - this.blinkT)) : 0;
  773|     const f = this.face;
  774| 
  775|     const happy = this.mood === 'happy' ? 1 : 0;
  776|     const love = this.mood === 'love' ? 1 : 0;
  777|     const sad = this.mood === 'sad' ? 1 : 0;
  778|     const surprised = this.mood === 'surprised' || this.mood === 'scared' ? 1 : 0;
  779|     const sleepy = this.mood === 'sleepy' ? 1 : 0;
  780|     const dizzy = this.mood === 'dizzy' ? 1 : 0;
  781| 
  782|     // target face values, then blend — never a hard set
  783|     let lidT = f.lid;
  784|     let openT = f.mouthOpen;
  785|     let wideT = f.mouthWide;
  786|     let blushT = f.blush;
  787|     let pupT = 1;
  788|     let spinT = 0;
  789| 
  790|     if (this.petted) { lidT = 0.72; blushT = 1; pupT = 1.06; wideT = 1.15; openT = Math.max(openT, 0.25); }
  791|     if (this.held) { lidT = 0; blushT = 0.2; pupT = 1.35; openT = Math.max(openT, 0.75); wideT = 0.8; }
  792|     if (dizzy) { pupT = 0.7; spinT = 1; }
  793|     if (sleepy) { lidT = Math.max(lidT, 0.62); }
  794|     if (happy || love) { lidT = Math.max(lidT, 0.55); blushT = Math.max(blushT, 0.8); pupT = 1.08; openT = Math.max(openT, 0.35); }
  795|     if (surprised) { lidT = 0; pupT = 1.3; openT = Math.max(openT, 0.85); wideT = 0.85; }
  796|     if (sad) { lidT = Math.max(lidT, 0.28); pupT = 1.12; wideT = 0.85; }
  797|     if (this.action === 'chomp' || this.action === 'eat') {
  798|       const chew = 0.5 + 0.5 * Math.sin(this.actionT * 22);
  799|       openT = Math.max(openT, chew * 0.95);
  800|       wideT = 1.1;
  801|       blushT = Math.max(blushT, 0.55);
  802|     }
  803|     if (this.action === 'wave') { lidT = Math.max(lidT, 0.5); openT = Math.max(openT, 0.7); wideT = 1.2; }
  804|     if (this.idleAct === 'yawn' && this.idleActW > 0.2) { openT = Math.max(openT, 0.95 * this.idleActW); wideT = 0.9; lidT = Math.max(lidT, 0.75 * this.idleActW); }
  805|     if (this.idleAct === 'sniff' && this.idleActW > 0.2) { wideT = 0.8; }
  806| 
  807|     const k = Math.min(1, dt / 0.22);
  808|     f.lid += (lidT - f.lid) * k;
  809|     f.mouthOpen += (openT - f.mouthOpen) * k;
  810|     f.mouthWide += (wideT - f.mouthWide) * k;
  811|     f.blush += (blushT - f.blush) * k;
  812|     f.pupil += (pupT - f.pupil) * Math.min(1, dt * 6);
  813|     f.gazeSpin += (spinT - f.gazeSpin) * Math.min(1, dt * 4);
  814| 
  815|     this.pupil += ((1 + surprised * 0.10 - happy * 0.05 + (sleepy + dizzy) * -0.08) - this.pupil) * Math.min(1, dt * 5);
  816|     this.lidClose += (f.lid - this.lidClose) * Math.min(1, dt * 6);
  817| 
  818|     const dizzySpin = f.gazeSpin * this.t * 5.5;
  819|     for (const e of h.eyes) {
  820|       // a one-eyed wink, when the blink roll said so
  821|       const lid = this.blinkSide > 0 ? (e.side > 0 ? closed : 0)
  822|         : this.blinkSide < 0 ? (e.side < 0 ? closed : 0)
  823|           : closed;
  824|       e.lidPivot.rotation.x = -Math.PI * 0.5 + Math.max(0, Math.min(0.985, lid + this.lidClose)) * Math.PI * 0.98;
  825|       // eyes lead the head: the socket carries the gaze, the head carries the pose
  826|       const spinY = f.gazeSpin > 0.05 ? Math.cos(dizzySpin) * 0.45 * f.gazeSpin : 0;
  827|       const spinP = f.gazeSpin > 0.05 ? Math.sin(dizzySpin) * 0.35 * f.gazeSpin : 0;
  828|       e.socket.rotation.set(this.gazePitch + spinP, this.gazeYaw + spinY, f.gazeSpin * Math.sin(dizzySpin * 0.7) * 0.4);
  829|       const pup = this.pupil * f.pupil * (1 + surprised * 0.05);
  830|       e.pupil.scale.set(pup, pup * (1 - sad * 0.12), 1);
  831|       const browUp = surprised * 1.0 + sad * 0.45;
  832|       e.brow.visible = browUp > 0.08;
  833|       if (e.brow.visible) {
  834|         e.brow.position.y = e.radius * (0.92 + browUp * 0.20);
  835|         // angry/sad tilt: inner end up for sad, down for angry
  836|         e.brow.rotation.z = (sad * -0.30 + (this.mood === 'scared' ? 0.25 : 0)) * (e.side >= 0 ? 1 : -1);
  837|       }
  838|     }
  839| 
  840|     const open = f.mouthOpen;
  841|     h.mouthOpen.visible = open > 0.06;
  842|     h.mouth.visible = open <= 0.06;
  843|     if (h.mouthOpen.visible) {
  844|       const w = f.mouthWide;
  845|       // cheeks puff 20% on a chomp: the open mouth and the blush both swell
  846|       h.mouthOpen.scale.set(w * (1 + open * 0.15), (0.6 + open * 0.75) * (1 + open * 0.12), 1 * (1 + open * 0.2));
  847|     }
  848|     if (h.mouth.visible) {
  849|       h.mouth.scale.set(f.mouthWide, 1 - (1 - f.mouthWide) * 0.5 + (sad ? -0.25 : 0), 1);
  850|       h.mouth.rotation.z = -Math.PI * 0.5 - 1.55 * 0.5 + (sad ? 0.5 : 0);
  851|     }
  852|     for (const b of h.blush) b.scale.setScalar(0.85 + f.blush * 0.55 + open * 0.08);
  853|     void act;
  854|   }

  856|   /** Per-species locomotion flavour, keyed off the template NAME: PIP and
  857|    *  MOCHI share the `walk` tag but must move completely differently. */
  858|   private _gaitFlavour(loco: number, speed01: number): { roll: number; spine: number; headPitch: number } {
  859|     const t = this.template;
  860|     if (loco <= 0.001) return { roll: 0, spine: 0, headPitch: 0 };
  861|     const s = Math.sin(this.phase * Math.PI * 2);
  862|     const key = (t.name || '').toUpperCase();
  863|     if (key.startsWith('PIP') || t.locomotion === 'walk') {
  864|       // PIP waddle: hard side-to-side sway, shoulders roll, head counter-sways
  865|       return { roll: s * 0.24 * loco * (0.45 + speed01 * 0.55), spine: 0.11 * loco, headPitch: 0 };
  866|     }
  867|     if (key.startsWith('MOCHI')) {
  868|       return { roll: Math.sin(this.phase * Math.PI * 4) * 0.05 * loco, spine: 0.19 * loco, headPitch: -0.05 * speed01 };
  869|     }
  870|     if (t.locomotion === 'skitter') {
  871|       // ZIK alternating-tripod: low body, quick shimmy, head holds still
  872|       return { roll: s * 0.07 * loco, spine: -0.05 * loco, headPitch: -0.06 * loco };
  873|     }
  874|     if (t.locomotion === 'hop') return { roll: 0, spine: 0, headPitch: 0 };
  875|     return { roll: s * 0.08 * loco, spine: 0.09 * loco, headPitch: 0 };
  876|   }

  878|   /** Additive action poses, all in body-local radians / metres. */
  879|   private _actionPose(id: ActionId, time: number): { pitch: number; lift: number; squash: number; roll: number } {
  880|     const e = Math.min(1, time / 0.22);
  881|     const osc = Math.sin(time * 9);
  882|     const R = { pitch: 0, lift: 0, squash: 1, roll: 0 };
  883|     switch (id) {
  884|       // eat: dip the head, then chomp — the mouth itself is driven by _updateFace
  885|       case 'eat': return { ...R, pitch: e * 0.40 + osc * 0.07, lift: -e * 0.03, squash: 1 + osc * 0.05 };
  886|       case 'chomp': return { ...R, pitch: e * 0.44 + Math.sin(time * 22) * 0.05, lift: -e * 0.03, squash: 1 + Math.sin(time * 22) * 0.055 };
  887|       case 'sniff': return { ...R, pitch: e * 0.46 + Math.sin(time * 7) * 0.07, lift: -e * 0.03, squash: 1 };
  888|       case 'play': return { ...R, pitch: -e * 0.12, lift: Math.abs(Math.sin(time * 5)) * e * 0.10, squash: 1 + Math.sin(time * 5) * 0.10, roll: Math.sin(time * 4) * 0.10 * e };
  889|       case 'pet': return { ...R, pitch: e * 0.14, lift: -e * 0.02, squash: 1 + Math.sin(time * 8) * 0.05 };
  890|       // roll onto the back: a big roll plus a drop, held for the stroke
  891|       case 'petRoll': {
  892|         const u = Math.min(1, time / 0.35);
  893|         return { pitch: -u * 0.55, lift: -u * 0.06, squash: 1 + Math.sin(time * 6) * 0.06, roll: u * 2.3 };
  894|       }
  895|       case 'sleep': return { ...R, pitch: e * 0.34, lift: -e * 0.07, squash: 1 - e * 0.08 };
  896|       case 'greet': return { ...R, pitch: -e * 0.10, lift: Math.abs(Math.sin(time * 7)) * e * 0.09, squash: 1 + Math.sin(time * 7) * 0.05 };
  897|       case 'wave': return { ...R, pitch: -e * 0.06, lift: Math.abs(Math.sin(time * 8)) * e * 0.05, squash: 1 };
  898|       case 'startle': return { ...R, pitch: -e * 0.22, lift: e * 0.07, squash: 1 - e * 0.08 };
  899|       case 'dig': return { ...R, pitch: e * 0.46 + Math.sin(time * 12) * 0.10, lift: -e * 0.04, squash: 1 };
  900|       case 'splash': return { ...R, pitch: e * 0.18, lift: Math.abs(Math.sin(time * 14)) * e * 0.05, squash: 1 };
  901|       case 'swim': return { ...R, pitch: e * 0.06, lift: Math.sin(time * 3) * e * 0.03, squash: 1 };
  902|       case 'fly': return { ...R, pitch: -e * 0.14, lift: Math.sin(time * 6) * e * 0.05, squash: 1 };
  903|       case 'perch': return { ...R, pitch: e * 0.06, squash: 1 };
  904|       case 'carry': return { ...R, pitch: Math.sin(time * 7) * 0.14, squash: 1.12, roll: Math.sin(time * 5) * 0.12 };
  905|       default: return R;
  906|     }
  907|   }
