 1001|       const vd = Math.hypot(vx, vz), lim = 0.85 * flat;
 1002|       if (vd > lim) { o.x = hx + (vx / vd) * lim; o.z = hz + (vz / vd) * lim; }
 1003|       o.y = gy(o.x, o.z);
 1004|       return o;
 1005|     };
 1006| 
 1007|     if (L.mode === 'plant') {
 1008|       // v13: a planted foot the solver cannot hold steps instead of sliding
 1009|       const strained = L.err > 0.01 && L.landT > 0.05;
 1010|       const hard = strained || Math.hypot(L.lock.x - hip.x, L.lock.z - hip.z) > flat || hip.distanceTo(L.lock) > hardR;
 1011|       const rx = L.lock.x - rest.x, rz = L.lock.z - rest.z;
 1012|       const along = rx * fx + rz * fz;
 1013|       const lagging = moving && along < -1.35 * half;
 1014|       const drifted = !moving && Math.hypot(rx, rz) > Math.max(0.3 * flat, 0.05 * reach);
 1015|       let air = 0;
 1016|       for (const o of this.legs) if (o !== L && o.mode === 'swing') air++;
 1017|       const maxAir = this.legs.length >= 6 ? 3 : this.legs.length >= 4 ? 2 : 1;
 1018|       const beat = onBeat && L.landT > 0.2 * cycle;
 1019|       const soft = (lagging || drifted) && air < maxAir && L.landT > 0.08;
 1020|       if (hard || beat || soft) {
 1021|         const ls = this.liftStats;
 1022|         if (strained) ls.strain++; else if (hard) ls.hard++; else if (beat) ls.beat++; else ls.soft++;
 1023|         L.mode = 'swing';
 1024|         L.forced = !beat;
 1025|         L.swingT = 0;
 1026|         L.swingFrom.copy(L.lock);
 1027|         if (beat && !hard) {
 1028|           L.swingDur = THREE.MathUtils.clamp((1 - DUTY) * cycle, 0.12, 0.40);
 1029|         } else {
 1030|           plan(0.2, L.swingTo);
 1031|           const dd = Math.hypot(L.swingTo.x - L.lock.x, L.swingTo.z - L.lock.z);
 1032|           L.swingDur = THREE.MathUtils.clamp(0.14 + 0.35 * dd / reach, 0.14, 0.30);
 1033|         }
 1034|         plan(L.swingDur, L.swingTo);
 1035|       }
 1036|     }
 1037| 
 1038|     if (L.mode === 'swing') {
 1039|       L.swingT = Math.min(1, L.swingT + dt / Math.max(0.05, L.swingDur));
 1040|       plan((1 - L.swingT) * L.swingDur, this._tw);
 1041|       L.swingTo.lerp(this._tw, 1 - Math.exp(-dt / 0.05));
 1042|       const sT = L.swingT;
 1043|       const e = sT * sT * (3 - 2 * sT);
 1044|       out.lerpVectors(L.swingFrom, L.swingTo, e);
 1045|       const dd = Math.hypot(L.swingTo.x - L.swingFrom.x, L.swingTo.z - L.swingFrom.z);
 1046|       const k = reach / Math.max(1e-3, L.L1 + L.L2);
 1047|       const liftH = L.lift * k * Math.min(1, 0.35 + dd / stride);
 1048|       out.y = Math.max(out.y, gy(out.x, out.z)) + Math.sin(sT * Math.PI) * liftH;
 1049|       if (sT >= 1) { L.mode = 'plant'; L.landPending = true; L.landT = 0; L.lock.copy(out); }
 1050|       return;
 1051|     }
 1052|     L.lock.y += (gy(L.lock.x, L.lock.z) - L.lock.y) * Math.min(1, dt * 20);
 1053|     out.copy(L.lock);
 1054|   }

 1065|     while (d < -Math.PI) d += Math.PI * 2;
 1066|     return d;

 1074|   /**
 1075|    * The face. One place writes eyes, lids, brows, mouth and blush, so an
 1076|    * expression can never half-apply.
 1077|    */
 1078|   private _updateFace(dt: number, act: number): void {
 1079|     const h = this.handle;
 1080|     const closed = this.blinkT > 0 ? Math.sin(Math.PI * (1 - this.blinkT)) : 0;
 1081|     const f = this.face;
 1082| 
 1083|     const happy = this.mood === 'happy' ? 1 : 0;
 1084|     const love = this.mood === 'love' ? 1 : 0;
 1085|     const sad = this.mood === 'sad' ? 1 : 0;
 1086|     const surprised = this.mood === 'surprised' || this.mood === 'scared' ? 1 : 0;
 1087|     const sleepy = this.mood === 'sleepy' ? 1 : 0;
 1088|     const dizzy = this.mood === 'dizzy' ? 1 : 0;
 1089| 
 1090|     // target face values, then blend — never a hard set
 1091|     let lidT = f.lid;
 1092|     let openT = f.mouthOpen;
 1093|     let wideT = f.mouthWide;
 1094|     let blushT = f.blush;
 1095|     let pupT = 1;
 1096|     let spinT = 0;
 1097| 
 1098|     if (this.petted) { lidT = 0.72; blushT = 1; pupT = 1.06; wideT = 1.15; openT = Math.max(openT, 0.25); }
 1099|     if (this.held) { lidT = 0; blushT = 0.2; pupT = 1.35; openT = Math.max(openT, 0.75); wideT = 0.8; }
 1100|     if (dizzy) { pupT = 0.7; spinT = 1; }
 1101|     if (sleepy) { lidT = Math.max(lidT, 0.62); }
 1102|     if (happy || love) { lidT = Math.max(lidT, 0.55); blushT = Math.max(blushT, 0.8); pupT = 1.08; openT = Math.max(openT, 0.35); }
 1103|     if (surprised) { lidT = 0; pupT = 1.3; openT = Math.max(openT, 0.85); wideT = 0.85; }
 1104|     if (sad) { lidT = Math.max(lidT, 0.28); pupT = 1.12; wideT = 0.85; }
 1105|     if (this.action === 'chomp' || this.action === 'eat') {
 1106|       const chew = 0.5 + 0.5 * Math.sin(this.actionT * 22);
 1107|       openT = Math.max(openT, chew * 0.95);
 1108|       wideT = 1.1;
 1109|       blushT = Math.max(blushT, 0.55);
 1110|     }
 1111|     if (this.action === 'wave') { lidT = Math.max(lidT, 0.5); openT = Math.max(openT, 0.7); wideT = 1.2; }
 1112|     if (this.idleAct === 'yawn' && this.idleActW > 0.2) { openT = Math.max(openT, 0.95 * this.idleActW); wideT = 0.9; lidT = Math.max(lidT, 0.75 * this.idleActW); }
 1113|     if (this.idleAct === 'sniff' && this.idleActW > 0.2) { wideT = 0.8; }
 1114| 
 1115|     const k = Math.min(1, dt / 0.22);
 1116|     f.lid += (lidT - f.lid) * k;
 1117|     f.mouthOpen += (openT - f.mouthOpen) * k;
 1118|     f.mouthWide += (wideT - f.mouthWide) * k;
 1119|     f.blush += (blushT - f.blush) * k;
 1120|     f.pupil += (pupT - f.pupil) * Math.min(1, dt * 6);
 1121|     f.gazeSpin += (spinT - f.gazeSpin) * Math.min(1, dt * 4);
 1122| 
 1123|     this.pupil += ((1 + surprised * 0.10 - happy * 0.05 + (sleepy + dizzy) * -0.08) - this.pupil) * Math.min(1, dt * 5);
 1124|     this.lidClose += (f.lid - this.lidClose) * Math.min(1, dt * 6);
 1125| 
 1126|     const dizzySpin = f.gazeSpin * this.t * 5.5;
 1127|     for (const e of h.eyes) {
 1128|       // a one-eyed wink, when the blink roll said so
 1129|       const lid = this.blinkSide > 0 ? (e.side > 0 ? closed : 0)
 1130|         : this.blinkSide < 0 ? (e.side < 0 ? closed : 0)
 1131|           : closed;
 1132|       e.lidPivot.rotation.x = -Math.PI * 0.5 + Math.max(0, Math.min(0.985, lid + this.lidClose)) * Math.PI * 0.98;
 1133|       // eyes lead the head: the socket carries the gaze, the head carries the pose
 1134|       const spinY = f.gazeSpin > 0.05 ? Math.cos(dizzySpin) * 0.45 * f.gazeSpin : 0;
 1135|       const spinP = f.gazeSpin > 0.05 ? Math.sin(dizzySpin) * 0.35 * f.gazeSpin : 0;
 1136|       e.socket.rotation.set(this.gazePitch + spinP, this.gazeYaw + spinY, f.gazeSpin * Math.sin(dizzySpin * 0.7) * 0.4);
 1137|       const pup = this.pupil * f.pupil * (1 + surprised * 0.05);
 1138|       e.pupil.scale.set(pup, pup * (1 - sad * 0.12), 1);
 1139|       const browUp = surprised * 1.0 + sad * 0.45;
 1140|       e.brow.visible = browUp > 0.08;
 1141|       if (e.brow.visible) {
 1142|         e.brow.position.y = e.radius * (0.92 + browUp * 0.20);
 1143|         // angry/sad tilt: inner end up for sad, down for angry
 1144|         e.brow.rotation.z = (sad * -0.30 + (this.mood === 'scared' ? 0.25 : 0)) * (e.side >= 0 ? 1 : -1);
 1145|       }
 1146|     }
 1147| 
 1148|     const open = f.mouthOpen;
 1149|     h.mouthOpen.visible = open > 0.06;
 1150|     h.mouth.visible = open <= 0.06;
 1151|     if (h.mouthOpen.visible) {
 1152|       const w = f.mouthWide;
 1153|       // cheeks puff 20% on a chomp: the open mouth and the blush both swell
 1154|       h.mouthOpen.scale.set(w * (1 + open * 0.15), (0.6 + open * 0.75) * (1 + open * 0.12), 1 * (1 + open * 0.2));
 1155|     }
 1156|     if (h.mouth.visible) {
 1157|       h.mouth.scale.set(f.mouthWide, 1 - (1 - f.mouthWide) * 0.5 + (sad ? -0.25 : 0), 1);
 1158|       h.mouth.rotation.z = -Math.PI * 0.5 - 1.55 * 0.5 + (sad ? 0.5 : 0);
 1159|     }
 1160|     for (const b of h.blush) b.scale.setScalar(0.85 + f.blush * 0.55 + open * 0.08);
 1161|     void act;
 1162|   }

 1164|   /** Per-species locomotion flavour, keyed off the template NAME: PIP and
 1165|    *  MOCHI share the `walk` tag but must move completely differently. */
 1166|   private _gaitFlavour(loco: number, speed01: number): { roll: number; spine: number; headPitch: number } {
 1167|     const t = this.template;
 1168|     if (loco <= 0.001) return { roll: 0, spine: 0, headPitch: 0 };
 1169|     const s = Math.sin(this.phase * Math.PI * 2);
 1170|     const key = (t.name || '').toUpperCase();
 1171|     if (key.startsWith('PIP') || t.locomotion === 'walk') {
 1172|       // PIP waddle: hard side-to-side sway, shoulders roll, head counter-sways
 1173|       return { roll: s * 0.24 * loco * (0.45 + speed01 * 0.55), spine: 0.11 * loco, headPitch: 0 };
 1174|     }
 1175|     if (key.startsWith('MOCHI')) {
 1176|       return { roll: Math.sin(this.phase * Math.PI * 4) * 0.05 * loco, spine: 0.19 * loco, headPitch: -0.05 * speed01 };
 1177|     }
 1178|     if (t.locomotion === 'skitter') {
 1179|       // ZIK alternating-tripod: low body, quick shimmy, head holds still
 1180|       return { roll: s * 0.07 * loco, spine: -0.05 * loco, headPitch: -0.06 * loco };
 1181|     }
 1182|     if (t.locomotion === 'hop') return { roll: 0, spine: 0, headPitch: 0 };
 1183|     return { roll: s * 0.08 * loco, spine: 0.09 * loco, headPitch: 0 };
 1184|   }

 1186|   /** Additive action poses, all in body-local radians / metres. */
 1187|   private _actionPose(id: ActionId, time: number): { pitch: number; lift: number; squash: number; roll: number } {
 1188|     const e = Math.min(1, time / 0.22);
 1189|     const osc = Math.sin(time * 9);
 1190|     const R = { pitch: 0, lift: 0, squash: 1, roll: 0 };
 1191|     switch (id) {
 1192|       // eat: dip the head, then chomp — the mouth itself is driven by _updateFace
 1193|       case 'eat': return { ...R, pitch: e * 0.40 + osc * 0.07, lift: -e * 0.03, squash: 1 + osc * 0.05 };
 1194|       case 'chomp': return { ...R, pitch: e * 0.44 + Math.sin(time * 22) * 0.05, lift: -e * 0.03, squash: 1 + Math.sin(time * 22) * 0.055 };
 1195|       case 'sniff': return { ...R, pitch: e * 0.46 + Math.sin(time * 7) * 0.07, lift: -e * 0.03, squash: 1 };
 1196|       case 'play': return { ...R, pitch: -e * 0.12, lift: Math.abs(Math.sin(time * 5)) * e * 0.10, squash: 1 + Math.sin(time * 5) * 0.10, roll: Math.sin(time * 4) * 0.10 * e };
 1197|       case 'pet': return { ...R, pitch: e * 0.14, lift: -e * 0.02, squash: 1 + Math.sin(time * 8) * 0.05 };
 1198|       // roll onto the back: a big roll plus a drop, held for the stroke
 1199|       case 'petRoll': {
 1200|         const u = Math.min(1, time / 0.35);
 1201|         return { pitch: -u * 0.55, lift: -u * 0.06, squash: 1 + Math.sin(time * 6) * 0.06, roll: u * 2.3 };
 1202|       }
 1203|       case 'sleep': return { ...R, pitch: e * 0.34, lift: -e * 0.07, squash: 1 - e * 0.08 };
 1204|       case 'greet': return { ...R, pitch: -e * 0.10, lift: Math.abs(Math.sin(time * 7)) * e * 0.09, squash: 1 + Math.sin(time * 7) * 0.05 };
 1205|       case 'wave': return { ...R, pitch: -e * 0.06, lift: Math.abs(Math.sin(time * 8)) * e * 0.05, squash: 1 };
 1206|       case 'startle': return { ...R, pitch: -e * 0.22, lift: e * 0.07, squash: 1 - e * 0.08 };
 1207|       case 'dig': return { ...R, pitch: e * 0.46 + Math.sin(time * 12) * 0.10, lift: -e * 0.04, squash: 1 };
 1208|       case 'splash': return { ...R, pitch: e * 0.18, lift: Math.abs(Math.sin(time * 14)) * e * 0.05, squash: 1 };
 1209|       case 'swim': return { ...R, pitch: e * 0.06, lift: Math.sin(time * 3) * e * 0.03, squash: 1 };
 1210|       case 'fly': return { ...R, pitch: -e * 0.14, lift: Math.sin(time * 6) * e * 0.05, squash: 1 };
 1211|       case 'perch': return { ...R, pitch: e * 0.06, squash: 1 };
 1212|       case 'carry': return { ...R, pitch: Math.sin(time * 7) * 0.14, squash: 1.12, roll: Math.sin(time * 5) * 0.12 };
 1213|       default: return R;
 1214|     }
 1215|   }
