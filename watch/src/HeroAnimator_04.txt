 1083|   /** Per-species locomotion flavour, keyed off the template NAME: PIP and
 1084|    *  MOCHI share the `walk` tag but must move completely differently. */
 1085|   private _gaitFlavour(loco: number, speed01: number): { roll: number; spine: number; headPitch: number } {
 1086|     const t = this.template;
 1087|     if (loco <= 0.001) return { roll: 0, spine: 0, headPitch: 0 };
 1088|     const s = Math.sin(this.phase * Math.PI * 2);
 1089|     const key = (t.name || '').toUpperCase();
 1090|     if (key.startsWith('PIP') || t.locomotion === 'walk') {
 1091|       // PIP waddle: hard side-to-side sway, shoulders roll, head counter-sways
 1092|       return { roll: s * 0.24 * loco * (0.45 + speed01 * 0.55), spine: 0.11 * loco, headPitch: 0 };
 1093|     }
 1094|     if (key.startsWith('MOCHI')) {
 1095|       return { roll: Math.sin(this.phase * Math.PI * 4) * 0.05 * loco, spine: 0.19 * loco, headPitch: -0.05 * speed01 };
 1096|     }
 1097|     if (t.locomotion === 'skitter') {
 1098|       // ZIK alternating-tripod: low body, quick shimmy, head holds still
 1099|       return { roll: s * 0.07 * loco, spine: -0.05 * loco, headPitch: -0.06 * loco };
 1100|     }
 1101|     if (t.locomotion === 'hop') return { roll: 0, spine: 0, headPitch: 0 };
 1102|     return { roll: s * 0.08 * loco, spine: 0.09 * loco, headPitch: 0 };
 1103|   }

 1105|   /** Additive action poses, all in body-local radians / metres. */
 1106|   private _actionPose(id: ActionId, time: number): { pitch: number; lift: number; squash: number; roll: number } {
 1107|     const e = Math.min(1, time / 0.22);
 1108|     const osc = Math.sin(time * 9);
 1109|     const R = { pitch: 0, lift: 0, squash: 1, roll: 0 };
 1110|     switch (id) {
 1111|       // eat: dip the head, then chomp — the mouth itself is driven by _updateFace
 1112|       case 'eat': return { ...R, pitch: e * 0.40 + osc * 0.07, lift: -e * 0.03, squash: 1 + osc * 0.05 };
 1113|       case 'chomp': return { ...R, pitch: e * 0.44 + Math.sin(time * 22) * 0.05, lift: -e * 0.03, squash: 1 + Math.sin(time * 22) * 0.055 };
 1114|       case 'sniff': return { ...R, pitch: e * 0.46 + Math.sin(time * 7) * 0.07, lift: -e * 0.03, squash: 1 };
 1115|       case 'play': return { ...R, pitch: -e * 0.12, lift: Math.abs(Math.sin(time * 5)) * e * 0.10, squash: 1 + Math.sin(time * 5) * 0.10, roll: Math.sin(time * 4) * 0.10 * e };
 1116|       case 'pet': return { ...R, pitch: e * 0.14, lift: -e * 0.02, squash: 1 + Math.sin(time * 8) * 0.05 };
 1117|       // roll onto the back: a big roll plus a drop, held for the stroke
 1118|       case 'petRoll': {
 1119|         const u = Math.min(1, time / 0.35);
 1120|         return { pitch: -u * 0.55, lift: -u * 0.06, squash: 1 + Math.sin(time * 6) * 0.06, roll: u * 2.3 };
 1121|       }
 1122|       case 'sleep': return { ...R, pitch: e * 0.34, lift: -e * 0.07, squash: 1 - e * 0.08 };
 1123|       case 'greet': return { ...R, pitch: -e * 0.10, lift: Math.abs(Math.sin(time * 7)) * e * 0.09, squash: 1 + Math.sin(time * 7) * 0.05 };
 1124|       case 'wave': return { ...R, pitch: -e * 0.06, lift: Math.abs(Math.sin(time * 8)) * e * 0.05, squash: 1 };
 1125|       case 'startle': return { ...R, pitch: -e * 0.22, lift: e * 0.07, squash: 1 - e * 0.08 };
 1126|       case 'dig': return { ...R, pitch: e * 0.46 + Math.sin(time * 12) * 0.10, lift: -e * 0.04, squash: 1 };
 1127|       case 'splash': return { ...R, pitch: e * 0.18, lift: Math.abs(Math.sin(time * 14)) * e * 0.05, squash: 1 };

 1129|       case 'fly': return { ...R, pitch: -e * 0.14, lift: Math.sin(time * 6) * e * 0.05, squash: 1 };
 1130|       case 'perch': return { ...R, pitch: e * 0.06, squash: 1 };
 1131|       case 'carry': return { ...R, pitch: Math.sin(time * 7) * 0.14, squash: 1.12, roll: Math.sin(time * 5) * 0.12 };
