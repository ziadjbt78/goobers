 1002|     while (d < -Math.PI) d += Math.PI * 2;
 1003|     return d;
 1004|   }

 1011|   /**
 1012|    * The face. One place writes eyes, lids, brows, mouth and blush, so an
 1013|    * expression can never half-apply.
 1014|    */
 1015|   private _updateFace(dt: number, act: number): void {
 1016|     const h = this.handle;
 1017|     const closed = this.blinkT > 0 ? Math.sin(Math.PI * (1 - this.blinkT)) : 0;
 1018|     const f = this.face;
 1019| 
 1020|     const happy = this.mood === 'happy' ? 1 : 0;
 1021|     const love = this.mood === 'love' ? 1 : 0;
 1022|     const sad = this.mood === 'sad' ? 1 : 0;
 1023|     const surprised = this.mood === 'surprised' || this.mood === 'scared' ? 1 : 0;
 1024|     const sleepy = this.mood === 'sleepy' ? 1 : 0;
 1025|     const dizzy = this.mood === 'dizzy' ? 1 : 0;
 1026| 
 1027|     // target face values, then blend — never a hard set
 1028|     let lidT = f.lid;
 1029|     let openT = f.mouthOpen;
 1030|     let wideT = f.mouthWide;
 1031|     let blushT = f.blush;
 1032|     let pupT = 1;
 1033|     let spinT = 0;
 1034| 
 1035|     if (this.petted) { lidT = 0.72; blushT = 1; pupT = 1.06; wideT = 1.15; openT = Math.max(openT, 0.25); }
 1036|     if (this.held) { lidT = 0; blushT = 0.2; pupT = 1.35; openT = Math.max(openT, 0.75); wideT = 0.8; }
 1037|     if (dizzy) { pupT = 0.7; spinT = 1; }
 1038|     if (sleepy) { lidT = Math.max(lidT, 0.62); }
 1039|     if (happy || love) { lidT = Math.max(lidT, 0.55); blushT = Math.max(blushT, 0.8); pupT = 1.08; openT = Math.max(openT, 0.35); }
 1040|     if (surprised) { lidT = 0; pupT = 1.3; openT = Math.max(openT, 0.85); wideT = 0.85; }
 1041|     if (sad) { lidT = Math.max(lidT, 0.28); pupT = 1.12; wideT = 0.85; }
 1042|     if (this.action === 'chomp' || this.action === 'eat') {
 1043|       const chew = 0.5 + 0.5 * Math.sin(this.actionT * 22);
 1044|       openT = Math.max(openT, chew * 0.95);
 1045|       wideT = 1.1;
 1046|       blushT = Math.max(blushT, 0.55);
 1047|     }
 1048|     if (this.action === 'wave') { lidT = Math.max(lidT, 0.5); openT = Math.max(openT, 0.7); wideT = 1.2; }
 1049|     if (this.idleAct === 'yawn' && this.idleActW > 0.2) { openT = Math.max(openT, 0.95 * this.idleActW); wideT = 0.9; lidT = Math.max(lidT, 0.75 * this.idleActW); }
 1050|     if (this.idleAct === 'sniff' && this.idleActW > 0.2) { wideT = 0.8; }
 1051| 
 1052|     const k = Math.min(1, dt / 0.22);
 1053|     f.lid += (lidT - f.lid) * k;
 1054|     f.mouthOpen += (openT - f.mouthOpen) * k;
 1055|     f.mouthWide += (wideT - f.mouthWide) * k;
 1056|     f.blush += (blushT - f.blush) * k;
 1057|     f.pupil += (pupT - f.pupil) * Math.min(1, dt * 6);
 1058|     f.gazeSpin += (spinT - f.gazeSpin) * Math.min(1, dt * 4);
 1059| 
 1060|     this.pupil += ((1 + surprised * 0.10 - happy * 0.05 + (sleepy + dizzy) * -0.08) - this.pupil) * Math.min(1, dt * 5);
 1061|     this.lidClose += (f.lid - this.lidClose) * Math.min(1, dt * 6);
 1062| 
 1063|     const dizzySpin = f.gazeSpin * this.t * 5.5;
 1064|     for (const e of h.eyes) {
 1065|       // a one-eyed wink, when the blink roll said so
 1066|       const lid = this.blinkSide > 0 ? (e.side > 0 ? closed : 0)
 1067|         : this.blinkSide < 0 ? (e.side < 0 ? closed : 0)
 1068|           : closed;
 1069|       e.lidPivot.rotation.x = -Math.PI * 0.5 + Math.max(0, Math.min(0.985, lid + this.lidClose)) * Math.PI * 0.98;
 1070|       // eyes lead the head: the socket carries the gaze, the head carries the pose
 1071|       const spinY = f.gazeSpin > 0.05 ? Math.cos(dizzySpin) * 0.45 * f.gazeSpin : 0;
 1072|       const spinP = f.gazeSpin > 0.05 ? Math.sin(dizzySpin) * 0.35 * f.gazeSpin : 0;
 1073|       e.socket.rotation.set(this.gazePitch + spinP, this.gazeYaw + spinY, f.gazeSpin * Math.sin(dizzySpin * 0.7) * 0.4);
 1074|       const pup = this.pupil * f.pupil * (1 + surprised * 0.05);
 1075|       e.pupil.scale.set(pup, pup * (1 - sad * 0.12), 1);
 1076|       const browUp = surprised * 1.0 + sad * 0.45;
 1077|       e.brow.visible = browUp > 0.08;
 1078|       if (e.brow.visible) {
 1079|         e.brow.position.y = e.radius * (0.92 + browUp * 0.20);
 1080|         // angry/sad tilt: inner end up for sad, down for angry
 1081|         e.brow.rotation.z = (sad * -0.30 + (this.mood === 'scared' ? 0.25 : 0)) * (e.side >= 0 ? 1 : -1);
 1082|       }
 1083|     }
 1084| 
 1085|     const open = f.mouthOpen;
 1086|     h.mouthOpen.visible = open > 0.06;
 1087|     h.mouth.visible = open <= 0.06;
 1088|     if (h.mouthOpen.visible) {
 1089|       const w = f.mouthWide;
 1090|       // cheeks puff 20% on a chomp: the open mouth and the blush both swell
 1091|       h.mouthOpen.scale.set(w * (1 + open * 0.15), (0.6 + open * 0.75) * (1 + open * 0.12), 1 * (1 + open * 0.2));
 1092|     }
 1093|     if (h.mouth.visible) {
 1094|       h.mouth.scale.set(f.mouthWide, 1 - (1 - f.mouthWide) * 0.5 + (sad ? -0.25 : 0), 1);
 1095|       h.mouth.rotation.z = -Math.PI * 0.5 - 1.55 * 0.5 + (sad ? 0.5 : 0);
 1096|     }
 1097|     for (const b of h.blush) b.scale.setScalar(0.85 + f.blush * 0.55 + open * 0.08);
 1098|     void act;
 1099|   }

 1101|   /** Per-species locomotion flavour, keyed off the template NAME: PIP and
 1102|    *  MOCHI share the `walk` tag but must move completely differently. */
 1103|   private _gaitFlavour(loco: number, speed01: number): { roll: number; spine: number; headPitch: number } {
 1104|     const t = this.template;
 1105|     if (loco <= 0.001) return { roll: 0, spine: 0, headPitch: 0 };
 1106|     const s = Math.sin(this.phase * Math.PI * 2);
 1107|     const key = (t.name || '').toUpperCase();
 1108|     if (key.startsWith('PIP') || t.locomotion === 'walk') {
 1109|       // PIP waddle: hard side-to-side sway, shoulders roll, head counter-sways
 1110|       return { roll: s * 0.24 * loco * (0.45 + speed01 * 0.55), spine: 0.11 * loco, headPitch: 0 };
 1111|     }
 1112|     if (key.startsWith('MOCHI')) {
 1113|       return { roll: Math.sin(this.phase * Math.PI * 4) * 0.05 * loco, spine: 0.19 * loco, headPitch: -0.05 * speed01 };
 1114|     }
 1115|     if (t.locomotion === 'skitter') {
 1116|       // ZIK alternating-tripod: low body, quick shimmy, head holds still
 1117|       return { roll: s * 0.07 * loco, spine: -0.05 * loco, headPitch: -0.06 * loco };
 1118|     }
 1119|     if (t.locomotion === 'hop') return { roll: 0, spine: 0, headPitch: 0 };
 1120|     return { roll: s * 0.08 * loco, spine: 0.09 * loco, headPitch: 0 };
 1121|   }

 1123|   /** Additive action poses, all in body-local radians / metres. */
 1124|   private _actionPose(id: ActionId, time: number): { pitch: number; lift: number; squash: number; roll: number } {
 1125|     const e = Math.min(1, time / 0.22);
 1126|     const osc = Math.sin(time * 9);
 1127|     const R = { pitch: 0, lift: 0, squash: 1, roll: 0 };
 1128|     switch (id) {
 1129|       // eat: dip the head, then chomp — the mouth itself is driven by _updateFace
 1130|       case 'eat': return { ...R, pitch: e * 0.40 + osc * 0.07, lift: -e * 0.03, squash: 1 + osc * 0.05 };
 1131|       case 'chomp': return { ...R, pitch: e * 0.44 + Math.sin(time * 22) * 0.05, lift: -e * 0.03, squash: 1 + Math.sin(time * 22) * 0.055 };
 1132|       case 'sniff': return { ...R, pitch: e * 0.46 + Math.sin(time * 7) * 0.07, lift: -e * 0.03, squash: 1 };
 1133|       case 'play': return { ...R, pitch: -e * 0.12, lift: Math.abs(Math.sin(time * 5)) * e * 0.10, squash: 1 + Math.sin(time * 5) * 0.10, roll: Math.sin(time * 4) * 0.10 * e };
 1134|       case 'pet': return { ...R, pitch: e * 0.14, lift: -e * 0.02, squash: 1 + Math.sin(time * 8) * 0.05 };
 1135|       // roll onto the back: a big roll plus a drop, held for the stroke
 1136|       case 'petRoll': {
 1137|         const u = Math.min(1, time / 0.35);
 1138|         return { pitch: -u * 0.55, lift: -u * 0.06, squash: 1 + Math.sin(time * 6) * 0.06, roll: u * 2.3 };
 1139|       }
 1140|       case 'sleep': return { ...R, pitch: e * 0.34, lift: -e * 0.07, squash: 1 - e * 0.08 };
 1141|       case 'greet': return { ...R, pitch: -e * 0.10, lift: Math.abs(Math.sin(time * 7)) * e * 0.09, squash: 1 + Math.sin(time * 7) * 0.05 };
 1142|       case 'wave': return { ...R, pitch: -e * 0.06, lift: Math.abs(Math.sin(time * 8)) * e * 0.05, squash: 1 };
 1143|       case 'startle': return { ...R, pitch: -e * 0.22, lift: e * 0.07, squash: 1 - e * 0.08 };
 1144|       case 'dig': return { ...R, pitch: e * 0.46 + Math.sin(time * 12) * 0.10, lift: -e * 0.04, squash: 1 };
 1145|       case 'splash': return { ...R, pitch: e * 0.18, lift: Math.abs(Math.sin(time * 14)) * e * 0.05, squash: 1 };
 1146|       case 'swim': return { ...R, pitch: e * 0.06, lift: Math.sin(time * 3) * e * 0.03, squash: 1 };
 1147|       case 'fly': return { ...R, pitch: -e * 0.14, lift: Math.sin(time * 6) * e * 0.05, squash: 1 };
 1148|       case 'perch': return { ...R, pitch: e * 0.06, squash: 1 };
 1149|       case 'carry': return { ...R, pitch: Math.sin(time * 7) * 0.14, squash: 1.12, roll: Math.sin(time * 5) * 0.12 };
 1150|       default: return R;
 1151|     }
 1152|   }
