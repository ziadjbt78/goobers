 1001|         if (strained) ls.strain++; else if (hard) ls.hard++; else if (beat) ls.beat++; else ls.soft++;
 1002|         L.mode = 'swing';
 1003|         L.forced = !beat;
 1004|         L.swingT = 0;
 1005|         L.swingFrom.copy(L.lock);
 1006|         if (beat && !hard) {
 1007|           L.swingDur = THREE.MathUtils.clamp((1 - DUTY) * cycle, 0.12, 0.40);
 1008|         } else {
 1009|           plan(0.2, L.swingTo);
 1010|           const dd = Math.hypot(L.swingTo.x - L.lock.x, L.swingTo.z - L.lock.z);
 1011|           L.swingDur = THREE.MathUtils.clamp(0.14 + 0.35 * dd / reach, 0.14, 0.30);
 1012|         }
 1013|         plan(L.swingDur, L.swingTo);
 1014|       }
 1015|     }
 1016| 
 1017|     if (L.mode === 'swing') {
 1018|       L.swingT = Math.min(1, L.swingT + dt / Math.max(0.05, L.swingDur));
 1019|       plan((1 - L.swingT) * L.swingDur, this._tw);
 1020|       L.swingTo.lerp(this._tw, 1 - Math.exp(-dt / 0.05));
 1021|       const sT = L.swingT;
 1022|       const e = sT * sT * (3 - 2 * sT);
 1023|       out.lerpVectors(L.swingFrom, L.swingTo, e);
 1024|       const dd = Math.hypot(L.swingTo.x - L.swingFrom.x, L.swingTo.z - L.swingFrom.z);
 1025|       const k = reach / Math.max(1e-3, L.L1 + L.L2);
 1026|       const liftH = L.lift * k * Math.min(1, 0.35 + dd / stride);
 1027|       out.y = Math.max(out.y, gy(out.x, out.z)) + Math.sin(sT * Math.PI) * liftH;
 1028|       if (sT >= 1) { L.mode = 'plant'; L.landPending = true; L.landT = 0; L.lock.copy(out); }
 1029|       return;
 1030|     }
 1031|     L.lock.y += (gy(L.lock.x, L.lock.z) - L.lock.y) * Math.min(1, dt * 20);
 1032|     out.copy(L.lock);
 1033|   }

 1053|   /**
 1054|    * The face. One place writes eyes, lids, brows, mouth and blush, so an
 1055|    * expression can never half-apply.
 1056|    */
 1057|   private _updateFace(dt: number, act: number): void {
 1058|     const h = this.handle;
 1059|     const closed = this.blinkT > 0 ? Math.sin(Math.PI * (1 - this.blinkT)) : 0;
 1060|     const f = this.face;
 1061| 
 1062|     const happy = this.mood === 'happy' ? 1 : 0;
 1063|     const love = this.mood === 'love' ? 1 : 0;
 1064|     const sad = this.mood === 'sad' ? 1 : 0;
 1065|     const surprised = this.mood === 'surprised' || this.mood === 'scared' ? 1 : 0;
 1066|     const sleepy = this.mood === 'sleepy' ? 1 : 0;
 1067|     const dizzy = this.mood === 'dizzy' ? 1 : 0;
 1068| 
 1069|     // target face values, then blend — never a hard set
 1070|     let lidT = f.lid;
 1071|     let openT = f.mouthOpen;
 1072|     let wideT = f.mouthWide;
 1073|     let blushT = f.blush;
 1074|     let pupT = 1;
 1075|     let spinT = 0;
 1076| 
 1077|     if (this.petted) { lidT = 0.72; blushT = 1; pupT = 1.06; wideT = 1.15; openT = Math.max(openT, 0.25); }
 1078|     if (this.held) { lidT = 0; blushT = 0.2; pupT = 1.35; openT = Math.max(openT, 0.75); wideT = 0.8; }
 1079|     if (dizzy) { pupT = 0.7; spinT = 1; }
 1080|     if (sleepy) { lidT = Math.max(lidT, 0.62); }
 1081|     if (happy || love) { lidT = Math.max(lidT, 0.55); blushT = Math.max(blushT, 0.8); pupT = 1.08; openT = Math.max(openT, 0.35); }
 1082|     if (surprised) { lidT = 0; pupT = 1.3; openT = Math.max(openT, 0.85); wideT = 0.85; }
 1083|     if (sad) { lidT = Math.max(lidT, 0.28); pupT = 1.12; wideT = 0.85; }
 1084|     if (this.action === 'chomp' || this.action === 'eat') {
 1085|       const chew = 0.5 + 0.5 * Math.sin(this.actionT * 22);
 1086|       openT = Math.max(openT, chew * 0.95);
 1087|       wideT = 1.1;
 1088|       blushT = Math.max(blushT, 0.55);
 1089|     }
 1090|     if (this.action === 'wave') { lidT = Math.max(lidT, 0.5); openT = Math.max(openT, 0.7); wideT = 1.2; }
 1091|     if (this.idleAct === 'yawn' && this.idleActW > 0.2) { openT = Math.max(openT, 0.95 * this.idleActW); wideT = 0.9; lidT = Math.max(lidT, 0.75 * this.idleActW); }
 1092|     if (this.idleAct === 'sniff' && this.idleActW > 0.2) { wideT = 0.8; }
 1093| 
 1094|     const k = Math.min(1, dt / 0.22);
 1095|     f.lid += (lidT - f.lid) * k;
 1096|     f.mouthOpen += (openT - f.mouthOpen) * k;
 1097|     f.mouthWide += (wideT - f.mouthWide) * k;
 1098|     f.blush += (blushT - f.blush) * k;
 1099|     f.pupil += (pupT - f.pupil) * Math.min(1, dt * 6);
 1100|     f.gazeSpin += (spinT - f.gazeSpin) * Math.min(1, dt * 4);
 1101| 
 1102|     this.pupil += ((1 + surprised * 0.10 - happy * 0.05 + (sleepy + dizzy) * -0.08) - this.pupil) * Math.min(1, dt * 5);
 1103|     this.lidClose += (f.lid - this.lidClose) * Math.min(1, dt * 6);
 1104| 
 1105|     const dizzySpin = f.gazeSpin * this.t * 5.5;
 1106|     for (const e of h.eyes) {
 1107|       // a one-eyed wink, when the blink roll said so
 1108|       const lid = this.blinkSide > 0 ? (e.side > 0 ? closed : 0)
 1109|         : this.blinkSide < 0 ? (e.side < 0 ? closed : 0)
 1110|           : closed;
 1111|       e.lidPivot.rotation.x = -Math.PI * 0.5 + Math.max(0, Math.min(0.985, lid + this.lidClose)) * Math.PI * 0.98;
 1112|       // eyes lead the head: the socket carries the gaze, the head carries the pose
 1113|       const spinY = f.gazeSpin > 0.05 ? Math.cos(dizzySpin) * 0.45 * f.gazeSpin : 0;
 1114|       const spinP = f.gazeSpin > 0.05 ? Math.sin(dizzySpin) * 0.35 * f.gazeSpin : 0;
 1115|       e.socket.rotation.set(this.gazePitch + spinP, this.gazeYaw + spinY, f.gazeSpin * Math.sin(dizzySpin * 0.7) * 0.4);
 1116|       const pup = this.pupil * f.pupil * (1 + surprised * 0.05);
 1117|       e.pupil.scale.set(pup, pup * (1 - sad * 0.12), 1);
 1118|       const browUp = surprised * 1.0 + sad * 0.45;
 1119|       e.brow.visible = browUp > 0.08;
 1120|       if (e.brow.visible) {
 1121|         e.brow.position.y = e.radius * (0.92 + browUp * 0.20);
 1122|         // angry/sad tilt: inner end up for sad, down for angry
 1123|         e.brow.rotation.z = (sad * -0.30 + (this.mood === 'scared' ? 0.25 : 0)) * (e.side >= 0 ? 1 : -1);
 1124|       }
 1125|     }
 1126| 
 1127|     const open = f.mouthOpen;
 1128|     h.mouthOpen.visible = open > 0.06;
 1129|     h.mouth.visible = open <= 0.06;
 1130|     if (h.mouthOpen.visible) {
 1131|       const w = f.mouthWide;
 1132|       // cheeks puff 20% on a chomp: the open mouth and the blush both swell
 1133|       h.mouthOpen.scale.set(w * (1 + open * 0.15), (0.6 + open * 0.75) * (1 + open * 0.12), 1 * (1 + open * 0.2));
 1134|     }
 1135|     if (h.mouth.visible) {
 1136|       h.mouth.scale.set(f.mouthWide, 1 - (1 - f.mouthWide) * 0.5 + (sad ? -0.25 : 0), 1);
 1137|       h.mouth.rotation.z = -Math.PI * 0.5 - 1.55 * 0.5 + (sad ? 0.5 : 0);
 1138|     }
 1139|     for (const b of h.blush) b.scale.setScalar(0.85 + f.blush * 0.55 + open * 0.08);
 1140|     void act;
 1141|   }

 1143|   /** Per-species locomotion flavour, keyed off the template NAME: PIP and
 1144|    *  MOCHI share the `walk` tag but must move completely differently. */
 1145|   private _gaitFlavour(loco: number, speed01: number): { roll: number; spine: number; headPitch: number } {
 1146|     const t = this.template;
 1147|     if (loco <= 0.001) return { roll: 0, spine: 0, headPitch: 0 };
 1148|     const s = Math.sin(this.phase * Math.PI * 2);
 1149|     const key = (t.name || '').toUpperCase();
 1150|     if (key.startsWith('PIP') || t.locomotion === 'walk') {
 1151|       // PIP waddle: hard side-to-side sway, shoulders roll, head counter-sways
 1152|       return { roll: s * 0.24 * loco * (0.45 + speed01 * 0.55), spine: 0.11 * loco, headPitch: 0 };
 1153|     }
 1154|     if (key.startsWith('MOCHI')) {
 1155|       return { roll: Math.sin(this.phase * Math.PI * 4) * 0.05 * loco, spine: 0.19 * loco, headPitch: -0.05 * speed01 };
 1156|     }
 1157|     if (t.locomotion === 'skitter') {
 1158|       // ZIK alternating-tripod: low body, quick shimmy, head holds still
 1159|       return { roll: s * 0.07 * loco, spine: -0.05 * loco, headPitch: -0.06 * loco };
 1160|     }
 1161|     if (t.locomotion === 'hop') return { roll: 0, spine: 0, headPitch: 0 };
 1162|     return { roll: s * 0.08 * loco, spine: 0.09 * loco, headPitch: 0 };
 1163|   }

 1165|   /** Additive action poses, all in body-local radians / metres. */
 1166|   private _actionPose(id: ActionId, time: number): { pitch: number; lift: number; squash: number; roll: number } {
 1167|     const e = Math.min(1, time / 0.22);
 1168|     const osc = Math.sin(time * 9);
 1169|     const R = { pitch: 0, lift: 0, squash: 1, roll: 0 };
 1170|     switch (id) {
 1171|       // eat: dip the head, then chomp — the mouth itself is driven by _updateFace
 1172|       case 'eat': return { ...R, pitch: e * 0.40 + osc * 0.07, lift: -e * 0.03, squash: 1 + osc * 0.05 };
 1173|       case 'chomp': return { ...R, pitch: e * 0.44 + Math.sin(time * 22) * 0.05, lift: -e * 0.03, squash: 1 + Math.sin(time * 22) * 0.055 };
 1174|       case 'sniff': return { ...R, pitch: e * 0.46 + Math.sin(time * 7) * 0.07, lift: -e * 0.03, squash: 1 };
 1175|       case 'play': return { ...R, pitch: -e * 0.12, lift: Math.abs(Math.sin(time * 5)) * e * 0.10, squash: 1 + Math.sin(time * 5) * 0.10, roll: Math.sin(time * 4) * 0.10 * e };
 1176|       case 'pet': return { ...R, pitch: e * 0.14, lift: -e * 0.02, squash: 1 + Math.sin(time * 8) * 0.05 };
 1177|       // roll onto the back: a big roll plus a drop, held for the stroke
 1178|       case 'petRoll': {
 1179|         const u = Math.min(1, time / 0.35);
 1180|         return { pitch: -u * 0.55, lift: -u * 0.06, squash: 1 + Math.sin(time * 6) * 0.06, roll: u * 2.3 };
 1181|       }
 1182|       case 'sleep': return { ...R, pitch: e * 0.34, lift: -e * 0.07, squash: 1 - e * 0.08 };
 1183|       case 'greet': return { ...R, pitch: -e * 0.10, lift: Math.abs(Math.sin(time * 7)) * e * 0.09, squash: 1 + Math.sin(time * 7) * 0.05 };
 1184|       case 'wave': return { ...R, pitch: -e * 0.06, lift: Math.abs(Math.sin(time * 8)) * e * 0.05, squash: 1 };
 1185|       case 'startle': return { ...R, pitch: -e * 0.22, lift: e * 0.07, squash: 1 - e * 0.08 };
 1186|       case 'dig': return { ...R, pitch: e * 0.46 + Math.sin(time * 12) * 0.10, lift: -e * 0.04, squash: 1 };
 1187|       case 'splash': return { ...R, pitch: e * 0.18, lift: Math.abs(Math.sin(time * 14)) * e * 0.05, squash: 1 };
 1188|       case 'swim': return { ...R, pitch: e * 0.06, lift: Math.sin(time * 3) * e * 0.03, squash: 1 };
 1189|       case 'fly': return { ...R, pitch: -e * 0.14, lift: Math.sin(time * 6) * e * 0.05, squash: 1 };
 1190|       case 'perch': return { ...R, pitch: e * 0.06, squash: 1 };
 1191|       case 'carry': return { ...R, pitch: Math.sin(time * 7) * 0.14, squash: 1.12, roll: Math.sin(time * 5) * 0.12 };
 1192|       default: return R;
 1193|     }
 1194|   }
