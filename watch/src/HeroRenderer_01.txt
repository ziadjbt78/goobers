  251|   for (let i = 0; i < bones.length; i++) {
  252|     skeleton.boneInverses[i].makeTranslation(boneAt[i].x, boneAt[i].y, boneAt[i].z).invert();
  253|   }
  254| 
  255|   const shell = new THREE.SkinnedMesh(geo, MAT.toon);
  256|   shell.name = 'shell';
  257|   shell.frustumCulled = false;
  258|   shell.bind(skeleton, _id);
  259|   group.add(shell);
  260| 
  261|   const outline = new THREE.SkinnedMesh(ogeo, MAT.outline);
  262|   outline.name = 'outline';
  263|   outline.frustumCulled = false;
  264|   outline.renderOrder = -1;
  265|   outline.bind(skeleton, _id);
  266|   group.add(outline);
  267| 
  268|   // ---- face ---------------------------------------------------------------
  269|   const headBone = bones[t.headBone];
  270|   const eyeWhite = hexToLinear('#fdfcf8');
  271|   const pupilCol: [number, number, number] = [0.012, 0.012, 0.020];
  272|   const skinLinear = hexToLinear(palette.base);
  273|   const lidLinear = mixRgb(skinLinear, [0, 0, 0], 0.10);
  274|   const browLinear = mixRgb(skinLinear, deep, 0.85);
  275| 
  276|   const eyes: EyeHandles[] = [];
  277|   for (const e of t.eyes) {
  278|     const r = e.radius;
  279|     const root = new THREE.Group();
  280|     root.position.copy(e.anchor).sub(headAt);
  281|     root.quaternion.setFromUnitVectors(new THREE.Vector3(0, 0, 1), e.dir.clone().normalize());
  282|     headBone.add(root);
  283| 
  284|     const socket = new THREE.Group();
  285|     root.add(socket);
  286| 
  287|     const scleraGeo = sphere(r, r, r * 0.62, 40, 26);
  288|     paint(scleraGeo, eyeWhite);
  289|     const sclera = new THREE.Mesh(scleraGeo, MAT.glossy);
  290|     sclera.frustumCulled = false;
  291|     socket.add(sclera);
  292| 
  293|     const irisGeo = sphere(r * 0.75, r * 0.75, r * 0.34, 34, 22);
  294|     paint(irisGeo, mixRgb(accent, [0, 0, 0], 0.25));
  295|     const iris = new THREE.Mesh(irisGeo, MAT.glossy);
  296|     iris.position.z = r * 0.50;
  297|     iris.frustumCulled = false;
  298|     socket.add(iris);
  299| 
  300|     const pupilGeo = sphere(r * 0.4125, r * 0.4125, r * 0.32, 28, 18);
  301|     paint(pupilGeo, pupilCol);
  302|     const pupil = new THREE.Mesh(pupilGeo, MAT.glossy);
  303|     pupil.position.z = r * 0.66;
  304|     pupil.frustumCulled = false;
  305|     socket.add(pupil);
  306| 
  307|     // fixed glints: parented to the eye ROOT, so gaze never drags them around
  308|     const bigGeo = sphere(r * 0.30, r * 0.30, r * 0.24, 18, 12);
  309|     paint(bigGeo, [1, 1, 1]);
  310|     const big = new THREE.Mesh(bigGeo, MAT.flat);
  311|     big.position.set(-r * 0.30, r * 0.30, r * 0.92);
  312|     big.frustumCulled = false;
  313|     root.add(big);
  314| 
  315|     const smallGeo = sphere(r * 0.14, r * 0.14, r * 0.12, 14, 10);
  316|     paint(smallGeo, [1, 1, 1]);
  317|     const small = new THREE.Mesh(smallGeo, MAT.flat);
  318|     small.position.set(r * 0.30, -r * 0.26, r * 0.95);
  319|     small.frustumCulled = false;
  320|     root.add(small);
  321| 
  322|     // eyelid: a solid skin dome that swings from behind the eye to over the eye
  323|     const lidPivot = new THREE.Group();
  324|     root.add(lidPivot);
  325|     const lidGeo = sphere(r * 1.07, r * 1.07, r * 1.07, 30, 20);
  326|     paint(lidGeo, lidLinear);
  327|     const lid = new THREE.Mesh(lidGeo, MAT.toon);
  328|     lid.position.set(0, r * 1.03, 0);
  329|     lid.scale.set(0.99, 0.94, 0.99);
  330|     lid.frustumCulled = false;
  331|     lidPivot.add(lid);
  332|     lidPivot.rotation.x = -Math.PI * 0.5;
  333| 
  334|     const browGeo = sphere(r * 0.62, r * 0.16, r * 0.14, 22, 12);
  335|     paint(browGeo, browLinear);
  336|     const brow = new THREE.Mesh(browGeo, MAT.toon);
  337|     brow.position.set(e.side * r * 0.10, r * 0.98, r * 0.52);
  338|     brow.visible = false;
  339|     brow.frustumCulled = false;
  340|     root.add(brow);
  341| 
  342|     // lower lid: the same dome, mirrored below the eye, parked outside the
  343|     // socket until the Face map lifts it
  344|     const lidLowerPivot = new THREE.Group();
  345|     root.add(lidLowerPivot);
  346|     const lowGeo = sphere(r * 1.07, r * 1.07, r * 1.07, 30, 20);
  347|     paint(lowGeo, lidLinear);
  348|     const lidLower = new THREE.Mesh(lowGeo, MAT.toon);
  349|     lidLower.position.set(0, -r * 1.03, 0);
  350|     lidLower.scale.set(0.99, 0.94, 0.99);
  351|     lidLower.frustumCulled = false;
  352|     lidLowerPivot.add(lidLower);
  353|     lidLowerPivot.rotation.x = -Math.PI * 0.5;
  354| 
  355|     // ---- v8 eyeMode symbols: built from primitives, no image files ---------
  356|     const arcColor = mixRgb(pupilCol, [0.05, 0.045, 0.085], 0.45);
  357|     const sym = (geo: THREE.BufferGeometry, col: [number, number, number], z: number): THREE.Mesh => {
  358|       paint(geo, col);
  359|       const m = new THREE.Mesh(geo, MAT.flat);
  360|       m.position.z = z;
  361|       m.visible = false;
  362|       m.frustumCulled = false;
  363|       root.add(m);
  364|       return m;
  365|     };
  366|     const arcHappy = sym(torusArc(r * 0.98, r * 0.17, 2.3, 30, 10), arcColor, r * 0.84);
  367|     arcHappy.rotation.z = Math.PI * 0.5 - 1.15;          // peak up: the "^" happy eye
  368|     const arcClosed = sym(torusArc(r * 1.04, r * 0.055, 1.9, 26, 8), arcColor, r * 0.86);
  369|     arcClosed.rotation.z = Math.PI * 0.5 - 0.95;
  370|     const heart = sym(heartShape(r * 0.62), hexToLinear('#ff5f86'), r * 0.80);
  371|     const star = sym(starShape(r * 0.60), pupilCol, r * 0.82);
  372|     const spiral = sym(spiralShape(r * 0.74), arcColor, r * 0.84);
  373| 
  374|     eyes.push({
  375|       root, socket, sclera, iris, pupil, big, small, lidPivot, lid, brow,
  376|       lidLowerPivot, lidLower, arcHappy, arcClosed, heart, star, spiral,
  377|       radius: r, side: e.side,
  378|     });
  379|   }
  380| 
  381|   // ---- mouth, blush, brows ------------------------------------------------
  382|   const headWorld = t.expression.mouthCenter;
  383|   const darkMouth = mixRgb(deep, [0, 0, 0], 0.35);
  384|   const mw = t.expression.mouthWidth;
  385|   const mouthGeo = torusArc(mw, mw * 0.20, 1.55, 26, 10);
  386|   paint(mouthGeo, darkMouth);
  387|   const mouth = new THREE.Mesh(mouthGeo, MAT.toon);
  388|   mouth.position.copy(headWorld).sub(headAt);
  389|   mouth.rotation.z = -Math.PI * 0.5 - 1.55 * 0.5;
  390|   mouth.frustumCulled = false;
  391|   headBone.add(mouth);
  392| 
  393|   const openGeo = sphere(mw * 0.42, mw * 0.60, mw * 0.22, 22, 14);
  394|   paint(openGeo, darkMouth);
  395|   const mouthOpen = new THREE.Mesh(openGeo, MAT.toon);
  396|   mouthOpen.position.copy(headWorld).add(new THREE.Vector3(0, -mw * 0.18, 0.008)).sub(headAt);
  397|   mouthOpen.visible = false;
  398|   mouthOpen.frustumCulled = false;
  399|   headBone.add(mouthOpen);
  400| 
  401|   const blushCol = mixRgb(hexToLinear('#ff7a95'), skinLinear, 0.42);
  402|   const blush: THREE.Mesh[] = [];
  403|   const blushRest: THREE.Vector3[] = [];
  404|   for (const b of t.expression.blush) {
  405|     const g = sphere(0.062, 0.040, 0.020, 24, 14);
  406|     paint(g, blushCol);
  407|     const m = new THREE.Mesh(g, MAT.toon);
  408|     m.position.copy(b).sub(headAt);
  409|     blushRest.push(m.position.clone());
  410|     m.rotation.y = Math.sign(b.x) * 0.42;
  411|     m.frustumCulled = false;
  412|     headBone.add(m);
  413|     blush.push(m);
  414|   }
  415| 
  416|   return {
  417|     group, pivot, bones, boneByName, boneIndex, skeleton, shell, outline, eyes, mouth, mouthOpen, blush, blushRest,
  418|     parts: t.parts,
  419|     height: t.height,
  420|     radius: t.radius,
  421|     verts: vTotal,
  422|     tris: indices.length / 3,
  423|     colorAt: (name) => colorMap.get(name) ?? [0.5, 0.5, 0.5],
  424|     dispose(): void {
  425|       geo.dispose();
  426|       ogeo.dispose();
  427|       mouth.geometry.dispose();
  428|       mouthOpen.geometry.dispose();
  429|       for (const m of blush) m.geometry.dispose();
  430|       for (const e of eyes) {
  431|         e.sclera.geometry.dispose(); e.iris.geometry.dispose(); e.pupil.geometry.dispose();
  432|         e.big.geometry.dispose(); e.small.geometry.dispose(); e.lid.geometry.dispose(); e.brow.geometry.dispose();
  433|         e.lidLower.geometry.dispose(); e.arcHappy.geometry.dispose(); e.arcClosed.geometry.dispose();
  434|         e.heart.geometry.dispose(); e.star.geometry.dispose(); e.spiral.geometry.dispose();
  435|       }
  436|     },
  437|   };
