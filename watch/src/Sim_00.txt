    1| /**
    2|  * Sim — the playable layer. Owns the creatures, their brains, the player's
    3|  * tools, the visual feedback every tool fires, the floating name tags and the
    4|  * event log the acceptance run reads.
    5|  *
    6|  * Every tool reaction is a small choreography rather than an instant state
    7|  * change: a list of `{ at, fn }` steps that the fixed-step loop walks through,
    8|  * so a feed really does sniff, chomp three times, then squint, and a spawn
    9|  * really does drop a seed, wobble, pop, stretch, blink and hop.
   10|  */

  145|   step(dt: number): void {
  146|     this.simTime += dt;
  147|     this.world.update(dt);
  148|     if (this.dayFrozen) {
  149|       this.world.day = this.frozenDay;
  150|       this.world.stage.setDayNight(this.frozenDay);
  151|     }
  152|     for (const a of this.agents) { a.debugPose = this.testPose; a.noPivotJuice = this.quietPivot; }
  153|     this.runChoreo(dt);
  154|     this.tickDrops(dt);
  155| 
  156|     for (const a of this.agents) {
  157|       const ov = this.overrides.get(a.id) ?? 0;
  158|       if (ov > 0) this.overrides.set(a.id, ov - dt);
  159|       const b = this.brains.get(a.id);
  160|       if (b && this.brainEnabled && ov <= 0) {
  161|         b.update(dt, a, this.world);
  162|         const want = ANIM_ACTION[b.action] ?? 'none';
  163|         if (a.action !== want && a.action !== 'petRoll' && a.action !== 'chomp' && a.action !== 'wave') a.begin(want);
  164|       }
  165|       a.update(dt);
  166|     }
  167| 
  168|     // ---- petting: the motion layer owns the pose, its roll-over at 2 s and
  169|     //      the heart beat; the Sim only decides how long it lasts -------------
  170|     if (this.petting) {
  171|       const p = this.petting;
  172|       this.petHold += dt;
  173|       p.mood = this.petHold > 2 ? 'love' : 'happy';
  174|       p.wiggle = 1;
  175|     }
  176| 
  177|     this.carriedAgent = !!this.carried;
  178|     this.updateEmotes();
  179|     stats.creatures = this.agents.length;
  180| 
  181|     // ---- grass is trampled flat around every creature's feet ---------------
  182|     this.trampleTimer -= dt;
  183|     if (this.trampleTimer <= 0) {
  184|       this.trampleTimer = 0.22;
  185|       this.world.props.trample(this.agents.map((a) => ({ x: a.pos.x, z: a.pos.z })));
  186|     }
  187| 
  188|     for (const a of this.agents) {
  189|       const r = Math.hypot(a.pos.x, a.pos.z);
  190|       if (r > WALK_R) { a.pos.x *= WALK_R / r; a.pos.z *= WALK_R / r; }
  191|     }
  192|   }

  194|   private runChoreo(dt: number): void {
  195|     if (!this.choreo.length) return;
  196|     for (let i = this.choreo.length - 1; i >= 0; i--) {
  197|       const c = this.choreo[i];
  198|       c.t += dt;
  199|       if (c.t >= c.at) { c.fn(); this.choreo.splice(i, 1); }
  200|     }
  201|   }

  212|   private tickDrops(dt: number): void {
  213|     for (let i = this.drops.length - 1; i >= 0; i--) {
  214|       const d = this.drops[i];
  215|       d.life -= dt;
  216|       d.vel.y -= 11 * dt;
  217|       d.mesh.position.addScaledVector(d.vel, dt);
  218|       const g = heightAt(d.mesh.position.x, d.mesh.position.z) + 0.085;
  219|       if (d.mesh.position.y < g) {
  220|         d.mesh.position.y = g;
  221|         if (d.vel.y < -0.4) {
  222|           d.vel.y = -d.vel.y * 0.42;
  223|           d.vel.x *= 0.6; d.vel.z *= 0.6;
  224|           this.vfx.ring(d.mesh.position.x, g, d.mesh.position.z, { r0: 0.06, r1: 0.5, life: 0.35, color: '#ffd0d8' });
  225|         } else { d.vel.set(0, 0, 0); }
  226|       }
  227|       if (d.life <= 0) {
  228|         this.stage.scene.remove(d.mesh);
  229|         this.drops.splice(i, 1);
  230|       }
  231|     }
  232|   }

  234|   private updateEmotes(): void {

  238|         a.emoteAnchor(this.tmp);
  239|         list.push({
  240|           glyph: a.emoteGlyph, pos: this.tmp.clone(),
  241|           life: 1 - Math.min(1, a.emoteT / Math.max(0.1, a.emoteMax)),
  242|           punch: a.emotePunch,
  243|         });
  244|       }
  245|     }
  246|     this.emotes.render(list, this.stage.camera);
  247|   }
