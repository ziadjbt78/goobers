    1| /**
    2|  * Sim — the playable layer. Owns the creatures, their brains, the player's
    3|  * tools, the visual feedback every tool fires, the floating name tags and the
    4|  * event log the acceptance run reads.
    5|  *
    6|  * Every tool reaction is a small choreography rather than an instant state
    7|  * change: a list of `{ at, fn }` steps that the fixed-step loop walks through,
    8|  * so a feed really does sniff, chomp three times, then squint, and a spawn
    9|  * really does drop a seed, wobble, pop, stretch, blink and hop.
   10|  */

  153|   /** v16: the whole island as JSON (species, DNA, name seed, position, needs). */
  154|   saveState(): string {
  155|     const f: SaveFile = { v: 1, build: BUILD_LABEL, seed: this.seed, simTime: this.simTime, day: this.world.day, creatures: [] };
  156|     for (const a of this.agents) {
  157|       const hero = this.heroOf.get(a.id);
  158|       if (!hero) continue;
  159|       const b = this.brains.get(a.id);
  160|       f.creatures.push({
  161|         hero, dna: a.dna, seed: this.seedOf.get(a.id) ?? 0,
  162|         x: a.pos.x, z: a.pos.z, h: a.heading,
  163|         needs: b ? { ...(b.needs as unknown as Record<string, number>) } : undefined,
  164|       });
  165|     }
  166|     return JSON.stringify(f);
  167|   }

  169|   /** v16: replace the island with a saved one. Returns false (and changes nothing) on a bad file. */
  170|   loadState(json: string): boolean {
  171|     let f: SaveFile;
  172|     try { f = JSON.parse(json) as SaveFile; } catch { return false; }
  173|     if (!f || f.v !== 1 || !Array.isArray(f.creatures) || !f.creatures.length) return false;
  174|     this.endInspect();
  175|     for (const a of [...this.agents]) this.remove(a);
  176|     this.choreo.length = 0;
  177|     for (const c of f.creatures) {
  178|       try {
  179|         const a = this.spawn(c.hero, new THREE.Vector3(c.x, heightAt(c.x, c.z), c.z), true, c.dna, c.seed);
  180|         const b = this.brains.get(a.id);
  181|         if (b && c.needs) Object.assign(b.needs as unknown as Record<string, number>, c.needs);
  182|       } catch { /* skip one bad record, keep the rest */ }
  183|     }
  184|     if (typeof f.day === 'number' && !this.dayFrozen) this.world.day = f.day;
  185|     return this.agents.length > 0;
  186|   }

  190|   /** v16 LOD: brains of creatures far from the camera think every 3rd step on the accumulated dt. */
  191|   private brainDt(a: Agent, dt: number): number {
  192|     const far = this.stage.camera.position.distanceTo(a.pos) > 16;
  193|     const acc = (this.brainAcc.get(a.id) ?? 0) + dt;
  194|     if (far && (this.lodTick + a.id) % 3 !== 0) { this.brainAcc.set(a.id, acc); return 0; }
  195|     this.brainAcc.set(a.id, 0);
  196|     return acc;
  197|   }

  203|   /** v9: captures freeze the day clock (0.5 = noon) so lighting never drifts */

  214|   step(dt: number): void {
  215|     this.simTime += dt;
  216|     this.lodTick++;
  217|     this.saveTimer += dt;
  218|     if (this.saveTimer > 15) { this.saveTimer = 0; this.autosave(); }
  219|     this.world.update(dt);
  220|     if (this.dayFrozen) {
  221|       this.world.day = this.frozenDay;
  222|       this.world.stage.setDayNight(this.frozenDay);
  223|     }
  224|     for (const a of this.agents) { a.debugPose = this.testPose; a.noPivotJuice = this.quietPivot; }
  225|     this.runChoreo(dt);
  226|     this.tickDrops(dt);
  227| 
  228|     for (const a of this.agents) {
  229|       const ov = this.overrides.get(a.id) ?? 0;
  230|       if (ov > 0) this.overrides.set(a.id, ov - dt);
  231|       const b = this.brains.get(a.id);
  232|       if (b && this.brainEnabled && ov <= 0) {
  233|         const bdt = this.brainDt(a, dt);
  234|         if (bdt > 0) {
  235|           b.update(bdt, a, this.world);
  236|           const want = ANIM_ACTION[b.action] ?? 'none';
  237|           if (a.action !== want && a.action !== 'petRoll' && a.action !== 'chomp' && a.action !== 'wave') a.begin(want);
  238|         }
  239|       }
  240|       a.update(dt);
  241|     }

  243|     // ---- petting: the motion layer owns the pose, its roll-over at 2 s and
  244|     //      the heart beat; the Sim only decides how long it lasts -------------
  245|     if (this.petting) {
  246|       const p = this.petting;
  247|       this.petHold += dt;
  248|       p.mood = this.petHold > 2 ? 'love' : 'happy';
  249|       p.wiggle = 1;
  250|     }
