  176|   /** v16: replace the island with a saved one. Returns false (and changes nothing) on a bad file. */
  177|   loadState(json: string): boolean {
  178|     let f: SaveFile;
  179|     try { f = JSON.parse(json) as SaveFile; } catch { return false; }
  180|     if (!f || f.v !== 1 || !Array.isArray(f.creatures) || !f.creatures.length) return false;
  181|     this.endInspect();
  182|     for (const a of [...this.agents]) this.remove(a);
  183|     this.choreo.length = 0;
  184|     for (const c of f.creatures) {
  185|       try {
  186|         const a = this.spawn(c.hero, new THREE.Vector3(c.x, heightAt(c.x, c.z), c.z), true, c.dna, c.seed);
  187|         // v17: exact restore (place() re-rolls spots in shallows / past the walk edge)
  188|         a.pos.set(c.x, heightAt(c.x, c.z), c.z);
  189|         a.heading = c.h;
  190|         a.rig.root.position.copy(a.pos);
  191|         a.rig.root.rotation.set(0, c.h, 0);
  192|         const b = this.brains.get(a.id);

  202|   /** v16 LOD: brains of creatures far from the camera think every 3rd step on the accumulated dt. */
  203|   private brainDt(a: Agent, dt: number): number {
  204|     const far = this.stage.camera.position.distanceTo(a.pos) > 16;
  205|     const acc = (this.brainAcc.get(a.id) ?? 0) + dt;
  206|     if (far && (this.lodTick + a.id) % 3 !== 0) { this.brainAcc.set(a.id, acc); return 0; }
  207|     this.brainAcc.set(a.id, 0);
  208|     return acc;
  209|   }
