  439|   /**
  440|    * Call. A ripple spreads on the ground and every creature nearby reacts
  441|    * exactly the way its personality says it should: bold ones sprint, shy ones
  442|    * peek first and creep in, lazy ones wave a limb and stay put.
  443|    */
  444|   call(pos: THREE.Vector3, radius = 10): number {
  445|     this.vfx.ring(pos.x, pos.y, pos.z, { r0: 0.3, r1: radius * 0.75, life: 1.1, color: '#9fe8ff' });
  446|     this.vfx.ring(pos.x, pos.y, pos.z, { r0: 0.2, r1: radius * 0.5, life: 0.8, color: '#ffffff' });
  447|     this.vfx.puff(pos.x, pos.y + 0.1, pos.z, 10, { life: 0.7, size: 0.08, spread: 1.6, up: 1.0, color: '#cfeeff' });
  448|     let n = 0;
  449|     for (const a of this.agents) {
  450|       const d = a.pos.distanceTo(pos);
  451|       if (d >= radius) continue;
  452|       n++;
  453|       const ang = Math.atan2(pos.x - a.pos.x, pos.z - a.pos.z);
  454|       const spot = new THREE.Vector3(pos.x - Math.sin(ang) * 1.0, 0, pos.z - Math.cos(ang) * 1.0);
  455|       spot.y = heightAt(spot.x, spot.z);
  456|       // the ears/antennae twitch and the head snaps to the sound first
  457|       a.faceTarget = { x: pos.x, z: pos.z };
  458|       a.say('bang', 1.1, 1.3);
  459|       a.mood = 'surprised';
  460|       this.hold(a, 3.6);
  461| 
  462|       const bold = a.trait('bold');
  463|       const lazy = a.trait('lazy');
  464|       const shy = 1 - a.trait('sociable');
  465|       if (lazy > 0.62 && bold < 0.55) {
  466|         // lazy: waves a limb, does not budge
  467|         this.after(0.5, () => { if (a.alive) {         a.begin('wave');
  468|         a.say('note', 1.6, 1.2);
  469|       } });
  470|     this.pushEvent('call_wave', a);
  471|         this.after(2.2, () => { if (a.alive) a.mood = 'neutral'; });
  472|         this.pushEvent('call_ignore', a);
  473|         continue;
  474|       }
  475|       if (shy > 0.58 && bold < 0.62) {
  476|         // shy: peek, wait, then creep over at a low throttle
  477|         this.after(1.5, () => {
  478|           if (!a.alive) return;
  479|           a.goTo(spot, 0.45);
  480|           a.mood = 'neutral';
  481|         });
  482|         this.pushEvent('call_peek', a);
  483|         continue;
  484|       }
  485|       // bold: straight there at full tilt, bouncing
  486|       this.after(0.25, () => {
  487|         if (!a.alive) return;
  488|         a.goTo(spot, 1);
  489|         a.mood = 'happy';
  490|         a.cheer();
  491|       });
  492|       this.pushEvent('call', a);
  493|     }
  494|     return n;
  495|   }
