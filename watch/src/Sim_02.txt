  581|   drop(at?: THREE.Vector3): void {
  582|     const a = this.carried;
  583|     if (!a) return;
  584|     a.carried = false;
  585|     a.mood = 'dizzy';
  586|     this.carried = null;
  587| 
  588|     // up in the air with the fling velocity: the motion layer's Dangle carries
  589|     // it down, and its own landDizzy takes over on the landing frame
  590|     if (at) {
  591|       const y = Math.max(at.y, heightAt(at.x, at.z) + 0.6);
  592|       a.motion.dangle.pos.set(at.x, y, at.z);
  593|       a.motion.grabPoint = null;
  594|       a.motion.drop(new THREE.Vector3(0, 0, 0));
  595|     } else {
  596|       const rr = Math.hypot(a.pos.x + this.carryVel.x * 0.35, a.pos.z + this.carryVel.z * 0.35);
  597|       const k = rr > WALK_R ? WALK_R / rr : 1;
  598|       a.motion.drop(new THREE.Vector3(this.carryVel.x * 0.35 * k, 0, this.carryVel.z * 0.35 * k));
  599|     }
  600| 
  601|     // the landing squash, the dust ring and the dizzy stars are the motion
  602|     // layer's own landDizzy action — the Sim keeps only the mood and the
  603|     // shake-off puff at the end
  604|     this.hold(a, 4.4);
  605|     this.after(2.0, () => {
  606|       if (!a.alive) return;
  607|       a.mood = 'neutral';
  608|       this.vfx.puff(a.pos.x, a.pos.y + 0.4, a.pos.z, 8, { life: 0.4, size: 0.05, spread: 1.2, up: 0.9, color: '#ffffff' });
  609|     });
  610|     // a playful creature hops straight back for another go
  611|     if (a.trait('playful') > 0.55) {
  612|       this.after(2.6, () => {
  613|         if (!a.alive) return;
  614|         a.mood = 'happy';
  615|         a.say('note', 1.4, 1.2);
  616|         a.faceTarget = { x: this.stage.camera.position.x, z: this.stage.camera.position.z };
  617|         a.goTo({ x: this.stage.camera.position.x, y: 0, z: this.stage.camera.position.z }, 0.5);
  618|       });
  619|     }
  620|     this.pushEvent('drop', a);
  621|   }

  660|   private updateTags(): void {
  661|     const layer = this.tagLayer;
  662|     if (!layer) return;
  663|     const cam = this.stage.camera;
  664|     const v = new THREE.Vector3();
  665|     const w = this.stage.renderer.domElement.clientWidth;
  666|     const h = this.stage.renderer.domElement.clientHeight;
  667|     let camDist = 0;
  668|     for (const a of this.agents) {
  669|       let el = this.tagEls.get(a.id);
  670|       if (!el) {
  671|         el = document.createElement('div');
  672|         el.className = 'w-tag';
  673|         el.textContent = a.name.split(' ')[0];
  674|         layer.appendChild(el);
  675|         this.tagEls.set(a.id, el);
  676|       }
  677|       v.copy(a.pos);
  678|       v.y += this.templateHeight(a) * 1.34;
  679|       camDist += cam.position.distanceTo(a.pos);
  680|       v.project(cam);
  681|       const behind = v.z > 1;
  682|       const x = (v.x * 0.5 + 0.5) * w;
  683|       const y = (-v.y * 0.5 + 0.5) * h;
  684|       // fade out as the camera closes in — the tag is for reading from far away
  685|       const dist = cam.position.distanceTo(a.pos);
  686|       const op = behind ? 0 : THREE.MathUtils.clamp((dist - 6) / 6, 0, 1) * 0.92;
  687|       el.style.transform = `translate(-50%,-100%) translate(${x.toFixed(1)}px,${y.toFixed(1)}px)`;
  688|       el.style.opacity = op.toFixed(2);
  689|     }
  690|     for (const [id, el] of this.tagEls) {
  691|       if (!this.agents.some((a) => a.id === id)) { el.remove(); this.tagEls.delete(id); }
  692|     }
  693|   }
