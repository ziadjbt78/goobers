  791|       el.style.opacity = op.toFixed(2);

  862|   /**
  863|    * v10 slip probe, rebuilt.
  864|    *
  865|    * The v9 version seeded its "previous foot" buffer from creature-space plant
  866|    * points, which produced ~8 m/frame of nonsense. It now reads the SAME world
  867|    * ankle positions the gate metric reads, and only counts frames where the
  868|    * planting solver says the foot is planted.
  869|    */
  870|   slipProbe(seconds = 8): Record<string, { slip: number; overstretch: number; frames: number; speed: number; freq: number }> {
  871|     const out: Record<string, { slip: number; overstretch: number; frames: number; speed: number; freq: number }> = {};
  872|     const prevBrain = this.brainEnabled;
  873|     this.brainEnabled = false;
  874|     const home = this.agents.map((a) => ({ a, p: a.pos.clone(), h: a.heading }));
  875|     const dt = 1 / 60;
  876|     for (const sp of ['PIP', 'MOCHI', 'BOP', 'ZIK']) {
  877|       const a = this.agents.find((x) => x.template.name.toUpperCase().startsWith(sp));
  878|       if (!a) continue;
  879|       for (const o of this.agents) { o.stop(); o.begin('none'); o.debugPose = 0; }
  880|       a.place(0, -6, 0);
  881|       for (let i = 0; i < 40; i++) { a.goTo({ x: 0, y: 0, z: 40 }, 1); a.update(dt); }
  882|       const prev = a.feet.map((f) => f.pos.clone());
  883|       const prevSw = a.feet.map(() => false);
  884|       const over0 = a.overstretchFrames;
  885|       let worst = 0, speedSum = 0, cycle = 0;
  886|       let lastPhase = a.motion.loco.phase;
  887|       const steps = Math.round(seconds / dt);
  888|       for (let i = 0; i < steps; i++) {
  889|         a.goTo({ x: 0, y: 0, z: 60 }, 1);
  890|         a.update(dt);
  891|         speedSum += a.motion.loco.speed;
  892|         if (a.motion.loco.phase < lastPhase) cycle++;
  893|         lastPhase = a.motion.loco.phase;
  894|         for (let k = 0; k < a.feet.length; k++) {
  895|           const f = a.feet[k];
  896|           if (!prevSw[k] && !f.swinging) {
  897|             const d = f.pos.distanceTo(prev[k]);
  898|             if (d > worst) worst = d;
  899|           }
  900|           prev[k].copy(f.pos);
  901|           prevSw[k] = f.swinging;
  902|         }
  903|       }
  904|       out[sp] = {
  905|         slip: Math.round(worst * 1e4) / 1e4,
  906|         overstretch: a.overstretchFrames - over0,
  907|         frames: steps,
  908|         speed: Math.round((speedSum / steps) * 1e3) / 1e3,
  909|         freq: Math.round((cycle / seconds) * 1e3) / 1e3,
  910|       };
  911|     }
  912|     for (const h of home) h.a.place(h.p.x, h.p.z, h.h);
  913|     this.brainEnabled = prevBrain;
  914|     return out;
  915|   }

  917|   /**
  918|    * v10 sign check. A sign is only called VERIFIED when the magnitude clears
  919|    * 0.02 m AND the two signs give opposite deltas. Anything else is reported as
  920|    * unverified rather than tie-broken.
  921|    */
  922|   signProbe(): Record<string, { pitch: number | null; roll: number | null; limb: number | null; wave: number | null; threshold: number; evidence: Record<string, string> }> {
  923|     type Row = { pitch: number | null; roll: number | null; limb: number | null; wave: number | null; threshold: number; evidence: Record<string, string> };
  924|     const out: Record<string, Row> = {};
  925|     const prevBrain = this.brainEnabled;
  926|     this.brainEnabled = false;
  927|     for (const sp of ['PIP', 'MOCHI', 'BOP', 'ZIK']) {
  928|       const a = this.agents.find((x) => x.template.name.toUpperCase().startsWith(sp));
  929|       if (!a) continue;
  930|       for (const o of this.agents) { o.stop(); o.debugPose = 0; }
  931|       a.place(0, -6, 0);
  932|       const headBone = a.handle.bones[a.template.headBone];
  933|       const bodyBone = a.handle.bones[a.template.bodyBone];
  934|       const tipOf = (b: THREE.Object3D): THREE.Object3D => (b.children.length ? (b.children[0] as THREE.Object3D) : b);
  935|       const tip = a.rig.arms.length ? tipOf(a.rig.arms[0]) : tipOf(a.rig.legs[0] ?? a.rig.body);
  936|       const footBone = a.template.feet.length ? a.handle.bones[a.template.feet[0].chain[2]] : null;
  937|       const height = a.template.height;
  938|       const THRESH = 0.02 * height;
  939|       // everything is measured in the BODY BONE's own frame (its parent space),
  940|       // not world space: a body-pivot rotation then shows up as a real
  941|       // displacement of the head relative to that pivot
  942|       const inBodyFrame = (v: THREE.Vector3): THREE.Vector3 => {
  943|         const parent = bodyBone.parent ?? bodyBone;
  944|         const inv = parent.getWorldQuaternion(new THREE.Quaternion()).invert();
  945|         return v.applyQuaternion(inv);
  946|       };
  947|       const sample = () => {
  948|         a.handle.group.updateMatrixWorld(true);
  949|         return {
  950|           h: inBodyFrame(headBone.getWorldPosition(new THREE.Vector3())),
  951|           f: footBone ? inBodyFrame(footBone.getWorldPosition(new THREE.Vector3())) : new THREE.Vector3(),
  952|           w: inBodyFrame(tip.getWorldPosition(new THREE.Vector3())),
  953|         };
  954|       };
  955|       const delta = (pose: number, ch: 'pitch' | 'roll' | 'limb' | 'wave', sign: number) => {
  956|         a.debugPose = 0;
  957|         (a.rig.axes as unknown as Record<string, number>)[ch] = sign;
  958|         a.update(1 / 60);
  959|         const b0 = sample();
  960|         a.debugPose = pose;
  961|         a.update(1 / 60);
  962|         const b1 = sample();
  963|         return {
  964|           fwd: b1.h.z - b0.h.z,
  965|           lat: b1.h.x - b0.h.x,
  966|           ff: b1.f.z - b0.f.z,
  967|           wy: b1.w.y - b0.w.y,
  968|         };
  969|       };
  970|       const pass = (dp: number, dm: number): number | null => {
  971|         const big = Math.max(Math.abs(dp), Math.abs(dm));
  972|         if (big <= THRESH) return null;
  973|         if ((dp > 0) === (dm > 0)) return null;
  974|         return dp >= dm ? 1 : -1;
  975|       };
  976|       out[sp] = { pitch: null, roll: null, limb: null, wave: null, threshold: Math.round(THRESH * 1e4) / 1e4, evidence: {} };
  977|       const run = (pose: number, ch: 'pitch' | 'roll' | 'limb' | 'wave', key: 'fwd' | 'lat' | 'ff' | 'wy') => {
  978|         const dd = delta(pose, ch, 1);
  979|         const plus = dd[key];
  980|         const minus = delta(pose, ch, -1)[key];
  981|         const chosen = pass(plus, minus);
  982|         out[sp].evidence[ch] = chosen === null
  983|           ? `UNVERIFIED  ${key}: +1 ${plus.toFixed(4)} / -1 ${minus.toFixed(4)} m  (threshold ${THRESH.toFixed(4)})`
  984|           : `verified    ${key}: +1 ${plus.toFixed(4)} / -1 ${minus.toFixed(4)} m -> ${chosen > 0 ? '+1' : '-1'}`;
  985|         if (chosen !== null) (a.rig.axes as unknown as Record<string, number>)[ch] = chosen;
  986|         return chosen;
  987|       };
  988|       out[sp].pitch = run(1, 'pitch', 'fwd');
  989|       out[sp].roll = run(2, 'roll', 'lat');
  990|       out[sp].wave = run(4, 'wave', 'wy');
  991|       // limb is an AXIS choice, so it is probed as one
  992|       {
  993|         const ax0 = a.rig.axes.limb;
  994|         const tryAxis = (ax: 'x' | 'z'): number => {
  995|           a.debugPose = 0; a.rig.axes.limb = ax; a.update(1 / 60);
  996|           const b0 = sample(); a.debugPose = 3; a.update(1 / 60); const b1 = sample();
  997|           return b1.f.z - b0.f.z;
  998|         };
  999|         const fx = tryAxis('x'), fz = tryAxis('z');
 1000|         const ok = Math.max(Math.abs(fx), Math.abs(fz)) > THRESH;
