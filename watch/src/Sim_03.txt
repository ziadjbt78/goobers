  771|     for (const [id, el] of this.tagEls) {

  840|   /**
  841|    * v10 slip probe, rebuilt.
  842|    *
  843|    * The v9 version seeded its "previous foot" buffer from creature-space plant
  844|    * points, which produced ~8 m/frame of nonsense. It now reads the SAME world
  845|    * ankle positions the gate metric reads, and only counts frames where the
  846|    * planting solver says the foot is planted.
  847|    */
  848|   slipProbe(seconds = 8): Record<string, { slip: number; overstretch: number; frames: number; speed: number; freq: number }> {
  849|     const out: Record<string, { slip: number; overstretch: number; frames: number; speed: number; freq: number }> = {};
  850|     const prevBrain = this.brainEnabled;
  851|     this.brainEnabled = false;
  852|     const home = this.agents.map((a) => ({ a, p: a.pos.clone(), h: a.heading }));
  853|     const dt = 1 / 60;
  854|     for (const sp of ['PIP', 'MOCHI', 'BOP', 'ZIK']) {
  855|       const a = this.agents.find((x) => x.template.name.toUpperCase().startsWith(sp));
  856|       if (!a) continue;
  857|       for (const o of this.agents) { o.stop(); o.begin('none'); o.debugPose = 0; }
  858|       a.place(0, -6, 0);
  859|       for (let i = 0; i < 40; i++) { a.goTo({ x: 0, y: 0, z: 40 }, 1); a.update(dt); }
  860|       const prev = a.feet.map((f) => f.pos.clone());
  861|       const prevSw = a.feet.map(() => false);
  862|       const over0 = a.overstretchFrames;
  863|       let worst = 0, speedSum = 0, cycle = 0;
  864|       let lastPhase = a.motion.loco.phase;
  865|       const steps = Math.round(seconds / dt);
  866|       for (let i = 0; i < steps; i++) {
  867|         a.goTo({ x: 0, y: 0, z: 60 }, 1);
  868|         a.update(dt);
  869|         speedSum += a.motion.loco.speed;
  870|         if (a.motion.loco.phase < lastPhase) cycle++;
  871|         lastPhase = a.motion.loco.phase;
  872|         for (let k = 0; k < a.feet.length; k++) {
  873|           const f = a.feet[k];
  874|           if (!prevSw[k] && !f.swinging) {
  875|             const d = f.pos.distanceTo(prev[k]);
  876|             if (d > worst) worst = d;
  877|           }
  878|           prev[k].copy(f.pos);
  879|           prevSw[k] = f.swinging;
  880|         }
  881|       }
  882|       out[sp] = {
  883|         slip: Math.round(worst * 1e4) / 1e4,
  884|         overstretch: a.overstretchFrames - over0,
  885|         frames: steps,
  886|         speed: Math.round((speedSum / steps) * 1e3) / 1e3,
  887|         freq: Math.round((cycle / seconds) * 1e3) / 1e3,
  888|       };
  889|     }
  890|     for (const h of home) h.a.place(h.p.x, h.p.z, h.h);
  891|     this.brainEnabled = prevBrain;
  892|     return out;
  893|   }

  895|   /**
  896|    * v10 sign check. A sign is only called VERIFIED when the magnitude clears
  897|    * 0.02 m AND the two signs give opposite deltas. Anything else is reported as
  898|    * unverified rather than tie-broken.
  899|    */
  900|   signProbe(): Record<string, { pitch: number | null; roll: number | null; limb: number | null; wave: number | null; threshold: number; evidence: Record<string, string> }> {
  901|     type Row = { pitch: number | null; roll: number | null; limb: number | null; wave: number | null; threshold: number; evidence: Record<string, string> };
  902|     const out: Record<string, Row> = {};
  903|     const prevBrain = this.brainEnabled;
  904|     this.brainEnabled = false;
  905|     for (const sp of ['PIP', 'MOCHI', 'BOP', 'ZIK']) {
  906|       const a = this.agents.find((x) => x.template.name.toUpperCase().startsWith(sp));
  907|       if (!a) continue;
  908|       for (const o of this.agents) { o.stop(); o.debugPose = 0; }
  909|       a.place(0, -6, 0);
  910|       const headBone = a.handle.bones[a.template.headBone];
  911|       const bodyBone = a.handle.bones[a.template.bodyBone];
  912|       const tipOf = (b: THREE.Object3D): THREE.Object3D => (b.children.length ? (b.children[0] as THREE.Object3D) : b);
  913|       const tip = a.rig.arms.length ? tipOf(a.rig.arms[0]) : tipOf(a.rig.legs[0] ?? a.rig.body);
  914|       const footBone = a.template.feet.length ? a.handle.bones[a.template.feet[0].chain[2]] : null;
  915|       const height = a.template.height;
  916|       const THRESH = 0.02 * height;
  917|       // everything is measured in the BODY BONE's own frame (its parent space),
  918|       // not world space: a body-pivot rotation then shows up as a real
  919|       // displacement of the head relative to that pivot
  920|       const inBodyFrame = (v: THREE.Vector3): THREE.Vector3 => {
  921|         const parent = bodyBone.parent ?? bodyBone;
  922|         const inv = parent.getWorldQuaternion(new THREE.Quaternion()).invert();
  923|         return v.applyQuaternion(inv);
  924|       };
  925|       const sample = () => {
  926|         a.handle.group.updateMatrixWorld(true);
  927|         return {
  928|           h: inBodyFrame(headBone.getWorldPosition(new THREE.Vector3())),
  929|           f: footBone ? inBodyFrame(footBone.getWorldPosition(new THREE.Vector3())) : new THREE.Vector3(),
  930|           w: inBodyFrame(tip.getWorldPosition(new THREE.Vector3())),
  931|         };
  932|       };
  933|       const delta = (pose: number, ch: 'pitch' | 'roll' | 'limb' | 'wave', sign: number) => {
  934|         a.debugPose = 0;
  935|         (a.rig.axes as unknown as Record<string, number>)[ch] = sign;
  936|         a.update(1 / 60);
  937|         const b0 = sample();
  938|         a.debugPose = pose;
  939|         a.update(1 / 60);
  940|         const b1 = sample();
  941|         return {
  942|           fwd: b1.h.z - b0.h.z,
  943|           lat: b1.h.x - b0.h.x,
  944|           ff: b1.f.z - b0.f.z,
  945|           wy: b1.w.y - b0.w.y,
  946|         };
  947|       };
  948|       const pass = (dp: number, dm: number): number | null => {
  949|         const big = Math.max(Math.abs(dp), Math.abs(dm));
  950|         if (big <= THRESH) return null;
  951|         if ((dp > 0) === (dm > 0)) return null;
  952|         return dp >= dm ? 1 : -1;
  953|       };
  954|       out[sp] = { pitch: null, roll: null, limb: null, wave: null, threshold: Math.round(THRESH * 1e4) / 1e4, evidence: {} };
  955|       const run = (pose: number, ch: 'pitch' | 'roll' | 'limb' | 'wave', key: 'fwd' | 'lat' | 'ff' | 'wy') => {
  956|         const dd = delta(pose, ch, 1);
  957|         const plus = dd[key];
  958|         const minus = delta(pose, ch, -1)[key];
  959|         const chosen = pass(plus, minus);
  960|         out[sp].evidence[ch] = chosen === null
  961|           ? `UNVERIFIED  ${key}: +1 ${plus.toFixed(4)} / -1 ${minus.toFixed(4)} m  (threshold ${THRESH.toFixed(4)})`
  962|           : `verified    ${key}: +1 ${plus.toFixed(4)} / -1 ${minus.toFixed(4)} m -> ${chosen > 0 ? '+1' : '-1'}`;
  963|         if (chosen !== null) (a.rig.axes as unknown as Record<string, number>)[ch] = chosen;
  964|         return chosen;
  965|       };
  966|       out[sp].pitch = run(1, 'pitch', 'fwd');
  967|       out[sp].roll = run(2, 'roll', 'lat');
  968|       out[sp].wave = run(4, 'wave', 'wy');
  969|       // limb is an AXIS choice, so it is probed as one
  970|       {
  971|         const ax0 = a.rig.axes.limb;
  972|         const tryAxis = (ax: 'x' | 'z'): number => {
  973|           a.debugPose = 0; a.rig.axes.limb = ax; a.update(1 / 60);
  974|           const b0 = sample(); a.debugPose = 3; a.update(1 / 60); const b1 = sample();
  975|           return b1.f.z - b0.f.z;
  976|         };
  977|         const fx = tryAxis('x'), fz = tryAxis('z');
  978|         const ok = Math.max(Math.abs(fx), Math.abs(fz)) > THRESH;
  979|         a.rig.axes.limb = ok ? (Math.abs(fx) >= Math.abs(fz) ? 'x' : 'z') : ax0;
  980|         out[sp].evidence.limb = ok
  981|           ? `verified axis '${a.rig.axes.limb}': x ${fx.toFixed(4)} / z ${fz.toFixed(4)} m fwd`
  982|           : `UNVERIFIED axis: x ${fx.toFixed(4)} / z ${fz.toFixed(4)} m fwd (threshold ${THRESH.toFixed(4)})`;
  983|         out[sp].limb = ok ? (a.rig.axes.limb === 'x' ? 1 : 2) : null;
  984|         a.debugPose = 0;
  985|       }
  986|     }
  987|     this.brainEnabled = prevBrain;
  988|     return out;
  989|   }
