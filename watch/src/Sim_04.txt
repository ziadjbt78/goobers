 1103|   /**
 1104|    * Integration step 4 — stride calibration. Drive each species' pedestal
 1105|    * animator with an external gait cycle at 0.6 and measure how far the REAL
 1106|    * ankle bones travel forward in body space over one full cycle. The gait
 1107|    * cycle advances `stride` per cycle, so `stride = range / 0.5`.
 1108|    */
 1109|   strideProbe(): Record<string, {
 1110|     stride: number; range: number; height: number; kind: string; factor: number; legReach: number;
 1111|     stanceFrac: number; stanceDisp: number; strideUserFormula: number; feet: number[];
 1112|   }> {

 1114|       stride: number; range: number; height: number; kind: string; factor: number; legReach: number;

 1119|       base: hexToLinear(pal.base), belly: hexToLinear(pal.belly), limb: hexToLinear(pal.limb),
 1120|       accent: hexToLinear(pal.accent), deep: hexToLinear(pal.deep),
 1121|     };

 1124|       const tpl = buildHeroTemplate(id, colors);

 1126|       const anim = new HeroAnimator(tpl, h);

 1128|       h.group.position.set(0, 0, 0);
 1129|       h.group.rotation.set(0, 0, 0);
 1130|       const n = tpl.feet.length;
 1131|       const lo = new Array<number>(n).fill(1e9);
 1132|       const hi = new Array<number>(n).fill(-1e9);
 1133|       const backSamples = new Array<number>(n).fill(0);
 1134|       const backSpan = new Array<number>(n).fill(0);
 1135|       const backLo = new Array<number>(n).fill(1e9);
 1136|       const backHi = new Array<number>(n).fill(-1e9);
 1137|       const N = 480;
 1138|       const prevZ = new Array<number>(n).fill(0);
 1139|       const havePrev = new Array<boolean>(n).fill(false);
 1140|       for (let i = 0; i < N; i++) {
