 1088|    * Integration step 5 — the BOP hop. A hopper has no legs to calibrate from, so

 1090|    * at walk throttle in a straight line and measure the distance per hop cycle.
 1091|    */
 1092|   hopProbe(seconds = 12, dt = 1 / 60): { hopDistance: number; cycles: number; hopsPerSecond: number; speed: number; stride: number; height: number; airTime: number } | null {
 1093|     const a = this.agents.find((x) => x.template.locomotion === 'hop');
 1094|     if (!a) return null;
 1095|     const loco = a.motion.loco;
 1096|     a.place(0, 3.0, 0);
 1097|     const start = loco.pos.clone();
 1098|     let cycles = 0;
 1099|     let prev = loco.phase;
 1100|     let dist = 0;
 1101|     const px = { x: loco.pos.x, z: loco.pos.z };
 1102|     let airTime = 0;
 1103|     for (let i = 0; i < seconds / dt; i++) {
 1104|       a.goTo({ x: 0, y: 0, z: 40 }, 0.42);
 1105|       a.update(dt);
 1106|       if (loco.phase < prev) cycles++;
 1107|       prev = loco.phase;
 1108|       dist += Math.hypot(loco.pos.x - px.x, loco.pos.z - px.z);
 1109|       px.x = loco.pos.x; px.z = loco.pos.z;
 1110|       // the airborne window of Juice.hopCurve is u in [0.18, 0.78]
 1111|       const u = loco.phase;
 1112|       if (u >= 0.18 && u <= 0.78) airTime += dt;
 1113|     }
 1114|     a.stop();
 1115|     a.place(start.x, start.z, 0);
 1116|     const hopDistance = cycles > 0 ? dist / cycles : 0;
 1117|     return {
 1118|       hopDistance: Math.round(hopDistance * 1e4) / 1e4,
 1119|       cycles,
 1120|       hopsPerSecond: Math.round((cycles / seconds) * 1e3) / 1e3,
 1121|       speed: Math.round(loco.speed * 1e3) / 1e3,
 1122|       stride: loco.cfg.stride,
 1123|       height: a.template.height,
 1124|       airTime: Math.round((airTime / Math.max(1, cycles)) * 1e3) / 1e3,
 1125|     };
 1126|   }
