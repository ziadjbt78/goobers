 1081|   /**
 1082|    * Integration step 4 — stride calibration. Drive each species' pedestal
 1083|    * animator with an external gait cycle at 0.6 and measure how far the REAL
 1084|    * ankle bones travel forward in body space over one full cycle. The gait
 1085|    * cycle advances `stride` per cycle, so `stride = range / 0.5`.
 1086|    */
 1087|   strideProbe(): Record<string, {
 1088|     stride: number; range: number; height: number; kind: string; factor: number; legReach: number;
 1089|     stanceFrac: number; stanceDisp: number; strideUserFormula: number; feet: number[];
 1090|   }> {
 1091|     const out: Record<string, {
 1092|       stride: number; range: number; height: number; kind: string; factor: number; legReach: number;
 1093|       stanceFrac: number; stanceDisp: number; strideUserFormula: number; feet: number[];
 1094|     }> = {};
 1095|     const pal = paletteAt(0);
 1096|     const colors = {
 1097|       base: hexToLinear(pal.base), belly: hexToLinear(pal.belly), limb: hexToLinear(pal.limb),
 1098|       accent: hexToLinear(pal.accent), deep: hexToLinear(pal.deep),
 1099|     };
 1100|     const v = new THREE.Vector3();
 1101|     for (const id of HERO_IDS) {
 1102|       const tpl = buildHeroTemplate(id, colors);
 1103|       const h = buildHero(tpl, pal);
 1104|       const anim = new HeroAnimator(tpl, h);
 1105|       anim.deterministic = true;
 1106|       h.group.position.set(0, 0, 0);
 1107|       h.group.rotation.set(0, 0, 0);
 1108|       const n = tpl.feet.length;
 1109|       const lo = new Array<number>(n).fill(1e9);
 1110|       const hi = new Array<number>(n).fill(-1e9);
 1111|       const backSamples = new Array<number>(n).fill(0);
 1112|       const backSpan = new Array<number>(n).fill(0);
 1113|       const backLo = new Array<number>(n).fill(1e9);
 1114|       const backHi = new Array<number>(n).fill(-1e9);
 1115|       const N = 480;
 1116|       const prevZ = new Array<number>(n).fill(0);
 1117|       const havePrev = new Array<boolean>(n).fill(false);
 1118|       for (let i = 0; i < N; i++) {
 1119|         anim.setGait(0.6, i / N);
 1120|         anim.update(1 / 60, null);
 1121|         h.group.updateMatrixWorld(true);
 1122|         for (let k = 0; k < n; k++) {
 1123|           const bone = h.bones[tpl.feet[k].chain[2]];
 1124|           if (!bone) continue;
 1125|           bone.getWorldPosition(v);
 1126|           h.group.worldToLocal(v);
 1127|           if (v.z < lo[k]) lo[k] = v.z;
 1128|           if (v.z > hi[k]) hi[k] = v.z;
 1129|           // "planted" = the foot is travelling BACKWARD relative to the body:
 1130|           // that is exactly the half of the cycle the in-place gait calls stance.
 1131|           if (havePrev[k] && v.z < prevZ[k]) {
 1132|             backSamples[k]++;
 1133|             if (v.z < backLo[k]) backLo[k] = v.z;
 1134|             if (v.z > backHi[k]) backHi[k] = v.z;
 1135|           }
 1136|           prevZ[k] = v.z;
 1137|           havePrev[k] = true;
 1138|         }
 1139|       }
 1140|       const ranges = lo.map((l, k) => hi[k] - l);
 1141|       const stanceFracs = backSamples.map((c) => c / N);
 1142|       const stanceDisps = backLo.map((l, k) => Math.max(0, backHi[k] - l));
 1143|       const range = ranges.reduce((a, b) => a + b, 0) / Math.max(1, ranges.length);
 1144|       const stanceFrac = stanceFracs.reduce((a, b) => a + b, 0) / Math.max(1, stanceFracs.length);
 1145|       const stanceDisp = stanceDisps.reduce((a, b) => a + b, 0) / Math.max(1, stanceDisps.length);
 1146|       // Body advance per stance = the stance displacement, over a stance that
 1147|       // lasts stanceFrac of the cycle. A cycle holds ONE stance per foot, so
 1148|       // stride = stanceDisp / stanceFrac. (For a duty-0.5 gait that is the
 1149|       // measured range itself, NOT range/0.5.)
 1150|       const stride = stanceDisp / Math.max(0.05, stanceFrac);
 1151|       void backSpan;
 1152|       const f0 = tpl.feet[0];
 1153|       const legReach = f0 ? tpl.bones[f0.chain[1]].pos.length() + tpl.bones[f0.chain[2]].pos.length() : 0;
 1154|       out[id.toUpperCase()] = {
 1155|         stride, range, height: tpl.height, kind: tpl.locomotion,
 1156|         legReach: Math.round(legReach * 1e4) / 1e4,
 1157|         factor: stride / tpl.height,
 1158|         stanceFrac: Math.round(stanceFrac * 1e4) / 1e4,
 1159|         stanceDisp: Math.round(stanceDisp * 1e4) / 1e4,
 1160|         strideUserFormula: Math.round((range / Math.max(0.05, stanceFrac)) * 1e4) / 1e4,
 1161|         feet: ranges.map((r) => Math.round(r * 1e4) / 1e4),
 1162|       };
 1163|       h.dispose();
 1164|     }
 1165|     return out;
 1166|   }

 1168|   /**
 1169|    * Integration step 5 — the BOP hop. A hopper has no legs to calibrate from, so
 1170|    * the stride has to come from the ground it actually covers: drive a real BOP
