    1| /**
    2|  * Pose math. Rotation-only bone work plus the 2-bone leg IK that plants feet.
    3|  *
    4|  * Everything is expressed in creature space and then converted through the
    5|  * parent bone's world matrix, so a limb can never stretch, scale, or translate
    6|  * away from its socket: the socket IS the bone origin.
    7|  *
    8|  * The aim step is child-offset aware: a splayed insect coxa whose child sits
    9|  * sideways off the bone axis still lands its knee exactly on the solved joint.
   10|  *
   11|  * v19 EXACT path (World): the whole solve runs in the hip parent's OWN space,
   12|  * squash included, and every aim is a minimal arc from the bone's rest
   13|  * direction. That removes the world-frame twist (heading-dependent flips on
   14|  * splayed legs, phantom knee rates) and the squash-shear ankle miss. The
   15|  * legacy path (pedestal / Heroes) is untouched.
   16|  */
   17| import * as THREE from 'three';
   18| import { solveIK2, ikLast } from './ik';
   19| 
   20| /** The conventional "down" limb axis, kept for callers that use it as a hint. */
   21| export const LIMB_AXIS = new THREE.Vector3(0, -1, 0);
   22| 
   23| const _q = new THREE.Quaternion();
   24| const _inv = new THREE.Quaternion();
   25| const _d = new THREE.Vector3();
   26| const _u = new THREE.Vector3();
   27| const _v = new THREE.Vector3();
   28| const _root = new THREE.Vector3();
   29| const _tgt = new THREE.Vector3();
   30| const _knee = new THREE.Vector3();
   31| const _pole = new THREE.Vector3();
   32| const _hipW = new THREE.Vector3();
   33| const _pos = new THREE.Vector3();
   34| const _lenA = new THREE.Vector3();
   35| const _lenB = new THREE.Vector3();
   36| 
   37| /**
   38|  * Point a bone so the direction of its CHILD, expressed in the bone's own
   39|  * local space, runs along `dirWorld`.
   40|  */
   41| export function aimBoneTowards(
   42|   bone: THREE.Bone,
   43|   childLocalDir: THREE.Vector3,
   44|   dirWorld: THREE.Vector3,
   45|   parentWorldQuat: THREE.Quaternion,
   46| ): void {
   47|   const u = _u.copy(childLocalDir);
   48|   if (u.lengthSq() < 1e-12) u.set(0, -1, 0);
   49|   u.normalize();
   50|   const d = _d.copy(dirWorld);
   51|   if (d.lengthSq() < 1e-12) d.set(0, -1, 0);
   52|   d.normalize();
   53|   _q.setFromUnitVectors(u, d);
   54|   bone.quaternion.copy(parentWorldQuat).invert().multiply(_q);
   55| }
   56| 
   57| /** Aim a limb whose child really does sit on local -Y. */
   58| export function aimBone(bone: THREE.Bone, dirWorld: THREE.Vector3, parentWorldQuat: THREE.Quaternion): void {
   59|   aimBoneTowards(bone, LIMB_AXIS, dirWorld, parentWorldQuat);
   60| }

   62| export interface LegSolveInput {
   63|   hip: THREE.Bone;
   64|   knee: THREE.Bone;
   65|   ankle: THREE.Bone;
   66|   /** optional: segment lengths are read from the bones, so callers cannot lie */
   67|   l1?: number;
   68|   l2?: number;
   69|   /** desired world position of the ankle */
   70|   targetWorld: THREE.Vector3;
   71|   /** world-space pole hint (roughly forward) */
   72|   poleWorld: THREE.Vector3;
   73|   /** previous knee position in the hip-parent's local space. Read to keep the
   74|    *  solve on the same branch, then written back with the new knee. */
   75|   prevKnee?: THREE.Vector3;
   76|   /** v19: exact parent-space solve (World path). */
   77|   exact?: boolean;
   78|   /** v19: world rotation the foot holds (creature heading). Default: world identity. */
   79|   footWorld?: THREE.Quaternion;
   80| }
   81| 
   82| const _xq = new THREE.Quaternion();
   83| const _xinv = new THREE.Quaternion();
   84| const _xh = new THREE.Quaternion();
   85| const _xt = new THREE.Vector3();
   86| const _xr = new THREE.Vector3();
   87| const _xp = new THREE.Vector3();
   88| const _xk = new THREE.Vector3();
   89| const _xu = new THREE.Vector3();
   90| const _xd = new THREE.Vector3();
   91| 
   92| function solveLegExact(i: LegSolveInput, hipParent: THREE.Object3D): boolean {
   93|   // getWorldQuaternion / worldToLocal refresh the ancestor chain themselves
   94|   const pw = hipParent.getWorldQuaternion(_xq);
   95|   const targetLocal = hipParent.worldToLocal(_xt.copy(i.targetWorld));
   96|   const hipLocal = _xr.copy(i.hip.position);
   97|   const poleLocal = _xp.copy(i.poleWorld).applyQuaternion(_xinv.copy(pw).invert()).normalize();
   98|   const L1 = i.knee.position.length();
   99|   const L2 = i.ankle.position.length();
  100|   solveIK2(hipLocal, targetLocal, L1, L2, poleLocal, _xk, i.prevKnee, true);
  101| 
  102|   // hip: minimal arc from its rest child direction onto the solved knee, in parent space
  103|   _xu.copy(i.knee.position).normalize();
  104|   _xd.copy(_xk).sub(hipLocal).normalize();
  105|   i.hip.quaternion.setFromUnitVectors(_xu, _xd);
  106| 
  107|   // knee: same, in the hip's frame, aiming the shin at the target
  108|   _xh.copy(i.hip.quaternion).invert();
  109|   _xd.copy(targetLocal).sub(hipLocal).applyQuaternion(_xh).sub(i.knee.position).normalize();
  110|   _xu.copy(i.ankle.position).normalize();
  111|   i.knee.quaternion.setFromUnitVectors(_xu, _xd);
  112| 
  113|   // ankle: foot flat on the ground, facing the creature's heading when given
  114|   _xh.copy(pw).multiply(i.hip.quaternion).multiply(i.knee.quaternion).invert();
  115|   if (i.footWorld) _xh.multiply(i.footWorld);
  116|   i.ankle.quaternion.copy(_xh);
  117| 
  118|   if (i.prevKnee) i.prevKnee.copy(_xk);
  119|   return ikLast.fallback;
  120| }

  122| /**
  123|  * Solve one leg. The hip's position is never touched; only the hip and knee
  124|  * rotations move, and the ankle is levelled to the world so the foot stays flat.
  125|  * Returns true when the pole grazed the limb axis (forensics).
  126|  */
  127| export function solveLeg(i: LegSolveInput): boolean {
  128|   const hipParent = i.hip.parent;
  129|   if (!hipParent) return false;
  130|   if (i.exact) return solveLegExact(i, hipParent);
  131|   hipParent.updateMatrixWorld(true);
  132| 
  133|   const pw = new THREE.Quaternion();
  134|   hipParent.getWorldQuaternion(pw);
  135|   const pwInv = _inv.copy(pw).invert();
  136| 
  137|   const targetLocal = _tgt.copy(i.targetWorld);
  138|   hipParent.worldToLocal(targetLocal);
  139| 
  140|   const hipLocal = _root.copy(i.hip.position);
  141|   const poleLocal = _pole.copy(i.poleWorld).applyQuaternion(pwInv).normalize();
  142| 
  143|   // Segment lengths come straight off the bone rest offsets.
  144|   const L1 = _lenA.copy(i.knee.position).length();
  145|   const L2 = _lenB.copy(i.ankle.position).length();
  146|   solveIK2(hipLocal, targetLocal, L1, L2, poleLocal, _knee, i.prevKnee);
  147| 
  148|   // hip: rotate its rest child direction onto the solved knee direction
  149|   const kneeDirWorld = _v.copy(_knee).sub(hipLocal).applyQuaternion(pw).normalize();
  150|   aimBoneTowards(i.hip, i.knee.position, kneeDirWorld, pw);
  151| 
  152|   // knee: find where it actually landed, then aim its child at the plant
  153|   const hipWorldQuat = new THREE.Quaternion().multiplyQuaternions(pw, i.hip.quaternion);
  154|   const hipWorldPos = _hipW.copy(hipLocal).applyQuaternion(pw).add(hipParent.getWorldPosition(_pos));
  155|   const kneeWorld = new THREE.Vector3().copy(i.knee.position).applyQuaternion(hipWorldQuat).add(hipWorldPos);
  156|   const shinDirWorld = _v.copy(i.targetWorld).sub(kneeWorld).normalize();
  157|   aimBoneTowards(i.knee, i.ankle.position, shinDirWorld, hipWorldQuat);
  158| 
  159|   // keep the shoe flat on the ground: cancel the shin's rotation
  160|   const shinWorldQuat = new THREE.Quaternion().multiplyQuaternions(hipWorldQuat, i.knee.quaternion);
  161|   i.ankle.quaternion.copy(shinWorldQuat).invert();
  162| 
  163|   // remember this frame's knee so tomorrow's solve stays on the same branch
  164|   if (i.prevKnee) i.prevKnee.copy(_knee);
  165|   return false;
  166| }

  168| /** A soft, critically damped 1D spring used for ears, tails and antennae. */
  169| export class Wobble {
  170|   v = 0;
  171|   x = 0;
  172|   step(drive: number, dt: number, k = 42, c = 7): number {
  173|     const a = -k * this.x - c * this.v + drive;
  174|     this.v += a * dt;
  175|     this.x += this.v * dt;
  176|     if (this.x > 0.5) { this.x = 0.5; this.v *= 0.4; }
  177|     if (this.x < -0.5) { this.x = -0.5; this.v *= 0.4; }
  178|     return this.x;
  179|   }
  180| }
