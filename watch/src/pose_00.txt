    1| /**
    2|  * Pose math. Rotation-only bone work plus the 2-bone leg IK that plants feet.
    3|  *
    4|  * Everything is expressed in creature space and then converted through the
    5|  * parent bone's world matrix, so a limb can never stretch, scale, or translate
    6|  * away from its socket: the socket IS the bone origin.
    7|  *
    8|  * The aim step is child-offset aware: a splayed insect coxa whose child sits
    9|  * sideways off the bone axis still lands its knee exactly on the solved joint.
   10|  * Assuming "the child is on local -Y" is what made the ZIK feet miss their
   11|  * plant targets by ~0.19 units.
   12|  */
   13| import * as THREE from 'three';
   14| import { solveIK2 } from './ik';
   15| 
   16| /** The conventional "down" limb axis, kept for callers that use it as a hint. */
   17| export const LIMB_AXIS = new THREE.Vector3(0, -1, 0);
   18| 
   19| const _q = new THREE.Quaternion();
   20| const _inv = new THREE.Quaternion();
   21| const _d = new THREE.Vector3();
   22| const _u = new THREE.Vector3();
   23| const _v = new THREE.Vector3();
   24| const _root = new THREE.Vector3();
   25| const _tgt = new THREE.Vector3();
   26| const _knee = new THREE.Vector3();
   27| const _pole = new THREE.Vector3();
   28| const _hipW = new THREE.Vector3();
   29| const _pos = new THREE.Vector3();
   30| const _lenA = new THREE.Vector3();
   31| const _lenB = new THREE.Vector3();
   32| 
   33| /**
   34|  * Point a bone so the direction of its CHILD, expressed in the bone's own
   35|  * local space, runs along `dirWorld`.
   36|  */
   37| export function aimBoneTowards(
   38|   bone: THREE.Bone,
   39|   childLocalDir: THREE.Vector3,
   40|   dirWorld: THREE.Vector3,
   41|   parentWorldQuat: THREE.Quaternion,
   42| ): void {
   43|   const u = _u.copy(childLocalDir);
   44|   if (u.lengthSq() < 1e-12) u.set(0, -1, 0);
   45|   u.normalize();
   46|   const d = _d.copy(dirWorld);
   47|   if (d.lengthSq() < 1e-12) d.set(0, -1, 0);
   48|   d.normalize();
   49|   _q.setFromUnitVectors(u, d);
   50|   bone.quaternion.copy(parentWorldQuat).invert().multiply(_q);
   51| }
   52| 
   53| /** Aim a limb whose child really does sit on local -Y. */
   54| export function aimBone(bone: THREE.Bone, dirWorld: THREE.Vector3, parentWorldQuat: THREE.Quaternion): void {
   55|   aimBoneTowards(bone, LIMB_AXIS, dirWorld, parentWorldQuat);
   56| }

   58| export interface LegSolveInput {
   59|   hip: THREE.Bone;
   60|   knee: THREE.Bone;
   61|   ankle: THREE.Bone;
   62|   /** optional: segment lengths are read from the bones, so callers cannot lie */
   63|   l1?: number;
   64|   l2?: number;
   65|   /** desired world position of the ankle */
   66|   targetWorld: THREE.Vector3;
   67|   /** world-space pole hint (roughly forward) */
   68|   poleWorld: THREE.Vector3;
   69|   /** previous knee position in the hip-parent's local space. Read to keep the
   70|    *  solve on the same branch, then written back with the new knee. */
   71|   prevKnee?: THREE.Vector3;
   72| }
   73| 
   74| /**
   75|  * Solve one leg. The hip's position is never touched; only the hip and knee
   76|  * rotations move, and the ankle is levelled to the world so the foot stays flat.
   77|  */
   78| export function solveLeg(i: LegSolveInput): void {
   79|   const hipParent = i.hip.parent;
   80|   if (!hipParent) return;
   81|   hipParent.updateMatrixWorld(true);
   82| 
   83|   const pw = new THREE.Quaternion();
   84|   hipParent.getWorldQuaternion(pw);
   85|   const pwInv = _inv.copy(pw).invert();
   86| 
   87|   const targetLocal = _tgt.copy(i.targetWorld);
   88|   hipParent.worldToLocal(targetLocal);
   89| 
   90|   const hipLocal = _root.copy(i.hip.position);
   91|   const poleLocal = _pole.copy(i.poleWorld).applyQuaternion(pwInv).normalize();
   92| 
   93|   // Segment lengths come straight off the bone rest offsets. Deriving them here
   94|   // means no caller can pass a length measured in the wrong space.
   95|   const L1 = _lenA.copy(i.knee.position).length();
   96|   const L2 = _lenB.copy(i.ankle.position).length();
   97|   solveIK2(hipLocal, targetLocal, L1, L2, poleLocal, _knee, i.prevKnee);
   98| 
   99|   // hip: rotate its rest child direction onto the solved knee direction
  100|   const kneeDirWorld = _v.copy(_knee).sub(hipLocal).applyQuaternion(pw).normalize();
  101|   aimBoneTowards(i.hip, i.knee.position, kneeDirWorld, pw);
  102| 
  103|   // knee: find where it actually landed, then aim its child at the plant
  104|   const hipWorldQuat = new THREE.Quaternion().multiplyQuaternions(pw, i.hip.quaternion);
  105|   const hipWorldPos = _hipW.copy(hipLocal).applyQuaternion(pw).add(hipParent.getWorldPosition(_pos));
  106|   const kneeWorld = new THREE.Vector3().copy(i.knee.position).applyQuaternion(hipWorldQuat).add(hipWorldPos);
  107|   const shinDirWorld = _v.copy(i.targetWorld).sub(kneeWorld).normalize();
  108|   aimBoneTowards(i.knee, i.ankle.position, shinDirWorld, hipWorldQuat);
  109| 
  110|   // keep the shoe flat on the ground: cancel the shin's rotation
  111|   const shinWorldQuat = new THREE.Quaternion().multiplyQuaternions(hipWorldQuat, i.knee.quaternion);
  112|   i.ankle.quaternion.copy(shinWorldQuat).invert();
  113| 
  114|   // remember this frame's knee so tomorrow's solve stays on the same branch
  115|   if (i.prevKnee) i.prevKnee.copy(_knee);
  116| }

  118| /** A soft, critically damped 1D spring used for ears, tails and antennae. */
  119| export class Wobble {
  120|   v = 0;
  121|   x = 0;
  122|   step(drive: number, dt: number, k = 42, c = 7): number {
  123|     const a = -k * this.x - c * this.v + drive;
  124|     this.v += a * dt;
  125|     this.x += this.v * dt;
  126|     if (this.x > 0.5) { this.x = 0.5; this.v *= 0.4; }
  127|     if (this.x < -0.5) { this.x = -0.5; this.v *= 0.4; }
  128|     return this.x;
  129|   }
  130| }
