  432|   /** Eats one ripe berry. Returns false when the bush is bare. */
  433|   eat(bushIndex: number): boolean {
  434|     const bush = this.bushes[bushIndex];
  435|     if (!bush) return false;
  436|     const i = bush.ripe.findIndex((v) => v === 0);
  437|     if (i < 0) return false;
  438|     bush.ripe[i] = 26 + Math.random() * 20; // seconds to regrow
  439|     bush.shake = 1;
  440|     this.refreshBerries();
  441|     return true;
  442|   }

  448|   nearestFruitingBush(p: THREE.Vector3, maxDist: number): number {
  449|     let best = -1;
  450|     let bestD = maxDist;
  451|     for (let i = 0; i < this.bushes.length; i++) {
  452|       if (!this.bushHasFruit(i)) continue;
  453|       const d = this.bushes[i].pos.distanceTo(p);
  454|       if (d < bestD) { bestD = d; best = i; }
  455|     }
  456|     return best;
  457|   }

  476|     const d = this.ball.pos.clone().sub(from).setY(0);

  485|   update(dt: number): void {
  486|     // berries regrow
  487|     let dirty = false;
  488|     for (const bush of this.bushes) {
  489|       for (let i = 0; i < bush.ripe.length; i++) {
  490|         if (bush.ripe[i] > 0) {
  491|           bush.ripe[i] -= dt;
  492|           if (bush.ripe[i] <= 0) { bush.ripe[i] = 0; dirty = true; }
  493|         }
  494|       }
  495|     }
  496|     if (dirty) this.refreshBerries();
  497| 
  498|     // a picked bush wobbles, so you can see which one gave up its fruit
  499|     for (let i = 0; i < this.bushes.length; i++) {
  500|       const bush = this.bushes[i];
