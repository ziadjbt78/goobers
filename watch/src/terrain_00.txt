    1| /**
    2|  * The island. One analytic height field drives everything: the mesh you see,
    3|  * where the feet plant, where the grass grows and how high the water sits.
    4|  * There is no heightmap to load and no collision mesh to keep in sync.
    5|  *
    6|  * The shoreline is a smoothstep shelf rather than a quadratic cliff, so the
    7|  * island reads as a rounded toon landmass with a gently curving beach instead
    8|  * of a blocky stepped rim.
    9|  */
   10| import * as THREE from 'three';
   11| import { MAT } from '../render/hero/materials';
   12| 
   13| export const ISLAND_R = 12.5;
   14| export const POND = { x: 4.6, z: -3.8, r: 3.1 };
   15| /** Water plane. Anything below it is the pond and the sea. */
   16| export const WATER_Y = 0.17;
   17| /** Creatures stay on the meadow and out of the sea. */
   18| export const WALK_R = 10.2;
   19| 
   20| const smoothstep = (a: number, b: number, x: number): number => {
   21|   const t = Math.min(1, Math.max(0, (x - a) / (b - a)));
   22|   return t * t * (3 - 2 * t);
   23| };
   24| 
   25| /** Ground height at a world point. Pure, cheap, called thousands of times a frame. */
   26| export function heightAt(x: number, z: number): number {
   27|   const r = Math.hypot(x, z);
   28|   const dome = 0.62 * (1 - (r / ISLAND_R) ** 2);
   29|   const hills = 0.30 * Math.sin(x * 0.44 + 0.7) * Math.cos(z * 0.39 - 0.3)
   30|     + 0.14 * Math.sin(x * 0.97 - 1.1) * Math.cos(z * 0.86 + 0.5);
   31|   const inland = smoothstep(ISLAND_R, ISLAND_R * 0.55, r);
   32|   let h = dome + hills * inland - 0.16;
   33|   const dp = Math.hypot(x - POND.x, z - POND.z);
   34|   h -= smoothstep(POND.r, 0, dp) * 0.62;
   35|   // smooth, C1 shoreline: one eased shelf out to the sea floor
   36|   const edge = smoothstep(ISLAND_R * 0.78, ISLAND_R * 1.34, r);
   37|   h -= edge * 5.4;
   38|   return h;
   39| }
   40| 
   41| export function normalAt(x: number, z: number, out = new THREE.Vector3()): THREE.Vector3 {
   42|   const e = 0.08;
   43|   const hx = heightAt(x + e, z) - heightAt(x - e, z);
   44|   const hz = heightAt(x, z + e) - heightAt(x, z - e);
   45|   return out.set(-hx, 2 * e, -hz).normalize();
   46| }
   47| 
   48| export function isDry(x: number, z: number): boolean {
   49|   return Math.hypot(x, z) < WALK_R && heightAt(x, z) > WATER_Y + 0.04;
   50| }
   51| 
   52| /** The island mesh: one displaced plane with graded vertex colours. */
   53| export function buildTerrain(): THREE.Mesh {
   54|   const SIZE = 46;
   55|   const SEG = 160;
   56|   const geo = new THREE.PlaneGeometry(SIZE, SIZE, SEG, SEG);
   57|   geo.rotateX(-Math.PI / 2);
   58|   const pos = geo.getAttribute('position') as THREE.BufferAttribute;
   59|   for (let i = 0; i < pos.count; i++) pos.setY(i, heightAt(pos.getX(i), pos.getZ(i)));
   60|   geo.computeVertexNormals();
   61| 
   62|   const grass = new THREE.Color('#8ed05a');
   63|   const grassDark = new THREE.Color('#5aa53f');
   64|   const sand = new THREE.Color('#f2e3b0');
   65|   const mud = new THREE.Color('#c9b078');
   66|   const reef = new THREE.Color('#79c6c0');
   67|   const underwater = new THREE.Color('#4f9fb0');
   68|   const col = new Float32Array(pos.count * 3);
   69|   const c = new THREE.Color();
   70|   for (let i = 0; i < pos.count; i++) {
   71|     const x = pos.getX(i);
   72|     const z = pos.getZ(i);
   73|     const y = pos.getY(i);
   74|     const r = Math.hypot(x, z);
   75|     if (y < WATER_Y - 0.55) c.copy(underwater).lerp(reef, smoothstep(-2.4, WATER_Y - 0.55, y));
   76|     else if (y < WATER_Y - 0.02) c.copy(reef).lerp(mud, smoothstep(WATER_Y - 0.55, WATER_Y, y));
   77|     else c.copy(sand).lerp(grass, smoothstep(WATER_Y, WATER_Y + 0.26, y));
   78|     if (y > WATER_Y + 0.24) c.lerp(grassDark, smoothstep(WATER_Y + 0.24, 1.05, y) * 0.40);
   79|     const n = 0.5 + 0.5 * Math.sin(x * 3.1 + z * 2.7) * Math.sin(x * 1.7 - z * 3.3);
   80|     c.offsetHSL(0, 0, (n - 0.5) * 0.038);
   81|     c.offsetHSL(0, 0, -smoothstep(WALK_R * 0.8, ISLAND_R, r) * 0.04);
   82|     col[i * 3] = c.r; col[i * 3 + 1] = c.g; col[i * 3 + 2] = c.b;
   83|   }
   84|   geo.setAttribute('color', new THREE.BufferAttribute(col, 3));
   85|   const mesh = new THREE.Mesh(geo, MAT.lit);
   86|   mesh.name = 'terrain';
   87|   mesh.frustumCulled = false;
   88|   return mesh;
   89| }
   90| 
   91| /**
   92|  * The sea: one big instanced-free disc at the waterline, toon-graded from a
   93|  * pale turquoise lagoon to a deep blue horizon, with a slow vertex ripple so it
   94|  * never looks like a painted plane. It replaces the old flat brown outer ring.
   95|  */
   96| export function buildSea(): THREE.Mesh {
   97|   const R = 260;
   98|   const geo = new THREE.CircleGeometry(R, 128, 1);
   99|   geo.rotateX(-Math.PI / 2);
  100|   const p = geo.getAttribute('position') as THREE.BufferAttribute;
  101|   const near = new THREE.Color('#8fdccb');
  102|   const mid = new THREE.Color('#4fb0d8');
  103|   const far = new THREE.Color('#2f6fb0');
  104|   const col = new Float32Array(p.count * 3);
  105|   const c = new THREE.Color();
  106|   for (let i = 0; i < p.count; i++) {
  107|     const r = Math.hypot(p.getX(i), p.getZ(i));
  108|     if (r < ISLAND_R * 1.15) c.copy(near);
  109|     else if (r < 60) c.copy(near).lerp(mid, smoothstep(ISLAND_R * 1.1, 60, r));
  110|     else c.copy(mid).lerp(far, smoothstep(60, 200, r));
  111|     col[i * 3] = c.r; col[i * 3 + 1] = c.g; col[i * 3 + 2] = c.b;
  112|   }
  113|   geo.setAttribute('color', new THREE.BufferAttribute(col, 3));
  114|   const mesh = new THREE.Mesh(geo, MAT.lit);
  115|   mesh.name = 'sea';
  116|   mesh.position.y = WATER_Y - 0.005;
  117|   mesh.frustumCulled = false;
  118|   mesh.renderOrder = -2;
  119|   return mesh;
  120| }

  122| /** A soft foam ring hugging the island where the land meets the waterline. */
  123| export function buildFoam(): THREE.Mesh {
  124|   let waterR = ISLAND_R;
  125|   for (let i = 0; i <= 200; i++) {
  126|     const r = ISLAND_R * 0.6 + (i / 200) * ISLAND_R * 0.8;
  127|     if (heightAt(r, 0) > WATER_Y) waterR = r;
  128|   }
  129|   const inner = Math.max(ISLAND_R * 0.7, waterR - 0.40);
  130|   const geo = new THREE.RingGeometry(inner, waterR + 0.26, 96, 1);
  131|   geo.rotateX(-Math.PI / 2);
  132|   const p = geo.getAttribute('position') as THREE.BufferAttribute;
  133|   const col = new Float32Array(p.count * 3);
  134|   const white = new THREE.Color('#ffffff');
  135|   const soft = new THREE.Color('#cdeef0');
  136|   const c = new THREE.Color();
  137|   for (let i = 0; i < p.count; i++) {
  138|     const r = Math.hypot(p.getX(i), p.getZ(i));
  139|     c.copy(soft).lerp(white, smoothstep(inner, waterR, r));
  140|     col[i * 3] = c.r; col[i * 3 + 1] = c.g; col[i * 3 + 2] = c.b;
  141|   }
  142|   geo.setAttribute('color', new THREE.BufferAttribute(col, 3));
  143|   const mesh = new THREE.Mesh(geo, MAT.lit);
  144|   mesh.name = 'foam';
  145|   mesh.position.y = WATER_Y + 0.012;
  146|   mesh.frustumCulled = false;
  147|   mesh.renderOrder = -1;
  148|   return mesh;
  149| }
